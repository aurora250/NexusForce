if(NOT NEXUSFORCE_BUILD_TESTS)
    message(STATUS "Skipping tests build (NEXUSFORCE_BUILD_TESTS=OFF)")
    return()
endif()

# A googletest source tree can replace the installed package.
# On Windows this is the way to give the test binary a single allocator,
# which the allocator override mode requires and which Linux already gets from its statically linked gtest.
set(NEXUSFORCE_GOOGLETEST_SOURCE_DIR "" CACHE PATH "googletest source tree used to build the tests")
if(NEXUSFORCE_GOOGLETEST_SOURCE_DIR AND EXISTS "${NEXUSFORCE_GOOGLETEST_SOURCE_DIR}/CMakeLists.txt")
    set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
    set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    add_subdirectory("${NEXUSFORCE_GOOGLETEST_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/googletest" EXCLUDE_FROM_ALL)
    message(STATUS "GTest is built from source: ${NEXUSFORCE_GOOGLETEST_SOURCE_DIR}")
else()
    find_package(GTest CONFIG QUIET)
endif()
if(NOT TARGET GTest::gtest)
    message(WARNING "GTest not found — skipping test build. Install GTest or enable the 'gtest' vcpkg feature.")
    return()
endif()

message(STATUS "GTest found — building tests")

set(NEXUSFORCE_GTEST_IS_SHARED OFF)
foreach(_nexusforce_gtest_config "" _DEBUG _RELEASE _RELWITHDEBINFO _MINSIZEREL _NOCONFIG)
    get_target_property(_nexusforce_gtest_location GTest::gtest IMPORTED_LOCATION${_nexusforce_gtest_config})
    if(_nexusforce_gtest_location AND _nexusforce_gtest_location MATCHES "\\.dll$")
        set(NEXUSFORCE_GTEST_IS_SHARED ON)
    endif()
endforeach()
unset(_nexusforce_gtest_location)
unset(_nexusforce_gtest_config)
if(NEXUSFORCE_GTEST_IS_SHARED)
    message(STATUS "GTest is a shared library; keep the allocator override enabled only if every C++ module uses it")
endif()

add_subdirectory(unit)
add_subdirectory(integration)
