68 static NEFORCE_CONSTEXPR20 value_type parse_view(const string_view view) {
69 if (view.empty()) return 0;
70
71 bool negative = false;
72 size_t start = 0;
73
74 while (start < view.size() && is_space(view[start])) ++start;
75 if (start == view.size()) return 0;
76
77 if (view[start] == '-') {
78 negative = true;
79 ++start;
80 } else if (view[start] == '+') {
81 ++start;
82 }
83
84 if (start + 1 < view.size() && view[start] == '0' &&
85 (view[start+1] == 'x' || view[start+1] == 'X')) {
86 start += 2;
87 }
88
89 uint64_t result = 0;
90 while (start < view.size()) {
91 const char c = view[start++];
92 if (is_xdigit(c)) {
93 const int digit = xdigit_value(c);
94 if (result > (numeric_traits<uint64_t>::max() >> 4)) {
95 NEFORCE_THROW_EXCEPTION(value_exception("Hexadecimal value too large"));
96 }
97 result = (result << 4) | static_cast<uint64_t>(digit);
98 } else if (!is_space(c)) {
99 NEFORCE_THROW_EXCEPTION(value_exception("Invalid hexadecimal character"));
100 }
101 }
102
103 if (negative) {
104 if (result > static_cast<uint64_t>(numeric_traits<int64_t>::max()) + 1) {
105 NEFORCE_THROW_EXCEPTION(value_exception("Hexadecimal value out of range"));
106 }
107 return -static_cast<int64_t>(result);
108 }
109 if (result > static_cast<uint64_t>(numeric_traits<int64_t>::max())) {
110 NEFORCE_THROW_EXCEPTION(value_exception("Hexadecimal value out of range"));
111 }
112 return static_cast<value_type>(result);
113 }
