337 buckets_.clear();
338 buckets_.reserve(other.buckets_.size());
339 buckets_.insert(buckets_.end(), other.buckets_.size(), nullptr);

363 pair<iterator, bool> insert_unique_noresize(link_type ptr) {
364 const size_type n = hashtable::bucket_index_value(ptr->data, buckets_.size());
365
366 node_type** buckets_ptr = &buckets_[n];
367 while (*buckets_ptr != nullptr) {
368 if (equals_(extracter_((*buckets_ptr)->data), extracter_(ptr->data))) {
369 ptr->next = (*buckets_ptr)->next;
370 hashtable::delete_node(*buckets_ptr);
371 *buckets_ptr = ptr;
372 return {{ptr, n, this}, false};
373 }
374 buckets_ptr = &(*buckets_ptr)->next;
375 }
376 ptr->next = nullptr;
377 *buckets_ptr = ptr;
378 ++size_;
379 return {iterator{ptr, n, this}, true};
380 }

387 iterator insert_equal_noresize(link_type ptr) {
388 const size_type n = hashtable::bucket_index_value(ptr->data, buckets_.size());
389 link_type first = buckets_[n];
390
391 link_type prev = nullptr;
392 link_type cur = first;
393 while (cur != nullptr &&
394 equals_(extracter_(cur->data), extracter_(ptr->data))) {
395 prev = cur;
396 cur = cur->next;
397 }
398
399 if (prev != nullptr) {
400 prev->next = ptr;
401 ptr->next = cur;
402 } else {
403 ptr->next = first;
404 buckets_[n] = ptr;
405 }
406
407 ++size_;
408 return {ptr, n, this};
409 }

512 size_type erase_bucket_range(size_type bucket, link_type first, link_type last) noexcept {
513 size_type count = 0;
514 if (first == nullptr) return 0;
515
516 if (buckets_[bucket] == first) {
517 count += hashtable::erase_bucket_to_node(bucket, last);
518 } else {

560 bool equal_small(const hashtable& rhs) const {
561 for (const_iterator iter = begin(); iter != end(); ++iter) {
562 const key_type& key = extracter_(*iter);
563
564 const size_t count_lhs = _NEFORCE count_if(begin(), end(),
565 [this, &key](const value_type& val) {
566 return equals_(extracter_(val), key);
567 });
568 const size_t count_rhs = _NEFORCE count_if(rhs.begin(), rhs.end(),
569 [&rhs, &key](const value_type& val) {
570 return rhs.equals_(rhs.extracter_(val), key);
571 });
572
573 if (count_lhs != count_rhs) {
574 return false;
575 }
576 }
577 return true;
578 }

585 bool equal_large(const hashtable& rhs) const {
586 vector<value_type> elements_lhs, elements_rhs;
587 elements_lhs.reserve(size_);
588 elements_rhs.reserve(size_);
589
590 for (const_iterator it = begin(); it != end(); ++it) {
591 elements_lhs.push_back(*it);
592 }
593 for (const_iterator it = rhs.begin(); it != rhs.end(); ++it) {
594 elements_rhs.push_back(*it);
595 }
596
597 _NEFORCE sort(elements_lhs.begin(), elements_lhs.end());
598 _NEFORCE sort(elements_rhs.begin(), elements_rhs.end());
599
600 return elements_lhs == elements_rhs;
601 }

876 void rehash(const size_type new_size) {
877 const auto min_buckets_for_size = static_cast<size_type>(
878 _NEFORCE ceil(static_cast<double>(size_) / max_load_factor()));
879 const size_type target = _NEFORCE max(new_size, min_buckets_for_size);
880 const size_type old_size = buckets_.size();
881
882 if (target <= old_size) return;
883
884 const size_type n = hashtable::next_size(target);
885 if (n < target) {
886 NEFORCE_THROW_EXCEPTION(value_exception("hashtable size exceeds max count"));
887 }
888
889 vector<link_type> new_buckets(n, nullptr);
890
891 for (size_type bucket = 0; bucket < old_size; ++bucket) {
892 link_type cur = buckets_[bucket];
893 while (cur != nullptr) {
894 link_type next = cur->next;
895 const size_type new_bucket = hashtable::bucket_index_value(cur->data, n);
896
897 cur->next = new_buckets[new_bucket];
898 new_buckets[new_bucket] = cur;
899
900 cur = next;
901 }
902 buckets_[bucket] = nullptr;
903 }
904
905 buckets_.swap(new_buckets);
906 }

1055 --size_;

1079 noexcept(is_nothrow_hashable_v<key_type>) {
1080 if (position.current_ == nullptr || position.container_ != this) {
1081 return hashtable::end();
1082 }

1084 const size_type n = position.bucket_;
1085 link_type const p = position.current_;
1086 link_type next_node = p->next;
1087
1088 link_type prev = nullptr;
1089 link_type curr = buckets_[n];
1090 while (curr != nullptr && curr != p) {
1091 prev = curr;
1092 curr = curr->next;
1093 }
1094
1095 if (curr == nullptr) {
1096 return hashtable::end();
1097 }
1098
1099 if (prev == nullptr) {
1100 buckets_[n] = next_node;
1101 } else {
1102 prev->next = next_node;
1103 }
1104
1105 hashtable::delete_node(p);
1106 --size_;
1107
1108 if (next_node != nullptr) {
1109 return iterator(next_node, n, this);
1110 }
1111
1112 for (size_type bucket = n + 1; bucket < buckets_.size(); ++bucket) {
1113 if (buckets_[bucket] != nullptr) {
1114 return iterator(buckets_[bucket], bucket, this);
1115 }
1116 }

1131 return hashtable::end();
