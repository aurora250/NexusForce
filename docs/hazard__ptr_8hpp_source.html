181 vector<void*> get_hazard_pointers() {
182 vector<void*> hazards;
183 hazard_pointer_record* current = head_.load(memory_order_acquire);
184
185 while (current) {
186 if (current->active.load(memory_order_acquire)) {
187 void* ptr = current->get_protected();
188 if (ptr) {
189 hazards.push_back(ptr);
190 }
191 }
192 current = current->next.load(memory_order_acquire);
193 }
194
195 return hazards;

204 void scan_and_reclaim() {
205 auto hazards = get_hazard_pointers();
206 _NEFORCE sort(hazards.begin(), hazards.end());
207
208 retire_list new_list;
209 hazard_pointer_obj_base* current = tl_retire_list_.head;

211 while (current) {
212 auto* next = current->next;
213
214 if (_NEFORCE binary_search(hazards.begin(), hazards.end(), current)) {
215 new_list.add(current);

217 current->destroy();
218 delete current;
219 }

253 hazard_pointer_record* acquire_record() {
254 hazard_pointer_record* current = head_.load(memory_order_acquire);
255 while (current) {
256 if (current->try_acquire()) {
257 return current;
258 }
259 current = current->next.load(memory_order_acquire);
260 }
261
262 hazard_pointer_record* new_record = new hazard_pointer_record();
263 new_record->active.store(true, memory_order_relaxed);
264
265 hazard_pointer_record* old_head = head_.load(memory_order_relaxed);
266 do {
267 new_record->next.store(old_head, memory_order_relaxed);
268 } while (!head_.compare_exchange_weak(
269 old_head, new_record,
270 memory_order_release, memory_order_relaxed));
