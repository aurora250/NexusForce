60template <typename Dur>
61bool __platform_wait_until_impl(const platform_wait_t* addr, const platform_wait_t old,
62 const time_point<wait_clock_t, Dur>& timeout) noexcept {
63 const bool has_timeout = (timeout != wait_clock_t::time_point::max());
64
65 if (!has_timeout) {
66 futex_wait(const_cast<void*>(static_cast<const void*>(addr)), old);
67 return true;
68 }
69
70 const auto now = wait_clock_t::now();
71 if (timeout <= now) {
72 return false;
73 }
74
75#ifdef NEFORCE_PLATFORM_WINDOWS
76 const auto dur = timeout - now;
77 const auto sec = time_cast<seconds>(dur);
78 const auto ns = time_cast<nanoseconds>(dur - sec);
79#else
80 const auto sys_timeout = steady_clock::to_system<Dur>(timeout);
81 const auto sys_now = system_clock::now();
82 if (sys_timeout <= sys_now) {
83 return false;
84 }
85 const auto sys_dur = sys_timeout - sys_now;
86 const auto sec = time_cast<seconds>(sys_dur);
87 const auto ns = time_cast<nanoseconds>(sys_dur - sec);
88#endif
89
90 return _NEFORCE futex_wait_until(const_cast<void*>(static_cast<const void*>(addr)), old, true, sec.count(),
91 ns.count(), true);
92}

155struct timed_backoff_spin_policy {
156 inner::wait_clock_t::time_point deadline;
157 inner::wait_clock_t::time_point start_time;
158
159 template <typename Clock, typename Dur>
160 timed_backoff_spin_policy(time_point<Clock, Dur> deadline_time = Clock::time_point::max(),
161 time_point<Clock, Dur> start_time_point = Clock::now()) noexcept :
162 deadline(inner::to_wait_clock(deadline_time)),
163 start_time(inner::to_wait_clock(start_time_point)) {}
164
165 bool operator()() const noexcept {
166 const auto now = inner::wait_clock_t::now();
167 if (deadline <= now) {
168 return false;
169 }
170
171 const auto elapsed = now - start_time;
172 if (elapsed > 128_ms) {
173 this_thread::sleep_for(64_ms);
174 } else if (elapsed > 64_us) {
175 this_thread::sleep_for(elapsed / 2);
176 } else if (elapsed > 4_us) {
177 this_thread::yield();
178 } else {
179 return false;
180 }
181 return true;
182 }
183};
