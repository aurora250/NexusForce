172 vector<void*> get_hazard_pointers() {

195 void scan_and_reclaim() {
196 auto hazards = get_hazard_pointers();
197 _NEFORCE sort(hazards.begin(), hazards.end());

199 retire_list new_list;

203 auto* next = current->next;

205 if (_NEFORCE binary_search(hazards.begin(), hazards.end(), current)) {

244 hazard_pointer_record* acquire_record() {
245 hazard_pointer_record* current = head_.load(memory_order_acquire);
246 while (current) {
247 if (current->try_acquire()) {
248 return current;
249 }
250 current = current->next.load(memory_order_acquire);
251 }
252
253 hazard_pointer_record* new_record = new hazard_pointer_record();
254 new_record->active.store(true, memory_order_relaxed);
255
256 hazard_pointer_record* old_head = head_.load(memory_order_relaxed);
257 do {
258 new_record->next.store(old_head, memory_order_relaxed);
259 } while (!head_.compare_exchange_weak(old_head, new_record, memory_order_release, memory_order_relaxed));
