92template <typename Iterator, typename T, typename Compare>
93constexpr pair<Iterator, Iterator> equal_range(Iterator first, Iterator last, const T& value, Compare comp) {
94 static_assert(is_ranges_fwd_iter_v<Iterator>, "Iterator must be forward_iterator");
95
96 auto len = _NEFORCE distance(first, last);
97 auto half = len;
98 Iterator middle, left, right;
99
100 while (len > 0) {
101 half = len >> 1;
102 middle = first;
103 _NEFORCE advance(middle, half);
104
105 if (comp(*middle, value)) {
106 first = middle;
107 ++first;
108 len = len - half - 1;
109 } else if (comp(value, *middle)) {
110 len = half;
111 } else {
112 left = _NEFORCE lower_bound(first, middle, value, comp);
113 _NEFORCE advance(first, len);
114 right = _NEFORCE upper_bound(++middle, first, value, comp);
115 return pair<Iterator, Iterator>(left, right);
116 }
117 }
118 return pair<Iterator, Iterator>(first, first);
119}

259template <typename Iterator, typename Compare>
260pair<iter_value_t<Iterator>, iter_value_t<Iterator>>
261constexpr minmax(Iterator first, Iterator last, Compare comp) {
262 static_assert(is_ranges_input_iter_v<Iterator>, "Iterator must be input_iterator");
263 using T = iter_value_t<Iterator>;
264 if (first == last) {
265 return _NEFORCE make_pair(T(), T());
266 }
267 T min_val = *first;
268 T max_val = *first;
269
270 ++first;
271 for (; first != last; ++first) {
272 if (comp(*first, min_val))
273 min_val = *first;
274 else if (!comp(*first, max_val))
275 max_val = *first;
276 }
277 return _NEFORCE make_pair(min_val, max_val);
278}
