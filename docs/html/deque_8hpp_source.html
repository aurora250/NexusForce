147 NEFORCE_NODISCARD reference dereference() const noexcept {
148 NEFORCE_DEBUG_VERIFY(current_ && container_, "Attempting to dereference on a null pointer");
149 NEFORCE_DEBUG_VERIFY(first_ <= current_ && current_ < last_, "Attempting to dereference out of boundary");
150 return *current_;

158 void increment() noexcept {
159 NEFORCE_DEBUG_VERIFY(current_ && container_, "Attempting to increment a null pointer");
160 ++current_;
161 if (current_ == last_) {
162 deque_iterator::change_buff(node_ + 1);
163 current_ = first_;
164 }

172 void decrement() noexcept {
173 NEFORCE_DEBUG_VERIFY(current_ && container_, "Attempting to decrement a null pointer");
174 if (current_ == first_) {
175 deque_iterator::change_buff(node_ - 1);
176 current_ = last_;
177 }
178 --current_;
179 }

187 void advance(difference_type off) noexcept {
188 NEFORCE_DEBUG_VERIFY((current_ && container_) || off == 0, "Attempting to advance a null pointer");
189 const difference_type offset = off + (current_ - first_);
190 if (offset >= 0 && offset < buffer_size) {
191 current_ += off;
192 } else {
193 difference_type node_offset =
194 offset > 0 ? offset / buffer_size : -static_cast<difference_type>((-offset - 1) / buffer_size) - 1;
195 deque_iterator::change_buff(node_ + node_offset);
196 current_ = first_ + (offset - node_offset * buffer_size);
197 }
198 }

362 void create_map_and_nodes(const size_type n) {
363 size_type node_nums = n / buffer_size + (n % buffer_size ? 1 : 0);
364 map_size_pair_.value = _NEFORCE max(init_map_size, node_nums + 2);
365
366 try {
367 map_pair_.value = deque::create_map(map_size_pair_.value);
368 } catch (...) {
369 map_pair_.value = nullptr;
370 map_size_pair_.value = 0;
371 throw;
372 }
373
374 map_pointer nstart = map_pair_.value + (map_size_pair_.value - node_nums) / 2;
375 map_pointer nfinish = node_nums == 0 ? nstart : nstart + node_nums - 1;
376
377 try {
378 deque::create_nodes(nstart, nfinish);
379 } catch (...) {
380 deque::destroy_nodes(map_pair_.value, map_pair_.value + map_size_pair_.value - 1);
381 map_pair_.get_base().deallocate(map_pair_.value, map_size_pair_.value);
382 map_pair_.value = nullptr;
383 map_size_pair_.value = 0;
384 throw;
385 }

387 start_.change_buff(nstart);
388 finish_.change_buff(nfinish);
389 start_.current_ = start_.first_;
390 finish_.current_ = n == 0 ? start_.first_ : finish_.first_ + (n % buffer_size ? n % buffer_size : buffer_size);
391 start_.container_ = this;
392 finish_.container_ = this;

503 void reallocate_map(const size_type n, const bool add_at_front) {
504 const size_type begin_left = start_.current_ - start_.first_;
505
506 if (add_at_front && begin_left < n) {
507 const size_t needed = (n - begin_left) / buffer_size + 1;
508
509 if (needed > static_cast<size_type>(start_.node_ - map_pair_.value)) {
510 const size_type new_size =
511 _NEFORCE max(map_size_pair_.value << 1, map_size_pair_.value + needed + init_map_size);
512 map_pointer map = deque::create_map(new_size);
513 const size_type old_buf = finish_.node_ - start_.node_ + 1;
514 const size_type new_buf = needed + old_buf;
515
516 auto begin = map + (new_size - new_buf) / 2;
517 auto mid = begin + needed;
518 auto end = mid + old_buf;
519
520 deque::create_nodes(begin, mid - 1);
521 for (auto begin1 = mid, begin2 = start_.node_; begin1 != end; ++begin1, ++begin2) {
522 *begin1 = *begin2;
523 }
524
525 map_pair_.get_base().deallocate(map_pair_.value, map_size_pair_.value);
526 map_pair_.value = map;
527 map_size_pair_.value = new_size;
528 start_ = iterator(*mid + (start_.current_ - start_.first_), mid, this);
529 finish_ = iterator(*(end - 1) + (finish_.current_ - finish_.first_), end - 1, this);
530 return;
531 }
532
533 deque::create_nodes(start_.node_ - needed, start_.node_ - 1);
534 return;
535 }
536
537 const size_type end_left = finish_.last_ - finish_.current_ - 1;
538
539 if (!add_at_front && end_left < n) {
540
541 const size_type needed = (n - end_left) / buffer_size + 1;
542
543 if (needed > static_cast<size_type>((map_pair_.value + map_size_pair_.value) - finish_.node_ - 1)) {
544 const size_type new_size =
545 _NEFORCE max(map_size_pair_.value << 1, map_size_pair_.value + needed + init_map_size);
546 map_pointer map = deque::create_map(new_size);
547 const size_type old_buf = finish_.node_ - start_.node_ + 1;
548 const size_type new_buf = needed + old_buf;
549
550 auto begin = map + (new_size - new_buf) / 2;
551 auto mid = begin + old_buf;
552 auto end = mid + needed;
553
554 for (auto begin1 = begin, begin2 = start_.node_; begin1 != mid; ++begin1, ++begin2) {
555 *begin1 = *begin2;
556 }
557 deque::create_nodes(mid, end - 1);
558
559 map_pair_.get_base().deallocate(map_pair_.value, map_size_pair_.value);
560 map_pair_.value = map;
561 map_size_pair_.value = new_size;
562 start_ = iterator(*begin + (start_.current_ - start_.first_), begin, this);
563 finish_ = iterator(*(mid - 1) + (finish_.current_ - finish_.first_), mid - 1, this);
564 return;
565 }
566
567 deque::create_nodes(finish_.node_ + 1, finish_.node_ + needed);
568 }

579 template <typename Iterator>
580 void insert_ranges_n(iterator position, Iterator first, Iterator last, size_type n) {
581 difference_type dist_before = position - start_;
582
583 if (dist_before < static_cast<difference_type>(size() / 2)) {
584 deque::reallocate_map(n, true);
585 auto old_start = start_;
586 auto new_start = start_ - n;
587 position = start_ + dist_before;
588
589 try {
590 if (dist_before >= n) {
591 iterator start_n = start_ + n;
592 _NEFORCE uninitialized_copy(start_, start_n, new_start);
593 start_ = new_start;
594 _NEFORCE copy(start_n, position, old_start);
595 _NEFORCE copy(first, last, position - n);
596 } else {
597 auto mid = _NEFORCE next(first, n - dist_before);
598 _NEFORCE uninitialized_copy(first, mid, _NEFORCE uninitialized_copy(start_, position, new_start));
599 start_ = new_start;
600 _NEFORCE copy(mid, last, old_start);
601 }
602 } catch (...) {
603 if (new_start.node_ != start_.node_) {
604 deque::destroy_nodes(new_start.node_, start_.node_ - 1);
605 }
606 throw;
607 }
608 } else {
609 deque::reallocate_map(n, false);
610 auto old_finish = finish_;
611 auto new_finish = finish_ + n;
612 const size_type dist_after = size() - dist_before;
613 position = finish_ - dist_after;
614
615 try {
616 if (dist_after > n) {
617 auto finish_n = finish_ - n;
618 _NEFORCE uninitialized_copy(finish_n, finish_, finish_);
619 finish_ = new_finish;
620 _NEFORCE copy_backward(position, finish_n, old_finish);
621 _NEFORCE copy(first, last, position);
622 } else {
623 auto mid = _NEFORCE next(first, dist_after);
624 _NEFORCE uninitialized_copy(position, finish_, _NEFORCE uninitialized_copy(mid, last, finish_));
625 finish_ = new_finish;
626 _NEFORCE copy(first, mid, position);
627 }
628 } catch (...) {
629 if (new_finish.node_ != finish_.node_) {
630 deque::destroy_nodes(finish_.node_ + 1, new_finish.node_);
631 }
632 throw;
633 }
634 }
