139 NEFORCE_NODISCARD reference dereference() const noexcept {
140 NEFORCE_DEBUG_VERIFY(current_ && container_, "Attempting to dereference on a null pointer");
141 NEFORCE_DEBUG_VERIFY(first_ <= current_ && current_ < last_, "Attempting to dereference out of boundary");
142 return *current_;
143 }

150 void increment() noexcept {
151 NEFORCE_DEBUG_VERIFY(current_ && container_, "Attempting to increment a null pointer");
152 ++current_;
153 if (current_ == last_) {
154 deque_iterator::change_buff(node_ + 1);
155 current_ = first_;
156 }
157 }
158
164 void decrement() noexcept {
165 NEFORCE_DEBUG_VERIFY(current_ && container_, "Attempting to decrement a null pointer");
166 if (current_ == first_) {
167 deque_iterator::change_buff(node_ - 1);
168 current_ = last_;
169 }
170 --current_;
171 }
172
179 void advance(difference_type off) noexcept {
180 NEFORCE_DEBUG_VERIFY((current_ && container_) || off == 0, "Attempting to advance a null pointer");
181 const difference_type offset = off + (current_ - first_);
182 if (offset >= 0 && offset < buffer_size) {
183 current_ += off;
184 } else {
185 difference_type node_offset = offset > 0 ?
186 offset / buffer_size :
187 -static_cast<difference_type>((-offset - 1) / buffer_size) - 1;
188 deque_iterator::change_buff(node_ + node_offset);
189 current_ = first_ + (offset - node_offset * buffer_size);
190 }
191 }

360 void create_map_and_nodes(const size_type n) {
361 size_type node_nums = n / buffer_size + (n % buffer_size ? 1 : 0);
362 map_size_pair_.value = _NEFORCE max(init_map_size, node_nums + 2);
363
364 try {
365 map_pair_.value = deque::create_map(map_size_pair_.value);
366 } catch (...) {
367 map_pair_.value = nullptr;
368 map_size_pair_.value = 0;
369 throw;
370 }
371
372 map_pointer nstart = map_pair_.value + (map_size_pair_.value - node_nums) / 2;
373 map_pointer nfinish = node_nums == 0 ? nstart : nstart + node_nums - 1;
374
375 try {
376 deque::create_nodes(nstart, nfinish);
377 } catch (...) {
378 deque::destroy_nodes(map_pair_.value, map_pair_.value + map_size_pair_.value - 1);
379 map_pair_.get_base().deallocate(map_pair_.value, map_size_pair_.value);
380 map_pair_.value = nullptr;
381 map_size_pair_.value = 0;
382 throw;
383 }

386 finish_.change_buff(nfinish);
387 start_.current_ = start_.first_;
388 finish_.current_ = n == 0 ?
389 start_.first_ :
390 finish_.first_ + (n % buffer_size ? n % buffer_size : buffer_size);
391 start_.container_ = this;
392 finish_.container_ = this;

508
509 if (add_at_front && begin_left < n) {

517 const size_type old_buf = finish_.node_ - start_.node_ + 1;
518 const size_type new_buf = needed + old_buf;

521 auto mid = begin + needed;
522 auto end = mid + old_buf;

534 finish_ = iterator(*(end - 1) + (finish_.current_ - finish_.first_), end - 1, this);

586 template <typename Iterator>
587 void insert_ranges_n(iterator position, Iterator first, Iterator last, size_type n) {
588 difference_type dist_before = position - start_;
589
590 if (dist_before < static_cast<difference_type>(size() / 2)) {
591 deque::reallocate_map(n, true);
592 auto old_start = start_;
593 auto new_start = start_ - n;
594 position = start_ + dist_before;
595
596 try {
597 if (dist_before >= n) {
598 iterator start_n = start_ + n;
599 _NEFORCE uninitialized_copy(start_, start_n, new_start);
600 start_ = new_start;
601 _NEFORCE copy(start_n, position, old_start);
602 _NEFORCE copy(first, last, position - n);
603 } else {
604 auto mid = _NEFORCE next(first, n - dist_before);
605 _NEFORCE uninitialized_copy(first, mid,
606 _NEFORCE uninitialized_copy(start_, position, new_start));
607 start_ = new_start;
608 _NEFORCE copy(mid, last, old_start);
609 }
610 } catch (...) {
611 if (new_start.node_ != start_.node_) {
612 deque::destroy_nodes(new_start.node_, start_.node_ - 1);
613 }
614 throw;
615 }
616 } else {
617 deque::reallocate_map(n, false);
618 auto old_finish = finish_;
619 auto new_finish = finish_ + n;
620 const size_type dist_after = size() - dist_before;
621 position = finish_ - dist_after;
622
623 try {
624 if (dist_after > n) {
625 auto finish_n = finish_ - n;
626 _NEFORCE uninitialized_copy(finish_n, finish_, finish_);
627 finish_ = new_finish;
628 _NEFORCE copy_backward(position, finish_n, old_finish);
629 _NEFORCE copy(first, last, position);
630 } else {
631 auto mid = _NEFORCE next(first, dist_after);
632 _NEFORCE uninitialized_copy(position, finish_,
633 _NEFORCE uninitialized_copy(mid, last, finish_));
634 finish_ = new_finish;
635 _NEFORCE copy(first, mid, position);
636 }
637 } catch (...) {
638 if (new_finish.node_ != finish_.node_) {
639 deque::destroy_nodes(finish_.node_ + 1, new_finish.node_);
640 }
641 throw;
642 }
