234template <int Bytes>
235NEFORCE_ALWAYS_INLINE_INLINE vec128_t shift_left_bytes(vec128_t v) noexcept {
236 static_assert(Bytes >= 0 && Bytes <= 16, "shift_left_bytes: Bytes must be in [0, 16]");
237#ifdef NEFORCE_SIMD_SSE2
238 return _mm_bslli_si128(v, Bytes);
239#elif defined(NEFORCE_SIMD_NEON)
240 if (Bytes == 0) {
241 return v;
242 }
243 return ::vextq_u8(::vdupq_n_u8(0), v, static_cast<int>(16 - Bytes));
244#else
245 if (Bytes == 0) {
246 return v;
247 }
248 vec128_t result;
249 for (int i = 0; i < Bytes; ++i) {
250 result.data[i] = 0;
251 }
252 for (int i = Bytes; i < 16; ++i) {
253 result.data[i] = v.data[i - Bytes];
254 }
255 return result;
256#endif
257}

266NEFORCE_ALWAYS_INLINE_INLINE vec128_t shift_right_bytes(vec128_t v) noexcept {
267 static_assert(Bytes >= 0 && Bytes <= 16, "shift_right_bytes: Bytes must be in [0, 16]");
268#ifdef NEFORCE_SIMD_SSE2
269 return _mm_bsrli_si128(v, Bytes);
270#elif defined(NEFORCE_SIMD_NEON)
271 if (Bytes == 0) {
272 return v;
273 }
274 return ::vextq_u8(v, ::vdupq_n_u8(0), static_cast<int>(Bytes));
275#else
276 if (Bytes == 0) {
277 return v;
278 }
279 vec128_t result;
280 for (int i = 0; i < 16 - Bytes; ++i) {
281 result.data[i] = v.data[i + Bytes];
282 }
283 for (int i = 16 - Bytes; i < 16; ++i) {
284 result.data[i] = 0;
285 }
286 return result;
287#endif

295NEFORCE_ALWAYS_INLINE_INLINE int popcount(vec128_t v) noexcept {
296#if defined(NEFORCE_SIMD_SSSE3) || defined(NEFORCE_SIMD_AVX2)
297 const ::__m128i low_nibble = ::_mm_and_si128(v, ::_mm_set1_epi8(0x0F));
298 const ::__m128i high_nibble = ::_mm_and_si128(::_mm_srli_epi16(v, 4), ::_mm_set1_epi8(0x0F));
299 const ::__m128i lookup = ::_mm_setr_epi8(0, 1, 1, 2, 1, 2, 2, 3, 1, 2, 2, 3, 2, 3, 3, 4);
300 const ::__m128i pop_low = ::_mm_shuffle_epi8(lookup, low_nibble);
301 const ::__m128i pop_high = ::_mm_shuffle_epi8(lookup, high_nibble);
302 const ::__m128i pop8 = ::_mm_add_epi8(pop_low, pop_high);
303 const ::__m128i sums = ::_mm_sad_epu8(pop8, ::_mm_setzero_si128());
304 return ::_mm_extract_epi16(sums, 0) + ::_mm_extract_epi16(sums, 4);
305#elif defined(NEFORCE_SIMD_SSE2)
306 const auto* raw = reinterpret_cast<const byte_t*>(&v);
307 int total = 0;
308 for (int i = 0; i < 16; ++i) {
309 byte_t b = raw[i];
310 total += (b & 1) + ((b >> 1) & 1) + ((b >> 2) & 1) + ((b >> 3) & 1) + ((b >> 4) & 1) + ((b >> 5) & 1) +
311 ((b >> 6) & 1) + ((b >> 7) & 1);
312 }
313 return total;
314#elif defined(NEFORCE_SIMD_NEON)
315 const uint8x16_t low_nibble = ::vandq_u8(v, ::vdupq_n_u8(0x0F));
316 const uint8x16_t high_nibble = ::vshrq_n_u8(v, 4);
317 const uint8x16_t lookup = {0, 1, 1, 2, 1, 2, 2, 3, 1, 2, 2, 3, 2, 3, 3, 4};
318 uint8x16_t pop = ::vaddq_u8(::vqtbl1q_u8(lookup, low_nibble), ::vqtbl1q_u8(lookup, high_nibble));
319# ifdef __aarch64__
320 return static_cast<int>(::vaddvq_u8(pop));
321# else
322 int total = 0;
323 const auto* bytes = reinterpret_cast<const byte_t*>(&pop);
324 for (int i = 0; i < 16; ++i) {
325 total += bytes[i];
326 }
327 return total;
328# endif
329#else
330 int total = 0;
331 for (int i = 0; i < 16; ++i) {
332 byte_t b = v.data[i];
333 total += (b & 1) + ((b >> 1) & 1) + ((b >> 2) & 1) + ((b >> 3) & 1) + ((b >> 4) & 1) + ((b >> 5) & 1) +
334 ((b >> 6) & 1) + ((b >> 7) & 1);
335 }
336 return total;
337#endif
