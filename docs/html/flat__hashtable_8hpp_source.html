302 pair<size_t, bool> probe_find_or_insert(const key_type& key, const byte_t h2) const noexcept {
303 const size_t h1 = hash_to_index(hasher_(key));
304 size_t idx = h1;
305 size_t first_deleted = npos;
306
307 for (size_t i = 0; i < capacity_; ++i) {
308 const byte_t meta = metadata_[idx];
309 if (meta == FLAT_HT_EMPTY) {
310 return {first_deleted != npos ? first_deleted : idx, false};
311 }
312 if (meta == FLAT_HT_DELETED) {
313 if (first_deleted == npos) {
314 first_deleted = idx;
315 }
316 } else if (meta == h2 && equals_(extracter_(data_[idx]), key)) {
317 return {idx, true};
318 }
319 idx = (idx + 1) & (capacity_ - 1);
320 }
321 return {first_deleted, false};
322 }

330 pair<size_t, bool> probe_find_or_insert_simd(const key_type& key, const byte_t h2) const noexcept {
331 const size_t h1 = hash_to_index(hasher_(key));
332 size_t idx = h1;
333 size_t first_deleted = npos;
334
335 const simd::vec128_t h2_vec = simd::fill_byte(h2);
336 const simd::vec128_t empty_vec = simd::fill_byte(FLAT_HT_EMPTY);
337 const simd::vec128_t deleted_vec = simd::fill_byte(FLAT_HT_DELETED);
338
339 for (size_t round = 0; round < capacity_; round += 16) {
340 // wrap-around safe load: metadata array is exactly capacity_ bytes,
341 // a 16-byte load from idx may cross the array boundary
342 simd::vec128_t meta_vec;
343 const size_t remaining = capacity_ - idx;
344 if (remaining >= 16) {
345 meta_vec = simd::load_unaligned(metadata_ + idx);
346 } else {
347 byte_t buf[16];
348 for (size_t k = 0; k < remaining; ++k) {
349 buf[k] = metadata_[idx + k];
350 }
351 for (size_t k = 0; k < 16 - remaining; ++k) {
352 buf[remaining + k] = metadata_[k];
353 }
354 meta_vec = simd::load_unaligned(buf);
355 }
356
357 const int h2_mask = simd::to_bitmask(simd::match_bytes(meta_vec, h2_vec));
358 const int empty_mask = simd::to_bitmask(simd::match_bytes(meta_vec, empty_vec));
359 const int deleted_mask = simd::to_bitmask(simd::match_bytes(meta_vec, deleted_vec));
360
361 int match = h2_mask;
362 while (match != 0) {
363 const int bit = countr_zero(static_cast<uintptr_t>(match));
364 const size_t slot = (idx + bit) & (capacity_ - 1);
365 if (equals_(extracter_(data_[slot]), key)) {
366 return {slot, true};
367 }
368 match &= (match - 1);
369 }
370
371 if (empty_mask != 0) {
372 const int empty_bit = countr_zero(static_cast<uintptr_t>(empty_mask));
373 const size_t empty_slot = (idx + empty_bit) & (capacity_ - 1);
374 if (first_deleted != npos) {
375 const size_t probe_dist_empty =
376 (empty_slot >= h1) ? (empty_slot - h1) : (capacity_ - h1 + empty_slot);
377 const size_t probe_dist_del =
378 (first_deleted >= h1) ? (first_deleted - h1) : (capacity_ - h1 + first_deleted);
379 if (probe_dist_del < probe_dist_empty) {
380 return {first_deleted, false};
381 }
382 }
383 return {empty_slot, false};
384 }
385
386 const int del = deleted_mask;
387 if ((del != 0) && first_deleted == npos) {
388 const int del_bit = countr_zero(static_cast<uintptr_t>(del));
389 first_deleted = (idx + del_bit) & (capacity_ - 1);
390 }
391
392 idx = (idx + 16) & (capacity_ - 1);
393 }
394 return {first_deleted, false};
395 }

407 void rehash_impl(const size_t min_capacity) {
408 const size_t needed = max(min_capacity, static_cast<size_t>(static_cast<double>(size_) / max_load_factor()));
409 const size_t new_capacity = next_power_of_2(needed);
410 if (new_capacity <= capacity_) {
411 return;
412 }
413
414 allocator_type& alloc = get_allocator();
415 Value* new_data = alloc.allocate(new_capacity);
416 auto* new_metadata = static_cast<byte_t*>(::operator new(new_capacity * sizeof(byte_t), std::nothrow));
417 if (new_metadata == nullptr) {
418 alloc.deallocate(new_data, new_capacity);
419 NEFORCE_THROW_EXCEPTION(memory_exception("flat_hashtable rehash metadata allocation failed"));
420 }
421 for (size_t i = 0; i < new_capacity; ++i) {
422 new_metadata[i] = FLAT_HT_EMPTY;
423 }
424
425 const size_t old_capacity = capacity_;
426 Value* const old_data = data_;
427 byte_t* const old_metadata = metadata_;
428
429 try {
430 for (size_t i = 0; i < old_capacity; ++i) {
431 if (old_metadata[i] != FLAT_HT_EMPTY && old_metadata[i] != FLAT_HT_DELETED) {
432 const key_type& key = extracter_(old_data[i]);
433 const size_t hash = hasher_(key);
434 const byte_t h2 = hash_to_h2(hash);
435 const size_t h1 = (hash >> 7) & (new_capacity - 1);
436
437 size_t new_idx = h1;
438 while (new_metadata[new_idx] != FLAT_HT_EMPTY) {
439 new_idx = (new_idx + 1) & (new_capacity - 1);
440 }
441 _NEFORCE construct(&new_data[new_idx], _NEFORCE move(old_data[i]));
442 new_metadata[new_idx] = h2;
443 }
444 }
445 } catch (...) {
446 for (size_t i = 0; i < new_capacity; ++i) {
447 if (new_metadata[i] != FLAT_HT_EMPTY && new_metadata[i] != FLAT_HT_DELETED) {
448 _NEFORCE destroy(&new_data[i]);
449 }
450 }
451 alloc.deallocate(new_data, new_capacity);
452 ::operator delete(new_metadata, std::nothrow);
453 throw;
454 }
455
456 if (old_data) {
457 alloc.deallocate(old_data, old_capacity);
458 }
459 if (old_metadata != nullptr) {
460 ::operator delete(old_metadata, std::nothrow);
461 }
462
463 data_ = new_data;
464 metadata_ = new_metadata;
465 capacity_ = new_capacity;
466 growth_left_ = static_cast<size_t>(static_cast<double>(capacity_) * max_load_factor()) - size_;
467 }

509 bool equal_small(const flat_hashtable& rhs) const {
510 for (const_iterator iter = begin(); iter != end(); ++iter) {
511 const key_type& key = extracter_(*iter);
512 const size_t count_lhs = _NEFORCE count_if(
513 begin(), end(), [this, &key](const value_type& val) { return equals_(extracter_(val), key); });
514 const size_t count_rhs = _NEFORCE count_if(rhs.begin(), rhs.end(), [&rhs, &key](const value_type& val) {
515 return rhs.equals_(rhs.extracter_(val), key);
516 });
517 if (count_lhs != count_rhs) {
518 return false;
519 }
520 }
521 return true;
522 }

524 bool equal_large(const flat_hashtable& rhs) const {
525 if (size_ != rhs.size_) {
526 return false;
527 }
528 vector<const value_type*> ptrs_lhs, ptrs_rhs;
529 ptrs_lhs.reserve(size_);
530 ptrs_rhs.reserve(size_);
531 for (const_iterator it = begin(); it != end(); ++it) {
532 ptrs_lhs.push_back(&(*it));
533 }
534 for (const_iterator it = rhs.begin(); it != rhs.end(); ++it) {
535 ptrs_rhs.push_back(&(*it));
536 }
537
538 auto key_less = [this](const value_type* a, const value_type* b) { return extracter_(*a) < extracter_(*b); };
539 auto rhs_key_less = [&rhs](const value_type* a, const value_type* b) {
540 return rhs.extracter_(*a) < rhs.extracter_(*b);
541 };
542 _NEFORCE sort(ptrs_lhs.begin(), ptrs_lhs.end(), key_less);
543 _NEFORCE sort(ptrs_rhs.begin(), ptrs_rhs.end(), rhs_key_less);
544
545 size_type i = 0, j = 0;
546 const size_type n = ptrs_lhs.size();
547 while (i < n && j < n) {
548 const key_type& key_l = extracter_(*ptrs_lhs[i]);
549 const key_type& key_r = rhs.extracter_(*ptrs_rhs[j]);
550 if (!equals_(key_l, key_r)) {
551 return false;
552 }
553 size_type i_start = i, j_start = j;
554 while (i < n && equals_(extracter_(*ptrs_lhs[i]), key_l)) {
555 ++i;
556 }
557 while (j < n && rhs.equals_(rhs.extracter_(*ptrs_rhs[j]), key_l)) {
558 ++j;
559 }
560 const size_type count_l = i - i_start;
561 const size_type count_r = j - j_start;
562 if (count_l != count_r) {
563 return false;
564 }
565 for (size_type k = i_start; k < i; ++k) {
566 const value_type& val = *ptrs_lhs[k];
567 bool found = false;
568 for (size_type l = j_start; l < j; ++l) {
569 if (ptrs_rhs[l] && *ptrs_rhs[l] == val) {
570 ptrs_rhs[l] = nullptr;
571 found = true;
572 break;
573 }
574 }
575 if (!found) {
576 return false;
577 }
578 }
579 for (size_type l = j_start; l < j; ++l) {
580 if (ptrs_rhs[l] != nullptr) {
581 return false;
582 }
583 }
584 }
585 return true;
586 }

867 template <typename... Args>
868 pair<iterator, bool> emplace_unique(Args&&... args) {
869 if (should_rehash()) {
870 const size_type new_cap = capacity_ == 0 ? 16 : capacity_ * 2;
871 rehash(max(new_cap, static_cast<size_t>(static_cast<double>(size_ + 1) / max_load_factor())));
872 }
873
874 value_type tmp(_NEFORCE forward<Args>(args)...);
875 const key_type& key = extracter_(tmp);
876 const size_t hash = hasher_(key);
877 const byte_t h2 = hash_to_h2(hash);
878
879 pair<size_t, bool> probe_result;
880#ifdef NEFORCE_SIMD_SSE2
881 probe_result = probe_find_or_insert_simd(key, h2);
882#else
883 probe_result = probe_find_or_insert(key, h2);
884#endif
885
886 if (probe_result.second) {
887 return {iterator(probe_result.first, this), false};
888 }
889
890 const size_t insert_idx = probe_result.first;
891 _NEFORCE construct(&data_[insert_idx], _NEFORCE move(tmp));
892 metadata_[insert_idx] = h2;

905 iterator emplace_equal(Args&&... args) {

908 rehash(max(new_cap, static_cast<size_t>(static_cast<double>(size_ + 1) / max_load_factor())));

922 const size_t insert_idx = (first_deleted != npos) ? first_deleted : idx;
923 construct_at(insert_idx, h2, _NEFORCE move(tmp));

928 }

936 construct_at(insert_idx, h2, _NEFORCE move(tmp));

1249 NEFORCE_NODISCARD pair<iterator, iterator> equal_range(const key_type& key) {
1250 if (capacity_ == 0) {
1251 return {end(), end()};
1252 }
1253 const size_t hash = hasher_(key);
1254 const byte_t h2 = hash_to_h2(hash);
1255 size_t idx = hash_to_index(hash);
1256
1257 size_t first_idx = npos;
1258 for (size_t i = 0; i < capacity_; ++i) {
1259 const byte_t meta = metadata_[idx];
1260 if (meta == FLAT_HT_EMPTY) {
1261 return {end(), end()};
1262 }
1263 if (meta == h2 && equals_(extracter_(data_[idx]), key)) {
1264 first_idx = idx;
1265 break;
1266 }
1267 idx = (idx + 1) & (capacity_ - 1);
1268 }
1269 if (first_idx == npos) {
1270 return {end(), end()};
1271 }
1272
1273 size_t last_idx = first_idx;
1274 do {
1275 last_idx = (last_idx + 1) & (capacity_ - 1);
1276 if (last_idx == first_idx) {
1277 return {iterator(first_idx, this), end()};
1278 }
1279 } while (metadata_[last_idx] != FLAT_HT_EMPTY && equals_(extracter_(data_[last_idx]), key));
1280
1281 while (last_idx != first_idx &&
1282 (metadata_[last_idx] == FLAT_HT_EMPTY || metadata_[last_idx] == FLAT_HT_DELETED)) {
1283 last_idx = (last_idx + 1) & (capacity_ - 1);
1284 }
1285 if (last_idx == first_idx || last_idx < first_idx) {
1286 // wraparound: the range extends past the end of the array,
1287 // second iterator points to end() since forward iteration
1288 // cannot jump from the end back to the beginning
1289 return {iterator(first_idx, this), end()};
1290 }
1291 return {iterator(first_idx, this), iterator(last_idx, this)};
