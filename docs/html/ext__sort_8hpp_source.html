200 if (first == last) return;

202 for (auto gap = dist / 2; gap > 0; gap /= 2) {
203 for (Iterator i = first + gap; i < last; ++i) {

241template <typename Iterator, typename Compare, typename IndexMapper>
242NEFORCE_CONSTEXPR20 void counting_sort(Iterator first, Iterator last, Compare comp, IndexMapper mapper) {
243 static_assert(is_ranges_rnd_iter_v<Iterator>, "Iterator must be random_access_iterator");
244 static_assert(is_invocable_v<Compare, decltype(*first), decltype(*first)>, "Compare must be invocable");
245 static_assert(is_invocable_v<IndexMapper, decltype(*first)>, "IndexMapper must be invocable");
246
247 if (first == last) return;
248 auto min_max = _NEFORCE minmax_element(first, last, comp);
249 auto min_val = mapper(*min_max.first);
250 auto max_val = mapper(*min_max.second);
251 const auto range = static_cast<size_t>(max_val - min_val + 1);
252 vector<int> count(range, 0);
253
254 for (Iterator it = first; it != last; ++it) {
255 auto value = mapper(*it);
256 if (value < min_val || value > max_val) {
257 NEFORCE_THROW_EXCEPTION(iterator_exception("element out of range for counting sort."));
258 }
259 ++count[static_cast<size_t>(value - min_val)];
260 }
261
262 for (size_t i = 1; i < count.size(); ++i) {
263 count[i] += count[i - 1];
264 }
265
266 vector<iter_value_t<Iterator>> sorted(_NEFORCE distance(first, last));
267 auto bound = _NEFORCE make_reverse_iterator(first);
268
269 for (auto rit = _NEFORCE make_reverse_iterator(last); rit != bound; ++rit) {
270 auto value = mapper(*rit);
271 const auto index = static_cast<size_t>(value - min_val);
272 size_t position = --count[index];
273 sorted[position] = *rit;
274 }
275 _NEFORCE copy(sorted.begin(), sorted.end(), first);
276}

555 constexpr int min_merge = 32;
556 iter_difference_t<Iterator> n = _NEFORCE distance(first, last);
557 for (Iterator i = first; i < last; i += min_merge) {
558 Iterator end = _NEFORCE min(i + min_merge, last);
