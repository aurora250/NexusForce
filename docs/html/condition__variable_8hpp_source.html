253 template <typename Clock, typename Dur>
254 cv_status wait_until(unique_lock<mutex>& lock, const time_point<Clock, Dur>& util) {
255 const typename Clock::time_point entry = Clock::now();
256 const auto atime = clock_type::now() + ceil<clock_type::duration>(util - entry);
257
258 if (this->__wait_until_impl(lock, atime) == cv_status::success) {
259 return cv_status::success;
260 }
261 if (Clock::now() < util) {
262 return cv_status::success;
263 }
264 return cv_status::timeout;
265 }
