130 constexpr explicit bitset(unsigned long long value) noexcept {
131 blocks.fill(0);
132 static_assert(sizeof(unsigned long long) >= sizeof(block_type),
133 "unsigned long long should be at least as large as block_type");
134 size_t i = 0;
135 while (value != 0 && i < block_count) {

363 constexpr bitset& operator<<=(const uint32_t pos) noexcept {
364 if (pos >= N) {
365 reset();
366 return *this;
367 }
368 const size_t block_shift = pos / bits_per_block;
369 const size_t bit_shift = pos % bits_per_block;
370
371 if (block_shift > 0) {
372 for (size_t i = block_count - 1; i >= block_shift; --i) {
373 blocks[i] = blocks[i - block_shift];
374 }
375 for (size_t i = 0; i < block_shift; ++i) {
376 blocks[i] = 0;
377 }
378 }
379 if (bit_shift > 0) {
380 for (size_t i = block_count - 1; i > 0; --i) {
381 blocks[i] = (blocks[i] << bit_shift) | (blocks[i - 1] >> (bits_per_block - bit_shift));
382 }
383 blocks[0] <<= bit_shift;
384 }
385 blocks[block_count - 1] &= last_block_mask();
386
387 return *this;

404 for (size_t i = 0; i < block_count - block_shift; ++i) {
