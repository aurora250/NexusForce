231template <typename T>
232shared_ptr<T> database_pool::acquire_impl() {
233 unique_lock<mutex> lk(queue_mtx_);
234
235 const bool got = cv_.wait_for(lk, pool_cfg_.acquire_timeout,
236 [this] { return !idle_queue_.empty() || !running_.load(memory_order_relaxed); });
237
238 if (!running_.load(memory_order_relaxed)) {
239 return nullptr;
240 }
241
242 if (!got || idle_queue_.empty()) {
243 const size_t cur = total_count_.load(memory_order_relaxed);
244 if (cur >= pool_cfg_.max_size) {
245 return nullptr;
246 }
247 idb_connect* raw = try_create_connect();
248 if (raw == nullptr) {
249 return nullptr;
250 }
251
252 total_count_.fetch_add(1, memory_order_relaxed);
253 T* typed = dynamic_cast<T*>(raw);
254 if (typed == nullptr) {
255 delete raw;
256 total_count_.fetch_sub(1, memory_order_relaxed);
257 return nullptr;
258 }
259 return shared_ptr<T>(typed, [this](T* p) { this->return_connect(p); });
260 }
261
262 const connection_entry entry = idle_queue_.front();
263 idle_queue_.pop();
264 lk.unlock_quiet();
265 cv_.notify_one();
266
267 idb_connect* raw = entry.conn;
268
269 if (!raw->is_valid()) {
270 if (!raw->reconnect(config_)) {
271 delete raw;
272 total_count_.fetch_sub(1, memory_order_relaxed);
273 cv_.notify_one();
274 return nullptr;
275 }
276 }
277
278 T* typed = dynamic_cast<T*>(raw);
279 if (typed == nullptr) {
280 this->return_connect(raw);
281 return nullptr;
282 }
283
284 return shared_ptr<T>(typed, [this](T* p) { this->return_connect(p); });
285}
