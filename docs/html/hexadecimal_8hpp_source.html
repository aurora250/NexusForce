60 static NEFORCE_CONSTEXPR20 value_type parse_view(const string_view view) {
61 if (view.empty()) return 0;
62
63 bool negative = false;
64 size_t start = 0;
65
66 while (start < view.size() && _NEFORCE is_space(view[start])) ++start;
67 if (start == view.size()) return 0;
68
69 if (view[start] == '-') {
70 negative = true;
71 ++start;
72 } else if (view[start] == '+') {
73 ++start;
74 }
75
76 if (start + 1 < view.size() && view[start] == '0' &&
77 (view[start+1] == 'x' || view[start+1] == 'X')) {
78 start += 2;
79 }
80
81 uint64_t result = 0;
82 while (start < view.size()) {
83 const char c = view[start++];
84 if (is_xdigit(c)) {
85 const int digit = digit_value(c);
86 if (result > (numeric_traits<uint64_t>::max() >> 4)) {
87 throw_exception(value_exception("Hexadecimal value too large"));
88 }
89 result = (result << 4) | static_cast<uint64_t>(digit);
90 } else if (!is_space(c)) {
91 throw_exception(value_exception("Invalid hexadecimal character"));
92 }
93 }
94
95 if (negative) {
96 if (result > static_cast<uint64_t>(numeric_traits<int64_t>::max()) + 1) {
97 throw_exception(value_exception("Hexadecimal value out of range"));
98 }
99 return -static_cast<int64_t>(result);
100 }
101 if (result > static_cast<uint64_t>(numeric_traits<int64_t>::max())) {
102 throw_exception(value_exception("Hexadecimal value out of range"));
103 }
104 return static_cast<value_type>(result);
105 }
