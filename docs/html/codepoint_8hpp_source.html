246 template <typename T>
247 static bool utf16_codepoint(const T* data, size_t index, const size_t len, uint32_t& cp, size_t& consumed,
248 const bool need_swap) {
249 if (index >= len) {
250 cp = 0xFFFD;
251 consumed = 0;
252 return false;
253 }
254
255 const auto raw1 = static_cast<uint16_t>(data[index]);
256 const auto c1 = static_cast<uint32_t>(need_swap ? endian::byteswap16(raw1) : raw1);
257 consumed = 1;
258
259 if (is_high_surrogate(c1)) {
260 if (index + 1 < len) {
261 const auto raw2 = static_cast<uint16_t>(data[index + 1]);
262 const auto c2 = static_cast<uint32_t>(need_swap ? endian::byteswap16(raw2) : raw2);
263
264 if (is_low_surrogate(c2)) {
265 cp = combine_surrogates(c1, c2).value();
266 consumed = 2;
267 return true;
268 }
269 }
270
271 cp = 0xFFFD;
272 return false;
273 }
274
275 if (is_low_surrogate(c1)) {
276 cp = 0xFFFD;
277 return false;
278 }
279
280 cp = c1;
281 return true;
282 }
