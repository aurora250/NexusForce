156 template <typename T>
157 static bool utf16_codepoint(const T* data, size_t index, const size_t len, uint32_t& cp, size_t& consumed,
158 const bool need_swap) {
159 if (index >= len) {
160 cp = 0xFFFD;
161 consumed = 0;
162 return false;
163 }
164
165 const auto raw1 = static_cast<uint16_t>(data[index]);
166 const auto c1 = static_cast<uint32_t>(need_swap ? endian::byteswap16(raw1) : raw1);
167 consumed = 1;
168
169 if (is_high_surrogate(c1)) {
170 if (index + 1 < len) {
171 const auto raw2 = static_cast<uint16_t>(data[index + 1]);
172 const auto c2 = static_cast<uint32_t>(need_swap ? endian::byteswap16(raw2) : raw2);
173
174 if (is_low_surrogate(c2)) {
175 cp = combine_surrogates(c1, c2).value();
176 consumed = 2;
177 return true;
178 }
179 }
180
181 cp = 0xFFFD;
182 return false;
183 }
184
185 if (is_low_surrogate(c1)) {
186 cp = 0xFFFD;
187 return false;
188 }
189
190 cp = c1;
191 return true;
192 }
