248 template <typename T>
249 static bool utf16_codepoint(const T* data, size_t index, const size_t len, uint32_t& cp, size_t& consumed,
250 const bool need_swap) {
251 if (index >= len) {
252 cp = 0xFFFD;
253 consumed = 0;
254 return false;
255 }
256
257 const auto raw1 = static_cast<uint16_t>(data[index]);
258 const auto c1 = static_cast<uint32_t>(need_swap ? endian::byteswap16(raw1) : raw1);
259 consumed = 1;
260
261 if (is_high_surrogate(c1)) {
262 if (index + 1 < len) {
263 const auto raw2 = static_cast<uint16_t>(data[index + 1]);
264 const auto c2 = static_cast<uint32_t>(need_swap ? endian::byteswap16(raw2) : raw2);
265
266 if (is_low_surrogate(c2)) {
267 cp = combine_surrogates(c1, c2).value();
268 consumed = 2;
269 return true;
270 }
271 }
272
273 cp = 0xFFFD;
274 return false;
275 }
276
277 if (is_low_surrogate(c1)) {
278 cp = 0xFFFD;
279 return false;
280 }
281
282 cp = c1;
283 return true;
284 }
