547 void emit(Types... args) {
548 vector<slot_entry> snapshot;
549 {
550 lock<mutex> lk(mutex_);
551 if (is_blocked()) {
552 return;
553 }
554 snapshot = slots_;
555 }
556
557 for (auto& slot: snapshot) {
558 if (!slot.connected_flag || !(*slot.connected_flag)) {
559 continue;
560 }
561
562 const callback_result res = slot.callback(args...);
563 if (res == callback_result::erase) {
564 *slot.connected_flag = false;
565 }
566 }
567
568 {
569 lock<mutex> lk(mutex_);
570 slots_.erase(
571 _NEFORCE remove_if(slots_.begin(), slots_.end(),
572 [](const slot_entry& s) { return !s.connected_flag || !(*s.connected_flag); }),
573 slots_.end());
574 }
