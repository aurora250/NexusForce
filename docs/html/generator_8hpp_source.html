840task<T> retry(Factory&& factory, const size_t max_attempts,
841 function<bool(const exception_ptr&)> should_retry = nullptr) {
842 static_assert(_NEFORCE is_invocable_r_v<task<T>, Factory>, "Factory must return task<T>");
843
844 exception_ptr last_exception;
845
846 for (size_t attempt = 0; attempt < max_attempts; ++attempt) {
847 try {
848 co_return co_await factory();
849 } catch (...) {
850 last_exception = _NEFORCE current_exception();
851
852 if (should_retry && !should_retry(last_exception)) {
853 rethrow_exception(last_exception);
854 }

856 rethrow_exception(last_exception);
857 }
858 }
859 }
