896task<T> retry(Factory&& factory, const size_t max_attempts,
897 function<bool(const exception_ptr&)> should_retry = nullptr) {
898 static_assert(_NEFORCE is_invocable_r_v<task<T>, Factory>, "Factory must return task<T>");
899
900 exception_ptr last_exception;
901
902 for (size_t attempt = 0; attempt < max_attempts; ++attempt) {
903 try {
904 co_return co_await factory();
905 } catch (...) {
906 last_exception = _NEFORCE current_exception();
907
908 if (should_retry && !should_retry(last_exception)) {
909 rethrow_exception(last_exception);
910 }
911 if (attempt == max_attempts - 1) {
912 rethrow_exception(last_exception);
913 }
914 }
915 }
