287template <typename T>
288NEFORCE_PURE_FUNCTION NEFORCE_CONSTEXPR14 T logarithm_e(const T x) {
289 static_assert(is_floating_point_v<T>, "floating point required");
290 if (x <= 0) {
291 NEFORCE_THROW_EXCEPTION(math_exception("Logarithm domain error"));
292 }
293 uint32_t N = constants::TAYLOR_CONVERGENCE;
294 const T a = (x - 1) / (x + 1);
295 const T a_sqrt = a * a;
296 T nk = 2 * N + 1;
297 T y = 1.0 / nk;
298 while (N--) {
299 nk -= 2;
300 y = 1.0 / nk + a_sqrt * y;
301 }
302 return 2.0 * a * y;
303}

609NEFORCE_PURE_FUNCTION NEFORCE_CONSTEXPR14 decimal_t sine(decimal_t x) noexcept {
610 decimal_t sign = 1.0;
611 if (x < 0) {
612 sign = -1.0;
613 x = -x;
614 }
615 constexpr decimal_t twoPi = 2 * constants::PI;
616 x = x - twoPi * floor(x / twoPi);
617
618 if (x > constants::PI) {
619 x -= constants::PI;
620 sign = -sign;
621 }
622 if (x > constants::PI / 2) {
623 x = constants::PI - x;
624 }
625
626 decimal_t i = 1;
627 int32_t neg = 1;
628 decimal_t term = 0;
629 decimal_t idx = x;
630 decimal_t fac = 1;
631 decimal_t taylor = x;
632 do {
633 fac = fac * (i + 1) * (i + 2);
634 idx *= x * x;
635 neg = -neg;
636 term = idx / fac * neg;
637 taylor += term;
638 i += 2;
639 } while (absolute(term) > constants::EPSILON);
640 const auto fin = sign * taylor;
641 return around_zero(fin) ? 0 : fin;
642}

687NEFORCE_PURE_FUNCTION NEFORCE_CONSTEXPR14 decimal_t __arctangent_taylor(const decimal_t x) noexcept {
688 const decimal_t x_sq = x * x;
689 decimal_t term = x;
690 decimal_t sum = x;
691 decimal_t n = 1.0;
692 while (absolute(term) > constants::EPSILON) {
693 term *= -x_sq;
694 n += 2.0;
695 sum += term / n;
696 }
697 return sum;
