418NEFORCE_CONST_FUNCTION NEFORCE_CONSTEXPR14 decimal_t logarithm_e(const decimal_t x) noexcept {
419 if (is_nan(x)) {
420 return x;
421 }
422 if (x < 0.0L) {
423 return numeric_traits<decimal_t>::quiet_nan();
424 }
425 if (x == 0.0L) {
426 return -numeric_traits<decimal_t>::infinity();
427 }
428 if (is_infinity(x) && x > 0) {
429 return x;
430 }
431
432 int64_t k = 0;
433 decimal_t m = x;
434
435 if (m >= 2.0L) {
436 while (m >= 2.0L) {
437 m *= 0.5L;
438 ++k;
439 }
440 } else if (m < 1.0L) {
441 while (m < 1.0L) {
442 m *= 2.0L;
443 --k;
444 }
445 }
446
447 const decimal_t a = (m - 1.0L) / (m + 1.0L);
448 const decimal_t a2 = a * a;
449 decimal_t term = a;
450 decimal_t s = a;
451 decimal_t n = 1.0L;
452
453 while (absolute(term) > constants::MACHINE_EPSILON * absolute(s)) {
454 term *= a2;
455 n += 2.0L;
456 s += term / n;
457 }
458
459 constexpr decimal_t LN2 = 0.69314718055994530941723212145817656807L;
460 return 2.0L * s + static_cast<decimal_t>(k) * LN2;
461}

532cube_root(const decimal_t x, const decimal_t precise = constants::DEFAULT_TOLERANCE) noexcept {
533 if (is_nan(x)) {
534 return x;

536 if (is_infinity(x)) {
537 return x;

540 return x;
541 }

863NEFORCE_CONSTEXPR14 void sincos_taylor(const decimal_t x, decimal_t& sin_val, decimal_t& cos_val) noexcept {
864 const decimal_t x2 = x * x;
865
866 decimal_t term_sin = x;
867 decimal_t sum_sin = x;
868
869 decimal_t term_cos = 1.0L;
870 decimal_t sum_cos = 1.0L;
871
872 decimal_t i = 1.0L;
873 bool sin_done = false, cos_done = false;
874
875 while (!sin_done || !cos_done) {
876 if (!sin_done) {
877 term_sin = -term_sin * x2 / ((i + 1.0L) * (i + 2.0L));
878 sum_sin += term_sin;
879 if (absolute(term_sin) <= constants::MACHINE_EPSILON * absolute(sum_sin)) {
880 sin_done = true;
881 }
882 }
883 if (!cos_done) {
884 term_cos = -term_cos * x2 / (i * (i + 1.0L));
885 sum_cos += term_cos;
886 if (absolute(term_cos) <= constants::MACHINE_EPSILON * absolute(sum_cos)) {
887 cos_done = true;
888 }
889 }
890 i += 2.0L;
891 }
892
893 sin_val = sum_sin;
894 cos_val = sum_cos;
895}
