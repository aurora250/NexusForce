137MSTL_CONST_FUNCTION MSTL_CONSTEXPR14 T power(const T& x, UT n) noexcept {
138 constexpr UT zero = UT(0);
139 constexpr UT one = UT(1);
140 constexpr UT two = UT(2);
141
142 if (n == zero) return 1;
143 T result(1);

149 n /= two;
150 }

159MSTL_PURE_FUNCTION MSTL_CONSTEXPR14 decimal_t logarithm_e(const decimal_t x) noexcept {
160 uint32_t N = _CONSTANTS TAYLOR_CONVERGENCE;
161 const decimal_t a = (x - 1) / (x + 1);
162 const decimal_t a_sqrt = a * a;
163 decimal_t nk = 2 * N + 1;
164 decimal_t y = 1.0 / nk;
165 while (N--) {
166 nk -= 2;
167 y = 1.0 / nk + a_sqrt * y;
168 }
169 return 2.0 * a * y;
170}

320MSTL_PURE_FUNCTION MSTL_CONSTEXPR14 decimal_t sine(decimal_t x) noexcept {
321 decimal_t sign = 1.0;
322 if (x < 0) {
323 sign = -1.0;
324 x = -x;
325 }
326 constexpr decimal_t twoPi = 2 * _CONSTANTS PI;
327 x = x - twoPi * floor(x / twoPi);
328
329 if (x > _CONSTANTS PI) {
330 x -= _CONSTANTS PI;
331 sign = -sign;
332 }
333 if (x > _CONSTANTS PI / 2) {
334 x = _CONSTANTS PI - x;
335 }
336
337 decimal_t i = 1;
338 int32_t neg = 1;
339 decimal_t sum = 0;
340 decimal_t idx = x;
341 decimal_t fac = 1;
342 decimal_t taylor = x;
343 do {
344 fac = fac * (i + 1) * (i + 2);
345 idx *= x * x;
346 neg = -neg;
347 sum = idx / fac * neg;
348 taylor += sum;
349 i += 2;
350 } while (absolute(sum) > _CONSTANTS EPSILON);
351 const auto fin = sign * taylor;
352 return around_zero(fin) ? 0 : fin;
353}
