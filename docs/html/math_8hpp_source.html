283template <typename T>
284NEFORCE_PURE_FUNCTION NEFORCE_CONSTEXPR14 T logarithm_e(const T x) noexcept {
285 static_assert(is_floating_point_v<T>, "floating point required");
286 uint32_t N = _CONSTANTS TAYLOR_CONVERGENCE;
287 const T a = (x - 1) / (x + 1);
288 const T a_sqrt = a * a;
289 T nk = 2 * N + 1;
290 T y = 1.0 / nk;
291 while (N--) {
292 nk -= 2;
293 y = 1.0 / nk + a_sqrt * y;
294 }
295 return 2.0 * a * y;
296}

602NEFORCE_PURE_FUNCTION NEFORCE_CONSTEXPR14 decimal_t sine(decimal_t x) noexcept {
603 decimal_t sign = 1.0;
604 if (x < 0) {
605 sign = -1.0;
606 x = -x;
607 }
608 constexpr decimal_t twoPi = 2 * _CONSTANTS PI;
609 x = x - twoPi * floor(x / twoPi);
610
611 if (x > _CONSTANTS PI) {
612 x -= _CONSTANTS PI;
613 sign = -sign;
614 }
615 if (x > _CONSTANTS PI / 2) {
616 x = _CONSTANTS PI - x;
617 }
618
619 decimal_t i = 1;
620 int32_t neg = 1;
621 decimal_t sum = 0;
622 decimal_t idx = x;
623 decimal_t fac = 1;
624 decimal_t taylor = x;
625 do {
626 fac = fac * (i + 1) * (i + 2);
627 idx *= x * x;
628 neg = -neg;
629 sum = idx / fac * neg;
630 taylor += sum;
631 i += 2;
632 } while (absolute(sum) > _CONSTANTS EPSILON);
633 const auto fin = sign * taylor;
634 return around_zero(fin) ? 0 : fin;
635}

684 const decimal_t x_sq = x * x;
685 decimal_t term = x;
