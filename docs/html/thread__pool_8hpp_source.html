63
67 void increment() noexcept { running_count.fetch_add(1, memory_order_relaxed); }
68
74 void decrement() noexcept {
75 if (running_count.fetch_sub(1, memory_order_release) == 1) {
76 running_count.notify_all();

84 size_t count = running_count.load(memory_order_acquire);
85 while (count != 0) {

629template <typename Func, typename... Args>
630submit_result<invoke_result_t<Func, Args...>> thread_pool::submit_task(const priority_type priority, Func&& func,
631 Args&&... args) {
632 static_assert(is_invocable_v<Func, Args...>, "Func must be invocable with Args");
633
634 using Result = invoke_result_t<Func, Args...>;
635
636 auto info = make_shared<task_info>(generate_task_id(), priority);
637
638 const auto current_group = get_current_task_group();
639 if (current_group) {
640 current_group->increment();
641 }
642
643 auto task = _NEFORCE make_shared<packaged_task<Result()>>(
644 [func = _NEFORCE forward<Func>(func), args = _NEFORCE make_tuple(_NEFORCE forward<Args>(args)...),
645 group = current_group, info]() mutable -> Result {
646 struct context_guard {
647 shared_ptr<task_info> info;
648 shared_ptr<task_group> group_inner;
649 shared_ptr<task_group> prev_group_inner;
650
651 explicit context_guard(shared_ptr<task_info> i, shared_ptr<task_group> g) :
652 info(move(i)),
653 group_inner(move(g)) {
654 info->status.store(task_info::status::running, memory_order_release);
655 info->start_time = timestamp::now();
656 info->worker_thread_id = get_worker_context() ? get_worker_context()->id : 0;
657
658 prev_group_inner = get_current_task_group();
659 get_current_task_group() = group_inner;
660 }
661
662 ~context_guard() noexcept {
663 try {
664 info->finish_time = timestamp::now();
665 auto expected = task_info::status::running;
666 info->status.compare_exchange_strong(expected, task_info::status::completed,
667 memory_order_release);
668
669 get_current_task_group() = prev_group_inner;
670 if (group_inner) {
671 group_inner->decrement();
672 }
673 // NOLINTNEXTLINE(bugprone-empty-catch)
674 } catch (...) {
675 /* ignore */
676 }
677 }
678 };
679
680 context_guard guard(info, group);
681 try {
682 return _NEFORCE apply(func, args);
683 } catch (const exception& e) {
684 info->status.store(task_info::status::failed, memory_order_release);
685 info->error = e.what();
686 throw;
687 } catch (...) {
688 info->status.store(task_info::status::failed, memory_order_release);
689 info->error = "Unknown exception";
690 throw;
691 }
692 });
693
694 future<Result> res = task->get_future();
695 task_type job([task] { (*task)(); });
696
697 if (static_cast<uint32_t>(priority) > 0) {
698 unique_lock<mutex> lock(task_queue_mtx_);
699
700 if (!not_full_.wait_for(lock, seconds(1), [&]() -> bool { return task_queue_.size() < task_threshhold_; })) {
701 info->status.store(task_info::status::failed, memory_order_release);
702 info->error = "Task queue is full";
703
704 auto dummy_task = _NEFORCE make_shared<packaged_task<Result()>>([]() -> Result { return Result(); });
705 (*dummy_task)();
706 return submit_result<Result>{dummy_task->get_future(), info};
707 }
708
709 task_queue_.emplace(move(job), priority, info);
710 ++task_size_;
711 ++total_submitted_tasks_;
712 not_empty_.notify_one();
713
714 } else {
715 auto* ctx = get_worker_context();
716
717 if (ctx != nullptr && ctx->queue.remain_size() > 0) {
718 ctx->queue.push_back(move(job));
719 ++total_submitted_tasks_;
720 } else {
721 unique_lock<mutex> lock(task_queue_mtx_);
722 if (!not_full_.wait_for(lock, seconds(1),
723 [&]() -> bool { return task_queue_.size() < task_threshhold_; })) {
724 info->status.store(task_info::status::failed, memory_order_release);
725 info->error = "Task queue is full";
726
727 auto dummy_task = _NEFORCE make_shared<packaged_task<Result()>>([]() -> Result { return Result(); });
728 (*dummy_task)();
729 return submit_result<Result>{dummy_task->get_future(), info};
730 }
731
732 task_queue_.emplace(move(job), static_cast<priority_type>(0), info);
733 ++task_size_;
734 ++total_submitted_tasks_;
735 not_empty_.notify_one();
736 }
737 }
738
739 if (pool_mode_.load() == pool_mode::cached && task_size_.load() > idle_thread_size_) {
740
741 inner::manual_thread* t_ptr = nullptr;
742 id_type thread_id = 0;
743
744 {
745 unique_lock<mutex> lock(task_queue_mtx_);
746 if (threads_map_.size() < thread_threshhold_) {
747 auto ptr =
748 _NEFORCE make_unique<inner::manual_thread>([this](const id_type id) { thread_function(id); });
749
750 thread_id = ptr->id();
751 t_ptr = ptr.get();
752 threads_map_.emplace(thread_id, move(ptr));
753 }
754 }
755
756 if (t_ptr != nullptr) {
757 {
758 lock<mutex> ctx_lock(worker_contexts_mtx_);
759 if (thread_id >= worker_contexts_ptr_.size()) {
760 worker_contexts_ptr_.reserve(thread_id + 1);
761 for (size_t i = worker_contexts_ptr_.size(); i <= thread_id; i++) {
762 atomic<worker_context*> tmp;
763 tmp.store(nullptr, memory_order_relaxed);
764 worker_contexts_ptr_.emplace_back(move(tmp));
765 }
766 }
767 }
768
769 t_ptr->start();
770 }
771 }
772
773 return submit_result<Result>{move(res), move(info)};
774}

777submit_result<invoke_result_t<Func, Args...>>
778thread_pool::submit_after(const int64_t delay_ms, const priority_type priority, Func&& func, Args&&... args) {
779 static_assert(is_invocable_v<Func, Args...>, "Func must be invocable with Args");
780
781 using Result = invoke_result_t<Func, Args...>;
782
783 auto info = make_shared<task_info>(generate_task_id(), priority);
784
785 auto task = _NEFORCE make_shared<packaged_task<Result()>>(
786 [func = _NEFORCE forward<Func>(func), tup = _NEFORCE make_tuple(_NEFORCE forward<Args>(args)...),
787 info]() mutable {
788 struct context_guard {
789 shared_ptr<task_info> info;
790
791 explicit context_guard(shared_ptr<task_info> i) :
792 info(move(i)) {
793 info->status.store(task_info::status::running, memory_order_release);
794 info->start_time = timestamp::now();
795 info->worker_thread_id = get_worker_context() ? get_worker_context()->id : 0;
796 }
797
798 ~context_guard() noexcept {
799 info->finish_time = timestamp::now();
800 auto expected = task_info::status::running;
801 info->status.compare_exchange_strong(expected, task_info::status::completed,
802 memory_order_release);
803 }
804 };
805
806 context_guard guard(info);
807
808 try {
809 return _NEFORCE apply(func, tup);
810 } catch (const exception& e) {
811 info->status.store(task_info::status::failed, memory_order_release);
812 info->error = e.what();
813 throw;
814 }
815 });
816
817 future<Result> res = task->get_future();
818
819 auto expire_time = steady_clock::now() + milliseconds(delay_ms);
820 timer_.add_task(expire_time, [this, task = _NEFORCE move(task), priority]() mutable {
821 this->submit_task(priority, [task]() { (*task)(); });
822 });
823
824 return submit_result<Result>{_NEFORCE move(res), info};
825}

827template <typename Func, typename... Args>
828thread_pool::periodic_token thread_pool::submit_every(int64_t interval_ms, const priority_type priority, Func&& func,
829 Args&&... args) {
830 auto state = make_shared<periodic_task_state>();
831 auto task = _NEFORCE make_shared<function<void()>>(
832 [func = _NEFORCE forward<Func>(func),
833 tup = _NEFORCE make_tuple(_NEFORCE forward<Args>(args)...)]() mutable { _NEFORCE apply(func, tup); });
834 auto handler_ptr = _NEFORCE make_shared<task_type>();
835 *handler_ptr = [this, state, task, interval_ms, priority, handler_ptr]() {
836 if (state->cancelled.load()) {
837 return;
838 }
839
840 this->submit_task(priority, [task]() { (*task)(); });
841
842 if (state->cancelled.load()) {
843 return;
844 }
845 auto next_time = steady_clock::now() + milliseconds(interval_ms);
846 timer_.add_task(next_time, [handler_ptr]() { (*handler_ptr)(); });
847 };
848
849 auto first_time = steady_clock::now() + milliseconds(interval_ms);
850 timer_.add_task(first_time, [handler_ptr]() { (*handler_ptr)(); });
851 return state;
852}
