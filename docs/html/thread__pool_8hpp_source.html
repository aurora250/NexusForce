49struct task_group {
50 task_group() = default;
51 ~task_group() = default;
52
53 atomic<size_t> running_count{0};
54
55 void increment() noexcept {
56 running_count.fetch_add(1, memory_order_relaxed);
57 }
58
59 void decrement() noexcept {
60 if (running_count.fetch_sub(1, memory_order_release) == 1) {
61 running_count.notify_all();
62 }
63 }
64
65 void wait() const noexcept {
66 size_t count = running_count.load(memory_order_acquire);
67 while (count != 0) {
68 running_count.wait(count);
69 count = running_count.load(memory_order_acquire);
70 }
71 }

368template <typename Func, typename... Args, enable_if_t<is_invocable_v<Func, Args...>, int>>
369submit_result<invoke_result_t<Func, Args...>>
370thread_pool::submit_task(const priority_type priority, Func&& func, Args&&... args) {
371 using Result = decltype(func(_NEFORCE forward<Args>(args)...));
372
373 auto info = make_shared<task_info>(generate_task_id(), priority);
374
375 const auto current_group = get_current_task_group();
376 if (current_group) {
377 current_group->increment();
378 }
379
380 auto task = _NEFORCE make_shared<packaged_task<Result()>>(
381 [func = _NEFORCE forward<Func>(func),
382 args = _NEFORCE make_tuple(_NEFORCE forward<Args>(args)...),
383 group = current_group,
384 info]() mutable -> Result {
385 struct context_guard {
386 shared_ptr<task_info> info;
387 shared_ptr<task_group> group_inner;
388 shared_ptr<task_group> prev_group_inner;
389
390 explicit context_guard(shared_ptr<task_info> i, shared_ptr<task_group> g)
391 : info(move(i)), group_inner(move(g)) {
392 info->status.store(TASK_STATUS::RUNNING, memory_order_release);
393 info->start_time = timestamp::now();
394 info->worker_thread_id = get_worker_context() ? get_worker_context()->id : 0;
395
396 prev_group_inner = get_current_task_group();
397 get_current_task_group() = group_inner;
398 }
399
400 ~context_guard() noexcept {
401 try {
402 info->finish_time = timestamp::now();
403 auto expected = TASK_STATUS::RUNNING;
404 info->status.compare_exchange_strong(expected,
405 TASK_STATUS::COMPLETED, memory_order_release);
406
407 get_current_task_group() = prev_group_inner;
408 if (group_inner) group_inner->decrement();
409 } catch (...) {
410 /* ignore */
411 }
412 }
413 };
414
415 context_guard guard(info, group);
416 try {
417 return _NEFORCE apply(func, args);
418 } catch (const exception& e) {
419 info->status.store(TASK_STATUS::FAILED, memory_order_release);
420 info->error = e.what();
421 throw;
422 } catch (...) {
423 info->status.store(TASK_STATUS::FAILED, memory_order_release);
424 info->error = "Unknown exception";
425 throw;
426 }
427 }
428 );
429
430 future<Result> res = task->get_future();
431 task_type job([task] { (*task)(); });
432
433 if (priority > 0) {
434 smart_lock<mutex> lock(task_queue_mtx_);
435
436 if (!not_full_.wait_for(lock, seconds(1), [&]()->bool {
437 return task_queue_.size() < task_threshhold_;
438 })) {
439 info->status.store(TASK_STATUS::FAILED, memory_order_release);
440 info->error = "Task queue is full";
441
442 auto dummy_task = _NEFORCE make_shared<packaged_task<Result()>>(
443 []() -> Result { return Result(); });
444 (*dummy_task)();
445 return submit_result<Result>{dummy_task->get_future(), info};
446 }
447
448 task_queue_.emplace(move(job), priority, info);
449 ++task_size_;
450 ++total_submitted_tasks_;
451 not_empty_.notify_one();
452
453 } else {
454 auto* ctx = get_worker_context();
455
456 if (ctx != nullptr && ctx->queue.remain_size() > 0) {
457 ctx->queue.push_back(move(job));
458 ++total_submitted_tasks_;
459 } else {
460 smart_lock<mutex> lock(task_queue_mtx_);
461 if (!not_full_.wait_for(lock, seconds(1), [&]()->bool {
462 return task_queue_.size() < task_threshhold_;
463 })) {
464 info->status.store(TASK_STATUS::FAILED, memory_order_release);
465 info->error = "Task queue is full";
466
467 auto dummy_task = _NEFORCE make_shared<packaged_task<Result()>>(
468 []() -> Result { return Result(); });
469 (*dummy_task)();
470 return submit_result<Result>{dummy_task->get_future(), info};
471 }
472
473 task_queue_.emplace(move(job), 0, info);
474 ++task_size_;
475 ++total_submitted_tasks_;
476 not_empty_.notify_one();
477 }
478 }
479
480 if (pool_mode_.load() == THREAD_POOL_MODE::MODE_CACHED
481 && task_size_.load() > idle_thread_size_
482 && threads_map_.size() < thread_threshhold_) {
483
484 auto ptr = _NEFORCE make_unique<_INNER manual_thread>(
485 [this](const id_type id) {
486 thread_function(id);
487 });
488 id_type thread_id = ptr->id();
489
490 if (thread_id >= worker_contexts_ptr_.size()) {
491 worker_contexts_ptr_.reserve(thread_id + 1);
492 for (size_t i = worker_contexts_ptr_.size() - 1; i <= thread_id; i++) {
493 atomic<worker_context*> tmp;
494 tmp.store(nullptr, memory_order_relaxed);
495 worker_contexts_ptr_.emplace_back(move(tmp));
496 }
497 }
498
499 threads_map_.emplace(thread_id, _NEFORCE move(ptr));
500 threads_map_[thread_id]->start();
501 ++idle_thread_size_;
502 }
503
504 return submit_result<Result>{_NEFORCE move(res), info};
505}

559thread_pool::periodic_token thread_pool::submit_every(int64_t interval_ms, const priority_type priority, Func &&func, Args &&...args) {
560 auto state = make_shared<periodic_task_state>();
561 auto task = _NEFORCE make_shared<function<void()>>(
562 [func = _NEFORCE forward<Func>(func), tup = _NEFORCE make_tuple(_NEFORCE forward<Args>(args)...)]() mutable {
563 _NEFORCE apply(func, tup);
564 }
565 );
566 auto handler_ptr = _NEFORCE make_shared<task_type>();
567 *handler_ptr = [this, state, task, interval_ms, priority, handler_ptr]() {
568 if (state->cancelled.load()) return;
569
570 this->submit_task(priority, [task]() {
571 (*task)();
572 });
573
574 if (state->cancelled.load()) return;
575 auto next_time = steady_clock::now() + milliseconds(interval_ms);
576 timer_.add_task(next_time, [handler_ptr]() { (*handler_ptr)(); });
577 };
578
579 auto first_time = steady_clock::now() + milliseconds(interval_ms);
580 timer_.add_task(first_time, [handler_ptr]() { (*handler_ptr)(); });
