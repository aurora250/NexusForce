508template <typename T>
509void any::internal_manage<T>::manage(const any_operation op, const any* value, ArgT* arg) {
510 auto ptr = reinterpret_cast<const T*>(&value->storage_.buffer_);
511 switch (op) {
512 case ACCESS: {
513 arg->obj_ptr_ = const_cast<T*>(ptr);
514 break;
515 }
516 case GET_TYPE_INFO: {
517 arg->type_ptr_ = &typeid(T);
518 break;
519 }
520 case COPY: {
521 ::new(&arg->any_ptr_->storage_.buffer_) T(*ptr);
522 arg->any_ptr_->manage_ = value->manage_;
523 break;
524 }
525 case DESTROY: {
526 ptr->~T();
527 break;
528 }
529 case SWAP: {
530 ::new(&arg->any_ptr_->storage_.buffer_) T(_NEFORCE move(*const_cast<T*>(ptr)));
531 ptr->~T();
532 arg->any_ptr_->manage_ = value->manage_;
533 const_cast<any*>(value)->manage_ = nullptr;
534 break;
535 }
536 default: {
537 NEFORCE_UNREACHABLE;
538 }
539 }
