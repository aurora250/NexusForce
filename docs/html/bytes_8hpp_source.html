175NEFORCE_ALWAYS_INLINE_INLINE int find_last_byte(vec128_t v, byte_t c) noexcept {
176#ifdef NEFORCE_SIMD_SSE2
177 const int mask = ::_mm_movemask_epi8(::_mm_cmpeq_epi8(v, ::_mm_set1_epi8(static_cast<char>(c))));
178 if (mask == 0) {
179 return -1;

204#endif

213NEFORCE_ALWAYS_INLINE_INLINE int count_byte(vec128_t v, byte_t c) noexcept {
214#ifdef NEFORCE_SIMD_SSE2
215 const ::__m128i match = ::_mm_cmpeq_epi8(v, ::_mm_set1_epi8(static_cast<char>(c)));
216 const ::__m128i ones = ::_mm_and_si128(match, ::_mm_set1_epi8(1));
217 const ::__m128i sums = ::_mm_sad_epu8(ones, ::_mm_setzero_si128());
218 return _mm_extract_epi16(sums, 0) + _mm_extract_epi16(sums, 4);
219#elif defined(NEFORCE_SIMD_NEON)
220 const uint8x16_t match = ::vceqq_u8(v, ::vdupq_n_u8(c));
221 const uint8x16_t ones = ::vandq_u8(match, ::vdupq_n_u8(1));
222# ifdef NEFORCE_ARCH_AARCH64
223 return static_cast<int>(::vaddvq_u8(ones));
224# else
225 int count = 0;
226 const auto* bytes = reinterpret_cast<const byte_t*>(&ones);
227 for (int i = 0; i < 16; ++i) {
228 count += bytes[i];
229 }
230 return count;
231# endif
232#else
233 int count = 0;
234 for (int i = 0; i < 16; ++i) {
235 if (v.data[i] == c) {
236 ++count;
237 }
238 }
239 return count;
240#endif
241}

320NEFORCE_ALWAYS_INLINE_INLINE vec128_t shuffle_bytes(vec128_t v, vec128_t indices) noexcept {
321#ifdef NEFORCE_SIMD_SSSE3
322 return ::_mm_shuffle_epi8(v, indices);
323#elif defined(NEFORCE_SIMD_SSE2)
324 alignas(16) byte_t src[16];
325 alignas(16) byte_t idx[16];
326 ::_mm_store_si128(reinterpret_cast<::__m128i*>(src), v);
327 ::_mm_store_si128(reinterpret_cast<::__m128i*>(idx), indices);
328 alignas(16) byte_t result[16] = {};

334#elif defined(NEFORCE_SIMD_NEON)
335 return ::vqtbl1q_u8(v, ::vandq_u8(indices, ::vdupq_n_u8(0x0F)));
336#else

353NEFORCE_ALWAYS_INLINE_INLINE vec128_t blend_bytes(vec128_t a, vec128_t b, vec128_t mask) noexcept {
354#ifdef NEFORCE_SIMD_SSE4_1
355 return ::_mm_blendv_epi8(a, b, mask);
356#elif defined(NEFORCE_SIMD_SSE2)
357 const ::__m128i sel = ::_mm_cmpgt_epi8(::_mm_setzero_si128(), mask);
358 return ::_mm_or_si128(::_mm_and_si128(sel, b), ::_mm_andnot_si128(sel, a));
359#elif defined(NEFORCE_SIMD_NEON)
360 const int8x16_t sel = ::vshrq_n_s8(::vreinterpretq_s8_u8(mask), 7);
361 return ::vbslq_u8(::vreinterpretq_u8_s8(sel), b, a);
362#else
363 vec128_t result;
364 for (int i = 0; i < 16; ++i) {
365 result.data[i] = (mask.data[i] & 0x80) ? b.data[i] : a.data[i];
366 }
367 return result;
368#endif
369}
