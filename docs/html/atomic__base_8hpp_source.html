546NEFORCE_ALWAYS_INLINE_INLINE bool atomic_cmpexch_weak(volatile T* ptr, remove_volatile_t<T>* expected,
547 remove_volatile_t<T> desired, const memory_order success,

549 static_assert(is_integral_v<T>, "T must be integral type");
550 NEFORCE_CONSTEXPR_ASSERT(is_valid_cmpexch_failure_order(failure));
551#ifdef NEFORCE_COMPILER_GNUC
552 return __atomic_compare_exchange_n(ptr, expected, desired, true, static_cast<int32_t>(success),
553 static_cast<int32_t>(failure));
554#else
555# if defined(NEFORCE_ARCH_X86) || defined(NEFORCE_ARCH_AARCH64)
556 const bool result = inner::interlocked_compare_exchange_impl<sizeof(T)>::call(ptr, expected, desired);
557 if (success == memory_order_seq_cst || failure == memory_order_seq_cst) {
558 ::_ReadWriteBarrier();

560 return result;
561# else

563 remove_volatile_t<T> loaded;
564 bool success_flag;
565# if defined(NEFORCE_ARCH_ARM)
566 NEFORCE_IF_CONSTEXPR(sizeof(T) == 1) {
567 asm volatile("ldrexb %[loaded], [%[ptr]]\n\t"

569 "bne 1f\n\t"
570 "strexb %w[success], %w[desired], [%[ptr]]\n\t"
571 "1:"
