103constexpr memory_order cmpexch_failure_order(const memory_order mo) noexcept {
104 constexpr auto mask = memory_order_modifier::memory_order_mask;
105 const memory_order base_mo = mo & mask;
106 const memory_order failure_order = base_mo == memory_order_acq_rel ? memory_order_acquire
107 : base_mo == memory_order_release ? memory_order_relaxed
108 : base_mo;
109 constexpr auto modifier_mask = memory_order_modifier::memory_order_modifier_mask;
110 const auto modifiers = static_cast<memory_order_modifier>(mo & modifier_mask);
111 return failure_order | modifiers;
112}

121 return (mo & memory_order_modifier::memory_order_mask) != memory_order_release &&
122 (mo & memory_order_modifier::memory_order_mask) != memory_order_acq_rel;

563NEFORCE_ALWAYS_INLINE_INLINE bool atomic_cmpexch_weak(volatile T* ptr, remove_volatile_t<T>* expected,
564 remove_volatile_t<T> desired, const memory_order success,
565 const memory_order failure) noexcept {
566 static_assert(is_integral_v<T>, "T must be integral type");
567 NEFORCE_CONSTEXPR_ASSERT(is_valid_cmpexch_failure_order(failure));
568#ifdef NEFORCE_COMPILER_GNUC
569 return __atomic_compare_exchange_n(ptr, expected, desired, true, static_cast<int32_t>(success),
570 static_cast<int32_t>(failure));
571#else
572# if defined(NEFORCE_ARCH_X86)
573 const bool result = inner::interlocked_compare_exchange_impl<sizeof(T)>::call(ptr, expected, desired);
574 if (success == memory_order_seq_cst || failure == memory_order_seq_cst) {
575 ::_ReadWriteBarrier();
576 }
577 return result;
578# else
579 remove_volatile_t<T> old_val = *expected;
580 remove_volatile_t<T> loaded;
581 bool success_flag;
582# if defined(NEFORCE_ARCH_ARM)
583 NEFORCE_IF_CONSTEXPR(sizeof(T) == 1) {
584 asm volatile("ldrexb %[loaded], [%[ptr]]\n\t"
585 "cmp %[loaded], %[old_val]\n\t"
586 "bne 1f\n\t"
587 "strexb %w[success], %w[desired], [%[ptr]]\n\t"
588 "1:"
589 : [loaded] "=&r"(loaded), [success] "=&r"(success_flag)
590 : [ptr] "r"(ptr), [old_val] "r"(old_val), [desired] "r"(desired)
591 : "cc", "memory");
592 }
593 else NEFORCE_IF_CONSTEXPR(sizeof(T) == 2) {
594 asm volatile("ldrexh %[loaded], [%[ptr]]\n\t"
595 "cmp %[loaded], %[old_val]\n\t"
596 "bne 1f\n\t"
597 "strexh %w[success], %w[desired], [%[ptr]]\n\t"
598 "1:"
599 : [loaded] "=&r"(loaded), [success] "=&r"(success_flag)
600 : [ptr] "r"(ptr), [old_val] "r"(old_val), [desired] "r"(desired)
601 : "cc", "memory");
602 }
603 else NEFORCE_IF_CONSTEXPR(sizeof(T) == 4) {
604 asm volatile("ldrex %[loaded], [%[ptr]]\n\t"
605 "cmp %[loaded], %[old_val]\n\t"
606 "bne 1f\n\t"
607 "strex %w[success], %w[desired], [%[ptr]]\n\t"
608 "1:"

610 : [ptr] "r"(ptr), [old_val] "r"(old_val), [desired] "r"(desired)
611 : "cc", "memory");
612 }

614 uint32_t loaded_lo, loaded_hi;
615 uint32_t old_lo = static_cast<uint32_t>(old_val);
616 uint32_t old_hi = static_cast<uint32_t>(static_cast<uint64_t>(old_val) >> 32);
617 uint32_t des_lo = static_cast<uint32_t>(static_cast<uint64_t>(desired));
618 uint32_t des_hi = static_cast<uint32_t>(static_cast<uint64_t>(desired) >> 32);

633 *expected = loaded;

637 }

641 "bne %[loaded], %[old_val], 1f\n\t"
642 "sc.w %[success], %[desired], (%[ptr])\n\t"
643 "1:"
644 : [loaded] "=&r"(loaded), [success] "=&r"(success_flag)
645 : [ptr] "r"(ptr), [old_val] "r"(old_val), [desired] "r"(desired)

647 }

649 asm volatile("lr.d %[loaded], (%[ptr])\n\t"
650 "bne %[loaded], %[old_val], 1f\n\t"
651 "sc.d %[success], %[desired], (%[ptr])\n\t"

653 : [loaded] "=&r"(loaded), [success] "=&r"(success_flag)
654 : [ptr] "r"(ptr), [old_val] "r"(old_val), [desired] "r"(desired)
655 : "memory");
656 }

662 "bne %[loaded], %[old_val], 1f\n\t"

683 *expected = loaded;

685 }
686 return success_flag == 0;
687# endif
688#endif
