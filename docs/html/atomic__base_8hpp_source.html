103constexpr memory_order cmpexch_failure_order(const memory_order mo) noexcept {
104 constexpr auto mask = memory_order_modifier::memory_order_mask;
105 const memory_order base_mo = mo & mask;
106 const memory_order failure_order =
107 base_mo == memory_order_acq_rel ? memory_order_acquire
108 : base_mo == memory_order_release ? memory_order_relaxed
109 : base_mo;
110 constexpr auto modifier_mask = memory_order_modifier::memory_order_modifier_mask;
111 const auto modifiers = static_cast<memory_order_modifier>(mo & modifier_mask);
112 return failure_order | modifiers;
113}

122 return (mo & memory_order_modifier::memory_order_mask) != memory_order_release
123 && (mo & memory_order_modifier::memory_order_mask) != memory_order_acq_rel;

583 static_assert(is_integral_v<T>, "T must be integral type");

585#ifdef NEFORCE_COMPILER_GNUC
586 return __atomic_compare_exchange_n(
587 ptr, expected, desired, true,
588 static_cast<int32_t>(success), static_cast<int32_t>(failure));
589#else
590#if defined(NEFORCE_ARCH_X86)
591 const bool result = inner::interlocked_compare_exchange_impl<sizeof(T)>::call(ptr, expected, desired);
592 if (success == memory_order_seq_cst || failure == memory_order_seq_cst) {
593 ::_ReadWriteBarrier();
594 }
595 return result;
596#else
597 remove_volatile_t<T> old_val = *expected;
598 remove_volatile_t<T> loaded;
599 bool success_flag;
600#if defined(NEFORCE_ARCH_ARM)
601 NEFORCE_IF_CONSTEXPR (sizeof(T) == 1) {
602 asm volatile(
603 "ldrexb %[loaded], [%[ptr]]\n\t"
604 "cmp %[loaded], %[old_val]\n\t"
605 "bne 1f\n\t"
606 "strexb %w[success], %w[desired], [%[ptr]]\n\t"
607 "1:"
608 : [loaded] "=&r" (loaded), [success] "=&r" (success_flag)

616 "strexh %w[success], %w[desired], [%[ptr]]\n\t"

620 : "cc", "memory");
