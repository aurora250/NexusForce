527NEFORCE_ALWAYS_INLINE_INLINE bool atomic_cmpexch_weak(volatile T* ptr, remove_volatile_t<T>* expected,
528 remove_volatile_t<T> desired, const memory_order success,
529 const memory_order failure) noexcept {
530 static_assert(is_integral_v<T>, "T must be integral type");
531 NEFORCE_CONSTEXPR_ASSERT(is_valid_cmpexch_failure_order(failure));
532#ifdef NEFORCE_COMPILER_GNUC
533 return __atomic_compare_exchange_n(ptr, expected, desired, true, static_cast<int32_t>(success),
534 static_cast<int32_t>(failure));
535#else
536# if defined(NEFORCE_ARCH_X86)
537 const bool result = inner::interlocked_compare_exchange_impl<sizeof(T)>::call(ptr, expected, desired);
538 if (success == memory_order_seq_cst || failure == memory_order_seq_cst) {
539 ::_ReadWriteBarrier();
540 }
541 return result;
542# else
543 remove_volatile_t<T> old_val = *expected;
544 remove_volatile_t<T> loaded;
545 bool success_flag;
546# if defined(NEFORCE_ARCH_ARM)
547 NEFORCE_IF_CONSTEXPR(sizeof(T) == 1) {
548 asm volatile("ldrexb %[loaded], [%[ptr]]\n\t"

550 "bne 1f\n\t"
551 "strexb %w[success], %w[desired], [%[ptr]]\n\t"
552 "1:"
553 : [loaded] "=&r"(loaded), [success] "=&r"(success_flag)
