365 template <typename Rep, typename Period>
366 future_status wait_for(const duration<Rep, Period>& relative_time) {
367 if (status.load(memory_order_acquire) == status::ready) {
368 return future_status::ready;
369 }
370
371 if (is_deferred_future()) {
372 return future_status::deferred;
373 }
374
375 if (relative_time > relative_time.zero() &&
376 status.load_when_equal_for(status::ready, memory_order_acquire, relative_time)) {
377 complete_async();
378 return future_status::ready;
379 }
380 return future_status::timeout;
381 }

390 template <typename Clock, typename Dur>
391 future_status wait_until(const time_point<Clock, Dur>& absolute_time) {
392 static_assert(is_clock_v<Clock>, "Clock type must be clock_t");
393 if (status.load(memory_order_acquire) == status::ready) {
394 return future_status::ready;
395 }
396
397 if (is_deferred_future()) {
398 return future_status::deferred;
399 }
400
401 if (status.load_when_equal_until(status::ready, memory_order_acquire, absolute_time)) {
402 complete_async();
403 return future_status::ready;
404 }
405 return future_status::timeout;
