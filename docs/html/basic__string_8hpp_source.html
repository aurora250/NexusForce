288
289#ifdef NEFORCE_USING_SSO

410 NEFORCE_CONSTEXPR20 basic_string& replace_fill(iterator first, size_type n1, const size_type n2, const value_type value) {
411#ifdef NEFORCE_USING_SSO
412 const size_type offset = first - begin();
413 const size_type old_size = size();
414 const size_type actual_n1 = _NEFORCE min(n1, old_size - offset);
415 if (actual_n1 == 0 && n2 == 0) return *this;
416
417 const size_type new_size = old_size - actual_n1 + n2;
418
419 if (!is_long() && new_size < sso_capacity) {
420 pointer p = storage_.short_ + offset;
421 if (old_size - offset - actual_n1 > 0) {
422 traits_type::move(p + n2, p + actual_n1, old_size - offset - actual_n1);
423 }
424 traits_type::assign(p, n2, value);
425 size_pair_.value = new_size;
426 traits_type::assign(storage_.short_ + new_size, 1, value_type());
427 return *this;
428 }
429
430 size_type new_cap = is_long() ? storage_.long_.cap : sso_buffer_size;
431 if (new_cap < new_size + 1) {
432 new_cap = _NEFORCE max(new_size + 1, new_cap + (new_cap >> 1));
433 }
434
435 pointer new_ptr = size_pair_.get_base().allocate(new_cap);
436 pointer dest = new_ptr;
437
438 dest = traits_type::copy(dest, data(), offset) + offset;
439 dest = traits_type::assign(dest, n2, value) + n2;
440 traits_type::copy(dest, data() + offset + actual_n1, old_size - offset - actual_n1);
441
442 if (is_long()) destroy_long();
443 storage_.long_.ptr = new_ptr;
444 storage_.long_.cap = new_cap;
445 size_pair_.value = new_size | long_flag;
446 traits_type::assign(storage_.long_.ptr + new_size, 1, value_type());
447
448 return *this;
449#else
450 if (static_cast<size_type>(end() - first) < n1) {
451 n1 = cend() - first;
452 }
453
454 if (n1 < n2) {
455 const size_type diff = n2 - n1;
456 NEFORCE_DEBUG_VERIFY(size_ + diff < max_size(), "basic_string index out of range.");
457 if (size_ > capacity_pair_.value - diff) {
458 reallocate(diff);
459 }
460
461 pointer raw_ptr = &*first;
462 traits_type::move(raw_ptr + n2, raw_ptr + n1, end() - (first + n1));
463 traits_type::assign(raw_ptr, n2, value);
464 size_ += diff;
465 }
466 else {
467 pointer raw_ptr = &*first;
468 traits_type::move(raw_ptr + n2, raw_ptr + n1, end() - (first + n1));
469 traits_type::assign(raw_ptr, n2, value);
470 size_ -= n1 - n2;
471 }
472
473 traits_type::assign(data_ + size_, 1, value_type());
474 return *this;
475#endif
