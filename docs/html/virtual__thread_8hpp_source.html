131class virtual_thread_scheduler {
132private:
133 queue<coroutine_handle<>> task_queue_;
134 vector<thread> workers_;
135 mutex mutex_;
136 condition_variable cv_;
137 atomic<bool> shutdown_;
138
142 virtual_thread_scheduler() :
143 shutdown_(false) {}
144
150 void worker_loop() {
151 while (true) {
152 coroutine_handle<> handle;
153
154 {
155 unique_lock<mutex> lock(mutex_);
156 cv_.wait(lock, [this] { return shutdown_ || !task_queue_.empty(); });
157
158 if (shutdown_ && task_queue_.empty()) {
159 return;
160 }
161
162 if (!task_queue_.empty()) {
163 handle = task_queue_.front();
164 task_queue_.pop();
165 }
166 }
167
168 if (handle) {
169 handle.resume();
170 if (handle.done()) {
171 handle.destroy();
172 }
173 }
174 }
175 }

177public:
182 static virtual_thread_scheduler& get_instance() {
183 static virtual_thread_scheduler instance;
184 return instance;
185 }
186
193 void schedule(coroutine_handle<> handle) {
194 {
195 lock<mutex> lock(mutex_);
196 task_queue_.push(handle);
197 }
198 cv_.notify_one();
199 }
200
207 void start_workers(size_t num_threads) {
208 for (size_t i = 0; i < num_threads; ++i) {
209 workers_.emplace_back([this] { worker_loop(); });
210 }
211 }

218 void shutdown() {
219 {
220 lock<mutex> lock(mutex_);
221 shutdown_ = true;
222 }
223 cv_.notify_all();

225 for (auto& worker: workers_) {
226 if (worker.joinable()) {
227 worker.join();
228 }
229 }
230 }
