146 void worker_loop() {
147 while (true) {
148 coroutine_handle<> handle;
149
150 {
151 unique_lock<mutex> lock(mutex_);
152 cv_.wait(lock, [this] { return shutdown_ || !task_queue_.empty(); });

154 if (shutdown_ && task_queue_.empty()) {
155 return;
156 }
157
158 if (!task_queue_.empty()) {
159 handle = task_queue_.front();
160 task_queue_.pop();
161 }
162 }

167 }
