120class virtual_thread_scheduler {
121private:
122 queue<coroutine_handle<>> task_queue_;
123 vector<thread> workers_;
124 mutex mutex_;
125 condition_variable cv_;
126 atomic<bool> shutdown_{false};
127
128public:
133 static virtual_thread_scheduler& get_instance() {
134 static virtual_thread_scheduler instance;
135 return instance;
136 }
137
142 void schedule(coroutine_handle<> handle) {
143 {
144 lock<mutex> lock(mutex_);
145 task_queue_.push(handle);
146 }
147 cv_.notify_one();
148 }
149
154 void start_workers(size_t num_threads) {
155 for (size_t i = 0; i < num_threads; ++i) {
156 workers_.emplace_back([this] { worker_loop(); });
157 }
158 }
159
165 void shutdown() {
166 {
167 lock<mutex> lock(mutex_);
168 shutdown_ = true;
169 }
170 cv_.notify_all();
171
172 for (auto& worker: workers_) {
173 if (worker.joinable()) {
174 worker.join();
175 }
176 }
177 }
178
182 ~virtual_thread_scheduler() { shutdown(); }
183
184private:
191 void worker_loop() {
192 while (true) {
193 coroutine_handle<> handle;
194
195 {
196 unique_lock<mutex> lock(mutex_);
197 cv_.wait(lock, [this] { return shutdown_ || !task_queue_.empty(); });
198
199 if (shutdown_ && task_queue_.empty()) {
200 return;
201 }
202
203 if (!task_queue_.empty()) {
204 handle = task_queue_.front();
205 task_queue_.pop();
206 }
207 }
208
209 if (handle) {
210 handle.resume();
211 if (handle.done()) {
212 handle.destroy();
213 }
214 }
215 }
216 }
