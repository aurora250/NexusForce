40public:
41 atomic<unsigned long> strong_count_;
42 atomic<unsigned long> weak_count_;

44protected:

55public:
60 __smart_ptr_counter() noexcept :
61 strong_count_(1),
62 weak_count_(1) {}

64 __smart_ptr_counter(__smart_ptr_counter&&) = delete;
65
66 virtual ~__smart_ptr_counter() = default;

71 void incref_strong() noexcept { strong_count_.fetch_add(1, memory_order_relaxed); }
72
76 void incref_weak() noexcept { weak_count_.fetch_add(1, memory_order_relaxed); }
77
82 void decref_strong() noexcept {
83 if (strong_count_.fetch_sub(1, memory_order_acq_rel) == 1) {
84 delete_object();
85 decref_weak();
86 }
87 }
88
93 void decref_weak() noexcept {
94 if (weak_count_.fetch_sub(1, memory_order_acq_rel) == 1) {
95 delete_this();
96 }
97 }
98
105 bool try_incref_strong() noexcept {
106 auto strong = strong_count_.load(memory_order_relaxed);
107 do {
108 if (strong == 0) {
109 return false;
110 }
111 } while (!strong_count_.compare_exchange_weak(strong, strong + 1, memory_order_release, memory_order_relaxed));
112 return true;
113 }
114
119 NEFORCE_NODISCARD uint64_t use_count() const noexcept { return strong_count_.load(memory_order_relaxed); }

125 NEFORCE_NODISCARD uint64_t weak_count() const noexcept { return weak_count_.load(memory_order_relaxed); }

848template <typename T, typename... Args>
849enable_if_t<!is_unbounded_array_v<T> && is_constructible_v<T, Args...>, shared_ptr<T>> make_shared(Args&&... args) {
850 auto const deleter = [](T* ptr) noexcept(is_nothrow_destructible_v<T>) { ptr->~T(); };
851 using Counter = inner::__smart_ptr_counter_impl_fused<T, decltype(deleter)>;
852 constexpr size_t align = max(alignof(T), alignof(Counter));
853 constexpr size_t offset = (sizeof(Counter) + align - 1) & ~(align - 1);
854 constexpr size_t size = offset + sizeof(T);
855#if NEFORCE_STANDARD_17
856 void* mem = operator new(size, std::align_val_t{align});
857 auto* counter = static_cast<Counter*>(mem);
858#else
859 void* mem = operator new(size + align - 1);
860 size_t aligned_addr = (reinterpret_cast<size_t>(mem) + (align - 1)) & ~(align - 1);
861 Counter* counter = reinterpret_cast<Counter*>(aligned_addr);
862#endif
863 T* object = reinterpret_cast<T*>(reinterpret_cast<byte_t*>(counter) + offset);
864 try {
865 _NEFORCE construct(object, _NEFORCE forward<Args>(args)...);
866 } catch (...) {
867#if NEFORCE_STANDARD_17
868 operator delete(mem, std::align_val_t{align});
869#else
870 operator delete(mem);
871#endif
872 NEFORCE_THROW_EXCEPTION(memory_exception("shared ptr construction failed."));
873 }
874 _NEFORCE construct(reinterpret_cast<Counter*>(counter), object, mem, align, _NEFORCE move(deleter));
875 inner::__setup_enable_shared_from(object, counter);

912template <typename T, typename Alloc, typename... Args>
913enable_if_t<!is_array_v<T> && is_constructible_v<T, Args...>, shared_ptr<T>> allocate_shared(Alloc& alloc,
914 Args&&... args) {
915 auto deleter = [](T* p) { p->~T(); };
916 using ControlBlock = inner::__smart_ptr_counter_impl_allocated<T, decltype(deleter), Alloc>;
917
918 const size_t align = _NEFORCE max(alignof(ControlBlock), alignof(T));
919 const size_t offset = (sizeof(ControlBlock) + align - 1) & ~(align - 1);
920 const size_t total_size = offset + sizeof(T);
921 const size_t raw_size = total_size + align - 1;
922
923 using alloc_traits = allocator_traits<remove_cv_t<Alloc>>;
924 using byte_allocator = typename alloc_traits::template alloc_rebind_t<Alloc, byte_t>;
925 byte_allocator byte_alloc(alloc);
926
927 byte_t* raw_mem = allocator_traits<byte_allocator>::allocate(byte_alloc, raw_size);
928
929 const auto raw_addr = reinterpret_cast<uintptr_t>(raw_mem);
930 const uintptr_t aligned_addr = (raw_addr + align - 1) & ~static_cast<uintptr_t>(align - 1);
931 auto* aligned_mem = reinterpret_cast<byte_t*>(aligned_addr);
932 T* object_ptr = reinterpret_cast<T*>(aligned_mem + offset);
933
934 try {
935 allocator_traits<Alloc>::construct(alloc, object_ptr, _NEFORCE forward<Args>(args)...);
936 } catch (...) {
937 allocator_traits<byte_allocator>::deallocate(byte_alloc, raw_mem, raw_size);
938 NEFORCE_THROW_EXCEPTION(memory_exception("shared ptr ref object construction failed."));
939 }
940
941 ControlBlock* ctrl_block = nullptr;
942 try {
943 ctrl_block = _NEFORCE construct(reinterpret_cast<ControlBlock*>(aligned_mem), object_ptr, raw_mem, raw_size,
944 deleter, alloc);
945 } catch (...) {
946 allocator_traits<Alloc>::destroy(alloc, object_ptr);
947 allocator_traits<byte_allocator>::deallocate(byte_alloc, raw_mem, raw_size);
948 NEFORCE_THROW_EXCEPTION(memory_exception("shared ptr control block construction failed."));
949 }
950
951 inner::__setup_enable_shared_from(object_ptr, ctrl_block);
952 return inner::__make_shared_fused(object_ptr, ctrl_block);
953}
