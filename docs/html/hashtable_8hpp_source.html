536 pair<iterator, bool> insert_unique_noresize(link_type ptr) {
537 const size_type n = hashtable::bucket_index_value(ptr->data, buckets_.size());
538
539 node_type** buckets_ptr = &buckets_[n];
540 while (*buckets_ptr != nullptr) {
541 if (equals_(extracter_((*buckets_ptr)->data), extracter_(ptr->data))) {
542 ptr->next = (*buckets_ptr)->next;
543 hashtable::delete_node(*buckets_ptr);
544 *buckets_ptr = ptr;
545 return {{ptr, n, this}, false};
546 }
547 buckets_ptr = &(*buckets_ptr)->next;
548 }
549 ptr->next = nullptr;
550 *buckets_ptr = ptr;
551 ++size_;
552 return {iterator{ptr, n, this}, true};
553 }

560 iterator insert_equal_noresize(link_type ptr) {
561 const size_type n = hashtable::bucket_index_value(ptr->data, buckets_.size());
562 link_type first = buckets_[n];
563
564 link_type prev = nullptr;
565 link_type cur = first;
566 while (cur != nullptr && equals_(extracter_(cur->data), extracter_(ptr->data))) {
567 prev = cur;
568 cur = cur->next;
569 }
570
571 if (prev != nullptr) {
572 prev->next = ptr;
573 ptr->next = cur;
574 } else {
575 ptr->next = first;
576 buckets_[n] = ptr;
577 }
578
579 ++size_;
580 return {ptr, n, this};
581 }

680 size_type erase_bucket_range(size_type bucket, link_type first, link_type last) noexcept {
681 size_type count = 0;
682 if (first == nullptr) {
683 return 0;

699 link_type next = curr->next;

732 bool equal_small(const hashtable& rhs) const {
733 for (const_iterator iter = begin(); iter != end(); ++iter) {
734 const key_type& key = extracter_(*iter);
735
736 const size_t count_lhs = _NEFORCE count_if(
737 begin(), end(), [this, &key](const value_type& val) { return equals_(extracter_(val), key); });
738 const size_t count_rhs = _NEFORCE count_if(rhs.begin(), rhs.end(), [&rhs, &key](const value_type& val) {
739 return rhs.equals_(rhs.extracter_(val), key);
740 });
741
742 if (count_lhs != count_rhs) {
743 return false;
744 }
745 }
746 return true;
747 }

754 bool equal_large(const hashtable& rhs) const {
755 vector<value_type> elements_lhs, elements_rhs;
756 elements_lhs.reserve(size_);
757 elements_rhs.reserve(size_);
758
759 for (const_iterator it = begin(); it != end(); ++it) {
760 elements_lhs.push_back(*it);
761 }
762 for (const_iterator it = rhs.begin(); it != rhs.end(); ++it) {
763 elements_rhs.push_back(*it);
764 }
765
766 _NEFORCE sort(elements_lhs.begin(), elements_lhs.end());
767 _NEFORCE sort(elements_rhs.begin(), elements_rhs.end());
768
769 return elements_lhs == elements_rhs;
770 }

1026 void rehash(const size_type new_size) {
1027 const auto min_buckets_for_size =
1028 static_cast<size_type>(_NEFORCE ceil(static_cast<double>(size_) / max_load_factor()));
1029 const size_type target = _NEFORCE max(new_size, min_buckets_for_size);
1030 const size_type old_size = buckets_.size();
1031
1032 if (target <= old_size) {
1033 return;
1034 }
1035
1036 const size_type n = hashtable::next_size(target);
1037 if (n < target) {
1038 NEFORCE_THROW_EXCEPTION(value_exception("hashtable size exceeds max count"));
1039 }
1040
1041 vector<link_type> new_buckets(n, nullptr);
1042
1043 for (size_type bucket = 0; bucket < old_size; ++bucket) {
1044 link_type cur = buckets_[bucket];
1045 while (cur != nullptr) {
1046 link_type next = cur->next;
1047 const size_type new_bucket = hashtable::bucket_index_value(cur->data, n);
1048
1049 cur->next = new_buckets[new_bucket];
1050 new_buckets[new_bucket] = cur;
1051
1052 cur = next;
1053 }
1054 buckets_[bucket] = nullptr;
1055 }
1056
1057 buckets_.swap(new_buckets);
1058 }

1218 iterator erase(const iterator& position) noexcept(is_nothrow_hashable_v<key_type>) {
1219 if (position.current_ == nullptr || position.container_ != this) {
1220 return hashtable::end();
1221 }
1222
1223 const size_type n = position.bucket_;
1224 link_type const p = position.current_;
1225 link_type next_node = p->next;
1226
1227 link_type prev = nullptr;
1228 link_type curr = buckets_[n];
1229 while (curr != nullptr && curr != p) {
1230 prev = curr;
1231 curr = curr->next;
1232 }
1233
1234 if (curr == nullptr) {
1235 return hashtable::end();
1236 }
1237
1238 if (prev == nullptr) {
1239 buckets_[n] = next_node;
1240 } else {
1241 prev->next = next_node;
1242 }
1243
1244 hashtable::delete_node(p);
1245 --size_;
1246
1247 if (next_node != nullptr) {
1248 return iterator(next_node, n, this);
1249 }
1250
1251 for (size_type bucket = n + 1; bucket < buckets_.size(); ++bucket) {
1252 if (buckets_[bucket] != nullptr) {
1253 return iterator(buckets_[bucket], bucket, this);

1268 }

1271 return hashtable::end();

1275

1278 } else {

1280 for (size_type bucket = first.bucket_ + 1; bucket < last.bucket_; ++bucket) {
1281 count_erased += hashtable::erase_bucket_completely(bucket);
1282 }
1283 if (last.bucket_ < buckets_.size()) {
1284 count_erased += hashtable::erase_bucket_range(last.bucket_, buckets_[last.bucket_], last.current_);
1285 }
