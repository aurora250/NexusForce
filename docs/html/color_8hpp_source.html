213 static constexpr int find_closest(const int value) noexcept {
214 constexpr int levels[6] = {0, 95, 135, 175, 215, 255};
215
216 int closest_idx = 0;
217 int min_diff = 255 * 255;
218 for (int i = 0; i < 6; ++i) {
219 const int diff = value - levels[i];
220 const int dist = diff * diff;
221 if (dist < min_diff) {
222 min_diff = dist;
223 closest_idx = i;
224 }
225 }
226 return closest_idx;
227 };

583 NEFORCE_NODISCARD constexpr color blend(const color& background) const noexcept {
584 if (a == 255) {
585 return *this;
586 }
587 if (a == 0) {
588 return background;
589 }
590
591 const decimal_t src_a = a / 255.0;
592 const decimal_t dst_a = background.a / 255.0;
593 const decimal_t inv_src_a = 1.0 - src_a;
594
595 // Result_A = Src_A + Dst_A * (1 - Src_A)
596 const decimal_t out_a = src_a + dst_a * inv_src_a;
597
598 if (out_a <= 0.0) {
599 return transparent();
600 }
601
602 // Result_RGB = (Src_RGB * Src_A + Dst_RGB * Dst_A * (1 - Src_A)) / Result_A
603 const decimal_t inv_out_a = 1.0 / out_a;
604 const int newR = static_cast<int>(_NEFORCE round((r * src_a + background.r * dst_a * inv_src_a) * inv_out_a));
605 const int newG = static_cast<int>(_NEFORCE round((g * src_a + background.g * dst_a * inv_src_a) * inv_out_a));
606 const int newB = static_cast<int>(_NEFORCE round((b * src_a + background.b * dst_a * inv_src_a) * inv_out_a));
607 const int newA = static_cast<int>(_NEFORCE round(out_a * 255.0));
608
609 return color(newR, newG, newB, newA);
610 }

632 static NEFORCE_CONSTEXPR20 color parse(const string_view hex) {
633 string_view clean_hex = hex;
634 if (clean_hex[0] == '#') {
635 clean_hex = clean_hex.tail(1);
636 }
637
638 if (clean_hex.length() == 6) {
639 const int r = hexadecimal::parse(clean_hex.substr(0, 2)).value();
640 const int g = hexadecimal::parse(clean_hex.substr(2, 2)).value();
641 const int b = hexadecimal::parse(clean_hex.substr(4, 2)).value();
642 return {r, g, b};
643 } else if (clean_hex.length() == 8) {
644 const int r = hexadecimal::parse(clean_hex.substr(0, 2)).value();
645 const int g = hexadecimal::parse(clean_hex.substr(2, 2)).value();
646 const int b = hexadecimal::parse(clean_hex.substr(4, 2)).value();
647 const int a = hexadecimal::parse(clean_hex.substr(6, 2)).value();
648 return {r, g, b, a};
649 } else {
650 NEFORCE_THROW_EXCEPTION(value_exception("Invalid hex string"));
651 }
652 unreachable();
653 }

662 NEFORCE_NODISCARD constexpr int to_ansi_256() const noexcept {
663 if (r == 0 && g == 0 && b == 0) {
664 return 16;
665 }
666 if (r == 255 && g == 255 && b == 255) {
667 return 231;
668 }
669
670 if (r == g && g == b) {
671 if (r < 8) {
672 return 16;
673 }
674 if (r > 248) {
675 return 231;
676 }
677 const int gray_index = (r - 8) / 10;
678 return 232 + (gray_index > 23 ? 23 : gray_index);
679 }
680
681 const int r_idx = find_closest(r);
682 const int g_idx = find_closest(g);
683 const int b_idx = find_closest(b);
684
685 // 16 + 36 * R + 6 * G + B
686 return 16 + 36 * r_idx + 6 * g_idx + b_idx;
687 }
