478 static NEFORCE_CONSTEXPR20 color parse(const string_view hex) {
479 string_view clean_hex = hex;
480 if (clean_hex[0] == '#') clean_hex = clean_hex.substr(1);
481
482 if (clean_hex.length() == 6) {
483 const int r = hexadecimal::parse(clean_hex.substr(0, 2)).value();
484 const int g = hexadecimal::parse(clean_hex.substr(2, 2)).value();
485 const int b = hexadecimal::parse(clean_hex.substr(4, 2)).value();
486 return {r, g, b};
487 } else if (clean_hex.length() == 8) {
488 const int r = hexadecimal::parse(clean_hex.substr(0, 2)).value();
489 const int g = hexadecimal::parse(clean_hex.substr(2, 2)).value();
490 const int b = hexadecimal::parse(clean_hex.substr(4, 2)).value();
491 const int a = hexadecimal::parse(clean_hex.substr(6, 2)).value();
492 return {r, g, b, a};
493 } else {
494 throw_exception(value_exception("Invalid hex string"));
495 }
496 NEFORCE_UNREACHABLE;

503 NEFORCE_NODISCARD constexpr int to_ansi_256() const noexcept {
504 if (r < 8 && g < 8 && b < 8) return 16;
505 if (r > 248 && g > 248 && b > 248) return 231;
506
507 if (r == g && g == b) {
508 const int gray_index = static_cast<int>((r - 8) / 247.0 * 24.0 + 0.5);
509 return 232 + gray_index;
510 }
511
512 const int r_idx = static_cast<int>((r / 255.0) * 5.0 + 0.5);
513 const int g_idx = static_cast<int>((g / 255.0) * 5.0 + 0.5);
514 const int b_idx = static_cast<int>((b / 255.0) * 5.0 + 0.5);
515
516 return 16 + 36 * r_idx + 6 * g_idx + b_idx;
517 }
