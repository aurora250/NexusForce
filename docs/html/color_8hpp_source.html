471 static NEFORCE_CONSTEXPR20 color parse(const string_view hex) {
472 string_view clean_hex = hex;
473 if (clean_hex[0] == '#') {
474 clean_hex = clean_hex.substr(1);
475 }
476
477 if (clean_hex.length() == 6) {
478 const int r = hexadecimal::parse(clean_hex.substr(0, 2)).value();
479 const int g = hexadecimal::parse(clean_hex.substr(2, 2)).value();
480 const int b = hexadecimal::parse(clean_hex.substr(4, 2)).value();
481 return {r, g, b};
482 } else if (clean_hex.length() == 8) {
483 const int r = hexadecimal::parse(clean_hex.substr(0, 2)).value();
484 const int g = hexadecimal::parse(clean_hex.substr(2, 2)).value();
485 const int b = hexadecimal::parse(clean_hex.substr(4, 2)).value();
486 const int a = hexadecimal::parse(clean_hex.substr(6, 2)).value();
487 return {r, g, b, a};
488 } else {
489 NEFORCE_THROW_EXCEPTION(value_exception("Invalid hex string"));
490 }
491 unreachable();

498 NEFORCE_NODISCARD constexpr int to_ansi_256() const noexcept {
499 if (r < 8 && g < 8 && b < 8) {
500 return 16;
501 }
502 if (r > 248 && g > 248 && b > 248) {
503 return 231;
504 }
505
506 if (r == g && g == b) {
507 const int gray_index = static_cast<int>((r - 8) / 247.0 * 24.0 + 0.5);
508 return 232 + gray_index;
509 }
510
511 const int r_idx = static_cast<int>((r / 255.0) * 5.0 + 0.5);
512 const int g_idx = static_cast<int>((g / 255.0) * 5.0 + 0.5);
513 const int b_idx = static_cast<int>((b / 255.0) * 5.0 + 0.5);
514
515 return 16 + 36 * r_idx + 6 * g_idx + b_idx;
516 }
