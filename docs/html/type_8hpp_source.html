84 void collect_properties(vector<pair<string, const meta_property*>>& result,
85 vector<reflect::type_id>* visited = nullptr) const {
86 vector<reflect::type_id> local_visited;
87 if (visited == nullptr) {
88 visited = &local_visited;
89 }
90
91 if (find(visited->begin(), visited->end(), type_id_) != visited->end()) {
92 return;
93 }
94 visited->push_back(type_id_);
95
96 for (const auto* base: base_types_) {
97 if (base != nullptr) {
98 base->collect_properties(result, visited);
99 }
100 }
101
102 for (const auto& property: properties_) {
103 const auto& name = property.first;
104 const auto& prop = property.second;
105 result.emplace_back(name, prop.get());
106 }
107 }

109 void collect_functions(vector<pair<string, const meta_function*>>& result,
110 vector<reflect::type_id>* visited = nullptr) const {
111 vector<reflect::type_id> local_visited;
112 if (visited == nullptr) {
113 visited = &local_visited;
114 }
115
116 if (find(visited->begin(), visited->end(), type_id_) != visited->end()) {
117 return;
118 }
119 visited->push_back(type_id_);
120
121 for (const auto* base: base_types_) {
122 if (base != nullptr) {
123 base->collect_functions(result, visited);
124 }
125 }
126
127 for (const auto& f: functions_) {
128 const auto& name = f.first;
129 const auto& func = f.second;
130 result.emplace_back(name, func.get());
131 }
