33template <typename T>
34class bounded_queue {
35private:
36 vector<T> buffer_;
37 size_t head_{0};
38 size_t tail_{0};
39 size_t count_{0};
40 size_t capacity_;
41
42public:
47 explicit bounded_queue(const size_t cap) :
48 buffer_(cap),
49 capacity_(cap) {}
50
51 bounded_queue(const bounded_queue&) = default;
52 bounded_queue& operator=(const bounded_queue&) = default;
53 bounded_queue(bounded_queue&&) noexcept = default;
54 bounded_queue& operator=(bounded_queue&&) noexcept = default;
55
57 NEFORCE_NODISCARD bool full() const noexcept { return count_ == capacity_; }
58
60 NEFORCE_NODISCARD bool empty() const noexcept { return count_ == 0; }
61
63 NEFORCE_NODISCARD size_t size() const noexcept { return count_; }
64
66 NEFORCE_NODISCARD size_t capacity() const noexcept { return capacity_; }
67
72 void push(T&& item) noexcept {
73 buffer_[tail_] = _NEFORCE move(item);
74 tail_ = (tail_ + 1) % capacity_;
75 ++count_;
76 }
77
82 NEFORCE_NODISCARD T pop() noexcept {
83 T item = _NEFORCE move(buffer_[head_]);
84 head_ = (head_ + 1) % capacity_;
85 --count_;
86 return item;
87 }
88
93 NEFORCE_NODISCARD T& front() noexcept { return buffer_[head_]; }
94};
