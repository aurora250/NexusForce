723constexpr void __rotate_cycle_aux(Iterator first, Iterator last, Iterator initial, Distance shift) {
724 iter_value_t<Iterator> value = *initial;
725 Iterator ptr1 = initial;
726 Iterator ptr2 = ptr1 + shift;
727 while (ptr2 != initial) {
728 *ptr1 = *ptr2;
729 ptr1 = ptr2;
730 if (last - ptr2 > shift) {

733 ptr2 = first + (shift - (last - ptr2));
