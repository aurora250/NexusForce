74 bool do_arrive(const byte_t old_phase, size_t current_index) {
75 const auto old_phase_value = old_phase;
76 const auto half_step = static_cast<byte_t>(old_phase_value + 1);
77 const auto full_step = static_cast<byte_t>(old_phase_value + 2);
78
79 size_t current_expected = expected_count_;
80 current_index %= ((expected_count_ + 1) >> 1);
81
82 for (int round = 0;; ++round) {
83 if (current_expected <= 1) {
84 return true;
85 }
86
87 size_t const end_node = ((current_expected + 1) >> 1), last_node = end_node - 1;
88
89 for (;; ++current_index) {
90 if (current_index == end_node) {
91 current_index = 0;
92 }
93
94 auto expected_phase = old_phase;
95 phase_ref_t phase_ref(state_array_[current_index].tickets[round]);
96
97 if (current_index == last_node && (current_expected & 1)) {
98 if (phase_ref.compare_exchange_strong(expected_phase, full_step, memory_order_acq_rel)) {
99 break;
100 }
101 } else if (phase_ref.compare_exchange_strong(expected_phase, half_step, memory_order_acq_rel)) {
102 return false;
103 } else if (expected_phase == half_step) {
104 if (phase_ref.compare_exchange_strong(expected_phase, full_step, memory_order_acq_rel)) {
105 break;
106 }
107 }
108 }
109
110 current_expected = last_node + 1;
111 current_index >>= 1;
112 }
113 }

146 NEFORCE_NODISCARD arrival_token arrive(ptrdiff_t update) {
147 constexpr hash<thread::id> hasher;
148 const size_t current_index = hasher(this_thread::id());
149 phase_ref_t phase_ref(current_phase_);
150 const auto old_phase = phase_ref.load(memory_order_relaxed);
151 const auto current_phase_value = static_cast<byte_t>(old_phase);
152
153 for (; update; --update) {
154 if (do_arrive(old_phase, current_index)) {
155 completion_function_();
156 expected_count_ += expected_adjustment_.load(memory_order_relaxed);
157 expected_adjustment_.store(0, memory_order_relaxed);
158 const auto new_phase = static_cast<byte_t>(current_phase_value + 2);
159 phase_ref.store(new_phase, memory_order_release);
160 phase_ref.notify_all();
161 }
162 }
163 return old_phase;
164 }
