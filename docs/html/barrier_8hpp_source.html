79 bool do_arrive(const byte_t old_phase, size_t current_index) {
80 const auto old_phase_value = old_phase;
81 const auto half_step = static_cast<byte_t>(old_phase_value + 1);
82 const auto full_step = static_cast<byte_t>(old_phase_value + 2);
83
84 size_t current_expected = expected_count_;
85 current_index %= ((expected_count_ + 1) >> 1);
86
87 for (int round = 0;; ++round) {
88 if (current_expected <= 1) {
89 return true;
90 }
91
92 size_t const end_node = ((current_expected + 1) >> 1), last_node = end_node - 1;
93
94 for (;; ++current_index) {
95 if (current_index == end_node) {
96 current_index = 0;
97 }
98
99 auto expected_phase = old_phase;
100 phase_ref_t phase_ref(state_array_[current_index].tickets[round]);
101
102 // NOLINTNEXTLINE(bugprone-branch-clone)
103 if (current_index == last_node && ((current_expected & 1) != 0U)) {
104 if (phase_ref.compare_exchange_strong(expected_phase, full_step, memory_order_acq_rel)) {
105 break;
106 }
107 } else if (phase_ref.compare_exchange_strong(expected_phase, half_step, memory_order_acq_rel)) {
108 return false;
109 } else if (expected_phase == half_step) {
110 if (phase_ref.compare_exchange_strong(expected_phase, full_step, memory_order_acq_rel)) {
111 break;
112 }
113 }
114 }
115
116 current_expected = last_node + 1;
117 current_index >>= 1;
118 }
119 }

150 NEFORCE_NODISCARD arrival_token arrive(ptrdiff_t update) {
151 constexpr hash<thread::id> hasher;
152 const size_t current_index = hasher(this_thread::id());
153 phase_ref_t phase_ref(current_phase_);
154 const auto old_phase = phase_ref.load(memory_order_relaxed);
155 const auto current_phase_value = static_cast<byte_t>(old_phase);
156
157 for (; update != 0; --update) {
158 if (do_arrive(old_phase, current_index)) {
159 completion_function_();
160 expected_count_ += expected_adjustment_.load(memory_order_relaxed);
161 expected_adjustment_.store(0, memory_order_relaxed);
162 const auto new_phase = static_cast<byte_t>(current_phase_value + 2);
163 phase_ref.store(new_phase, memory_order_release);
164 phase_ref.notify_all();
165 }
166 }
167 return old_phase;
168 }
