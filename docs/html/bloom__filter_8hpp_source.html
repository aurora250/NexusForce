179 static size_t compute_m(const size_t n, const double p) noexcept {
180 const double ln2 = logarithm_e(2.);
181 const double m = -static_cast<double>(n) * logarithm_e(p) / (ln2 * ln2);
182 return static_cast<size_t>(ceil(m));
183 }

193 static size_t compute_k(const size_t n, const size_t m) noexcept {
194 const double k = (static_cast<double>(m) / n) * logarithm_e(2.);
195 return static_cast<size_t>(max(static_cast<decimal_t>(1), round(k)));
196 }

205 pair<size_t, size_t> hash_values(const T& key) const noexcept {
206 size_t h1 = hasher_(key);
207 size_t h2 = rotate_l(h1, 17);
208 if (h2 == 0) {
209 h2 = 1;
210 }
211 return {h1, h2};
212 }

461 void from_bytes(const byte_vector& bytes) {
