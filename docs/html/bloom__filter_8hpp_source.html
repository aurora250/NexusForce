179 static size_t compute_m(const size_t n, const double p) noexcept {
180 if (p <= 0.0 || p >= 1.0) {
181 return 0;
182 }
183 const double ln2 = logarithm_e(2.);
184 const double m = -static_cast<double>(n) * logarithm_e(p) / (ln2 * ln2);
185 return static_cast<size_t>(ceil(m));
186 }
187
196 static size_t compute_k(const size_t n, const size_t m) noexcept {
197 if (n == 0 || m == 0) {
198 return 0;
199 }
200 const double k = (static_cast<double>(m) / n) * logarithm_e(2.);
201 return static_cast<size_t>(max(static_cast<decimal_t>(1), round(k)));
202 }

211 pair<size_t, size_t> hash_values(const T& key) const noexcept {
212 size_t h1 = hasher_(key);
213 size_t h2 = rotate_l(h1, 17);
214 if (h2 == 0) {
215 h2 = 1;
216 }
217 return {h1, h2};
218 }
