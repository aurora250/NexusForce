491template <typename T>
492void any::internal_manage<T>::manage(const any_operation op, const any* value, ArgT* arg) {
493 auto ptr = reinterpret_cast<const T*>(&value->storage_.buffer_);
494 switch (op) {
495 case ACCESS: {
496 arg->obj_ptr_ = const_cast<T*>(ptr);
497 break;
498 }
499 case GET_TYPE_INFO: {
500 arg->type_ptr_ = &typeid(T);
501 break;
502 }
503 case COPY: {
504 ::new (&arg->any_ptr_->storage_.buffer_) T(*ptr);
505 arg->any_ptr_->manage_ = value->manage_;
506 break;
507 }
508 case DESTROY: {
509 ptr->~T();
510 break;
511 }
512 case SWAP: {
513 ::new (&arg->any_ptr_->storage_.buffer_) T(_NEFORCE move(*const_cast<T*>(ptr)));
514 ptr->~T();
515 arg->any_ptr_->manage_ = value->manage_;
516 const_cast<any*>(value)->manage_ = nullptr;
517 break;
518 }
519 default: {
520 unreachable();
521 }
522 }
523}
