520template <typename T>
521void any::internal_manage<T>::manage(const any_operation op, const any* value, ArgT* arg) {
522 auto ptr = reinterpret_cast<const T*>(&value->storage_.buffer_);
523 switch (op) {
524 case ACCESS: {
525 arg->obj_ptr_ = const_cast<T*>(ptr);
526 break;
527 }
528 case GET_TYPE_INFO: {
529 arg->type_ptr_ = &typeid(T);
530 break;
531 }
532 case COPY: {
533 ::new(&arg->any_ptr_->storage_.buffer_) T(*ptr);
534 arg->any_ptr_->manage_ = value->manage_;
535 break;
536 }
537 case DESTROY: {
538 ptr->~T();
539 break;
540 }
541 case SWAP: {
542 ::new(&arg->any_ptr_->storage_.buffer_) T(_NEFORCE move(*const_cast<T*>(ptr)));
543 ptr->~T();
544 arg->any_ptr_->manage_ = value->manage_;
545 const_cast<any*>(value)->manage_ = nullptr;
546 break;
547 }
548 default: {
549 unreachable();
550 }
551 }
552}

555void any::external_manage<T>::manage(const any_operation op, const any* value, ArgT* arg) {

557 switch (op) {

559 arg->obj_ptr_ = const_cast<T*>(ptr);
560 break;
561 }
562 case GET_TYPE_INFO: {
563 arg->type_ptr_ = &typeid(T);
564 break;
565 }
566 case COPY: {
567 arg->any_ptr_->storage_.ptr_ = ::new T(*ptr);
568 arg->any_ptr_->manage_ = value->manage_;

574 }
