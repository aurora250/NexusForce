92 template <typename T>
93 toml_builder& value_impl(unique_ptr<T> value) {
94 if (contexts_.empty()) {
95 NEFORCE_THROW_EXCEPTION(toml_exception("Cannot add value to root (root must be a table)"));
96 }
97
98 const auto& top = contexts_.top();
99 if (top.type == array) {
100 top.array_ptr->add_element(_NEFORCE move(value));
101 } else if (top.type == table || top.type == inline_table) {
102 if (current_key_.empty()) {
103 NEFORCE_THROW_EXCEPTION(toml_exception("No key set for value in table"));
104 }
105 if (top.table_ptr->has_member(current_key_)) {
106 NEFORCE_THROW_EXCEPTION(toml_exception(("Duplicate key: " + current_key_).data()));
107 }
108 top.table_ptr->add_member(current_key_, _NEFORCE move(value));
109 current_key_.clear();
110 }
111 return *this;
112 }
