38constexpr enable_if_t<is_signed_v<T>, T> str_to_ints(const string_view sv, char** endptr, int base) {
39 using UT = make_unsigned_t<T>;
40
41 const char* start = sv.data();
42 const size_t len = sv.size();
43 const char* end = start + len;
44
45 if (len == 0) {
46 if (endptr != nullptr) {
47 *endptr = const_cast<char*>(start);
48 }
49 return 0;
50 }
51
52 const char* p = start;
53 while (p != end && is_space(*p)) {
54 ++p;
55 }
56 const char* start_conversion = p;
57
58 int sign = 1;
59 if (p != end && *p == '+') {
60 ++p;
61 } else if (p != end && *p == '-') {
62 sign = -1;
63 ++p;
64 }
65
66 if (base != 0 && (base < 2 || base > 36)) {
67 if (endptr != nullptr) {
68 *endptr = const_cast<char*>(start_conversion);
69 }
70 return 0;
71 }
72
73 if (base == 0) {
74 if (p != end && *p == '0') {
75 if (p + 1 != end && (*(p + 1) == 'x' || *(p + 1) == 'X')) {
76 base = 16;
77 p += 2;
78 } else {
79 base = 8;
80 }
81 } else {
82 base = 10;
83 }
84 } else if (base == 16 && p + 1 < end && *p == '0' && (*(p + 1) == 'x' || *(p + 1) == 'X')) {
85 p += 2;
86 }
87
88 const UT umax = static_cast<UT>(numeric_traits<T>::max());
89 const UT umin_abs = static_cast<UT>(numeric_traits<T>::max()) + static_cast<UT>(1);
90 const UT limit = (sign > 0) ? umax : umin_abs;
91 const UT cutoff = limit / static_cast<UT>(base);
92 const UT cutlim = limit % static_cast<UT>(base);
93
94 UT result = 0;
95 bool any_converted = false;
96 bool overflow = false;
97
98 while (p != end) {
99 UT digit(0);
100 const char c = *p;
101 if (c >= '0' && c <= '9') {
102 digit = static_cast<UT>(c - '0');
103 } else if (c >= 'a' && c <= 'z') {
104 digit = static_cast<UT>(c - 'a') + static_cast<UT>(10);
105 } else if (c >= 'A' && c <= 'Z') {
106 digit = static_cast<UT>(c - 'A') + static_cast<UT>(10);
107 } else {
108 break;
109 }
110 if (digit >= static_cast<UT>(base)) {
111 break;
112 }
113
114 any_converted = true;
115 if (!overflow) {
116 if (result > cutoff || (result == cutoff && digit > cutlim)) {
117 overflow = true;
118 } else {
119 result = result * static_cast<UT>(base) + digit;
120 }
121 }
122 ++p;
123 }
124
125 if (endptr != nullptr) {
126 *endptr = any_converted ? const_cast<char*>(p) : const_cast<char*>(start_conversion);
127 }
128
129 if (!any_converted) {
130 return 0;
131 }
132
133 if (overflow) {
134 return (sign > 0) ? numeric_traits<T>::max() : numeric_traits<T>::min();
135 }
136
137 if (sign > 0) {
138 return static_cast<T>(result);
139 }
140 if (result == umin_abs) {
141 return numeric_traits<T>::min();
142 }
143 return static_cast<T>(~result + static_cast<UT>(1));
144}

160 const size_t len = sv.size();
161 const char* end = start + len;

164 if (endptr != nullptr) {
165 *endptr = const_cast<char*>(start);
166 }
167 return 0;
168 }

171 while (p != end && is_space(*p)) {

175
176 int sign = 1;
177 if (p != end && *p == '+') {
178 ++p;
179 } else if (p != end && *p == '-') {
180 sign = -1;
181 ++p;
182 }
183
184 if (base != 0 && (base < 2 || base > 36)) {
185 if (endptr != nullptr) {
186 *endptr = const_cast<char*>(start_conversion);
187 }
188 return 0;
189 }
190
191 if (base == 0) {
192 if (p != end && *p == '0') {
193 if (p + 1 != end && (*(p + 1) == 'x' || *(p + 1) == 'X')) {
194 base = 16;
195 p += 2;
196 } else {
197 base = 8;
198 }
199 } else {
200 base = 10;
201 }
202 }
203
204 if (base == 16 && p + 1 < end && *p == '0' && (*(p + 1) == 'x' || *(p + 1) == 'X')) {
205 p += 2;
206 }
207
208 const T cutoff = numeric_traits<T>::max() / base;
209 const T cutlim = numeric_traits<T>::max() % base;
210 T result(0);
211 bool any_converted = false;
212 bool overflow = false;
213
214 while (p != end) {
215 T digit(0);
216 const char c = *p;
217 if (c >= '0' && c <= '9') {
218 digit = static_cast<T>(c - '0');
219 } else if (c >= 'a' && c <= 'z') {
220 digit = static_cast<T>(c - 'a') + static_cast<T>(10);
221 } else if (c >= 'A' && c <= 'Z') {
222 digit = static_cast<T>(c - 'A') + static_cast<T>(10);
223 } else {
224 break;
225 }
226 if (digit >= static_cast<T>(base)) {
227 break;
228 }
229

231 if (!overflow) {
232 if (result > cutoff || (result == cutoff && digit > cutlim)) {
233 overflow = true;
234 } else {

236 }

238 ++p;
239 }
240
241 if (endptr != nullptr) {
242 *endptr = any_converted ? const_cast<char*>(p) : const_cast<char*>(start_conversion);
243 }
244
245 if (!any_converted) {
246 return static_cast<T>(0);
247 }
248 if (overflow) {
249 return numeric_traits<T>::max();
250 }
251
252 if (sign < 0) {
253 // for unsigned, negative sign yields two's complement wrap,
254 // but we follow C standard: strtoul("-1", ...) returns ULLONG_MAX.
255 // So just cast via signed negation then to unsigned.
256 return static_cast<T>(static_cast<typename make_signed<T>::type>(0) -
257 static_cast<typename make_signed<T>::type>(result));
258 }
259 return result;

306constexpr enable_if_t<is_floating_point_v<T>, T> str_to_floats(const string_view sv, char** endptr) {
307 const char* start = sv.data();
308 const size_t len = sv.size();
309 const char* end = start + len;
310
311 if (len == 0) {
312 if (endptr) {
313 *endptr = const_cast<char*>(start);
314 }
315 return static_cast<T>(0);
316 }
317
318 const char* p = start;
319 while (p != end && is_space(*p)) {
320 ++p;
321 }
322 const char* start_conversion = p;
323
324 int sign = 1;
325 if (p != end && *p == '+') {
326 ++p;
327 } else if (p != end && *p == '-') {
328 sign = -1;
329 ++p;
330 }
331
332 const char* p_start = p;
333
334 if (p != end && (p[0] == 'i' || p[0] == 'I')) {
335 bool is_inf = false;
336
337 if (p + 8 <= end) {
338 if ((p[1] == 'n' || p[1] == 'N') && (p[2] == 'f' || p[2] == 'F') && (p[3] == 'i' || p[3] == 'I') &&
339 (p[4] == 'n' || p[4] == 'N') && (p[5] == 'i' || p[5] == 'I') && (p[6] == 't' || p[6] == 'T') &&
340 (p[7] == 'y' || p[7] == 'Y')) {
341 if (p + 8 == end || !is_alpha_or_digit(p[8])) {
342 p += 8;
343 is_inf = true;
344 }
345 }
346 }
347
348 if (!is_inf && p + 3 <= end) {
349 if ((p[1] == 'n' || p[1] == 'N') && (p[2] == 'f' || p[2] == 'F')) {
350 if (p + 3 == end || !is_alpha_or_digit(p[3])) {
351 p += 3;
352 is_inf = true;
353 }
354 }
355 }
356
357 if (is_inf) {
358 if (endptr) {
359 *endptr = const_cast<char*>(p);
360 }
361 const T inf_val = numeric_traits<T>::infinity();
362 return (sign < 0) ? -inf_val : inf_val;
363 }
364 }
365
366 if (p != end && (p[0] == 'n' || p[0] == 'N')) {
367 if (p + 3 <= end) {
368 const char c1 = p[1], c2 = p[2];
369 if ((c1 == 'a' || c1 == 'A') && (c2 == 'n' || c2 == 'N')) {
370 const bool terminated = (p + 3 == end) || !is_alpha_or_digit(p[3]);
371 if (terminated) {
372 p += 3;
373 if (p != end && *p == '(') {
374 ++p;
375 while (p != end && *p != ')') {
376 ++p;
377 }
378 if (p != end && *p == ')') {
379 ++p;
380 }
381 }
382 if (endptr) {
383 *endptr = const_cast<char*>(p);
384 }
385 return numeric_traits<T>::quiet_nan();
386 }
387 }
388 }
389 }
390
391 p = p_start;
392
393 T significand = 0;
394 int exponent = 0;
395 int digits_count = 0;
396 bool has_digits = false;
397
398 while (p != end && *p >= '0' && *p <= '9') {
399 has_digits = true;
400 if (digits_count < numeric_traits<T>::max_digits10) {
401 significand = significand * static_cast<T>(10) + static_cast<T>(*p - '0');
402 } else {
403 exponent++;
404 }
405 digits_count++;
406 ++p;
407 }
408
409 if (p != end && *p == '.') {
410 ++p;
411 while (p != end && *p >= '0' && *p <= '9') {
412 has_digits = true;
413 if (digits_count < numeric_traits<T>::max_digits10) {
414 significand = significand * static_cast<T>(10) + static_cast<T>(*p - '0');
415 exponent--;
416 }
417 digits_count++;
418 ++p;
419 }
420 }
421
422 if (!has_digits) {
423 if (endptr) {
424 *endptr = const_cast<char*>(start_conversion);
425 }
426 return static_cast<T>(0);
427 }
428
429 if (p != end && (*p == 'e' || *p == 'E')) {
430 const char* e_pos = p;
431 ++p;
432
433 int exp_sign = 1;
434 if (p != end && *p == '+') {
435 ++p;
436 } else if (p != end && *p == '-') {
437 exp_sign = -1;
438 ++p;
439 }
440
441 if (p != end && *p >= '0' && *p <= '9') {
442 int exp_val = 0;
443 while (p != end && *p >= '0' && *p <= '9') {
444 if (exp_val < 100000) {
445 exp_val = exp_val * 10 + (*p - '0');
446 }
447 ++p;
448 }
449 exponent += exp_sign * exp_val;
450 } else {
451 p = e_pos;
452 }
453 }
454
455 T result = significand;
456
457 if (exponent != 0) {
458 constexpr int max_exp = numeric_traits<T>::max_exponent10 + 50;
459 constexpr int min_exp = numeric_traits<T>::min_exponent10 - 50;
460 constexpr int max_table_exp = 32;
461
462 if (exponent > max_exp || exponent > max_table_exp) {
463 if (endptr) {
464 *endptr = const_cast<char*>(p);
465 }
466 return (sign > 0) ? numeric_traits<T>::infinity() : -numeric_traits<T>::infinity();
467 } else if (exponent < min_exp || exponent < -max_table_exp) {
468 if (endptr) {
469 *endptr = const_cast<char*>(p);
470 }
471 return static_cast<T>(0);
472 } else {
473 result *= fast_pow10<T>(exponent);
474 }
475 }
476
477 const T inf = numeric_traits<T>::infinity();
478 if (result == inf || result == -inf) {
479 if (endptr) {
480 *endptr = const_cast<char*>(p);
481 }
482 return (sign > 0) ? inf : -inf;
483 }
484
485 result = (sign > 0) ? result : -result;
486
487 if (endptr) {
488 *endptr = const_cast<char*>(p);
489 }
490 return result;
491}
