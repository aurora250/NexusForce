38constexpr T str_to_ints(const string_view sv, char** endptr, int base) {
39 const char* start = sv.data();
40 const size_t len = sv.size();
41 const char* end = start + len;
42
43 if (len == 0) {
44 if (endptr) *endptr = const_cast<char*>(start);
45 return 0;
46 }
47
48 const char* p = start;
49 while (p != end && is_space(*p)) ++p;
50 const char* start_conversion = p;
51
52 int sign = 1;
53 if (p != end && *p == '+') {
54 ++p;
55 } else if (p != end && *p == '-') {
56 sign = -1;
57 ++p;
58 }
59
60 if (base != 0 && (base < 2 || base > 36)) {
61 if (endptr) *endptr = const_cast<char*>(start_conversion);
62 return 0;
63 }
64
65 if (base == 0) {
66 if (p != end && *p == '0') {
67 if (p + 1 != end && (*(p + 1) == 'x' || *(p + 1) == 'X')) {
68 base = 16;
69 p += 2;
70 } else {
71 base = 8;
72 }
73 } else {
74 base = 10;
75 }
76 }
77
78 if (base == 16 && p + 1 < end && *p == '0' && (*(p + 1) == 'x' || *(p + 1) == 'X')) {
79 p += 2;
80 }
81
82 const T cutoff = numeric_traits<T>::min() / base;
83 const T cutlim = numeric_traits<T>::min() % base;
84 T result = 0;
85 bool any_converted = false;
86 bool overflow = false;
87
88 while (p != end) {
89 int digit = 0;
90 const char c = *p;
91 if (c >= '0' && c <= '9') {
92 digit = c - '0';
93 } else if (c >= 'a' && c <= 'z') {
94 digit = c - 'a' + 10;
95 } else if (c >= 'A' && c <= 'Z') {
96 digit = c - 'A' + 10;
97 } else {
98 break;
99 }
100 if (digit >= base) break;
101
102 any_converted = true;
103 if (!overflow) {
104 if (result < cutoff || (result == cutoff && digit > -cutlim)) {
105 overflow = true;
106 } else {
107 result = result * base - digit;
108 }
109 }
110 ++p;
111 }
112
113 if (endptr) {
114 *endptr = any_converted ? const_cast<char*>(p) : const_cast<char*>(start_conversion);
115 }
116
117 if (!any_converted) return 0;
118 if (overflow)
119 return (sign > 0) ? numeric_traits<T>::max() : numeric_traits<T>::min();
120
121 if (sign > 0) {
122 if (result == numeric_traits<T>::min())
123 return numeric_traits<T>::max();
124 return -result;
125 }
126 return result;
127}

275template <typename T>
276constexpr T str_to_floats(const string_view sv, char** endptr) {
277 const char* start = sv.data();
278 const size_t len = sv.size();
279 const char* end = start + len;
280
281 if (len == 0) {
282 if (endptr) *endptr = const_cast<char*>(start);
283 return static_cast<T>(0);
284 }
285
286 const char* p = start;
287 while (p != end && _NEFORCE is_space(*p)) ++p;
288 const char* start_conversion = p;
289
290 int sign = 1;
291 if (p != end && *p == '+') {
292 ++p;
293 } else if (p != end && *p == '-') {
294 sign = -1;
295 ++p;
296 }
297
298 const char* p_start = p;
299
300 if (p != end && (p[0] == 'i' || p[0] == 'I')) {
301 if (p + 3 <= end) {
302 const char c1 = p[1], c2 = p[2];
303 if ((c1 == 'n' || c1 == 'N') && (c2 == 'f' || c2 == 'F')) {
304 if (p + 3 == end || !is_alpha_or_digit(p[3])) {
305 p += 3;
306 T inf_val = numeric_traits<T>::infinity();
307 if (endptr) *endptr = const_cast<char*>(p);
308 return (sign < 0) ? -inf_val : inf_val;
309 }
310 }
311 }
312 }
313
314 if (p != end && (p[0] == 'n' || p[0] == 'N')) {
315 if (p + 3 <= end) {
316 const char c1 = p[1], c2 = p[2];
317 if ((c1 == 'a' || c1 == 'A') && (c2 == 'n' || c2 == 'N')) {
318 if (p + 3 == end || !is_alpha_or_digit(p[3])) {
319 p += 3;
320 if (p != end && *p == '(') {
321 ++p;
322 while (p != end && *p != ')') ++p;
323 if (p != end && *p == ')') ++p;
324 }
325 if (endptr) *endptr = const_cast<char*>(p);
326 return numeric_traits<T>::quiet_nan();
327 }
328 }
329 }
330 }
331
332 p = p_start;
333
334 T significand = 0;
335 int exponent = 0;
336 int digits_count = 0;
337 bool has_digits = false;
338
339 while (p != end && *p >= '0' && *p <= '9') {
340 has_digits = true;
341 if (digits_count < numeric_traits<T>::digits10) {
342 significand = significand * 10 + (*p - '0');
343 } else {
344 exponent++;
345 }
346 digits_count++;
347 ++p;
348 }
349
350 if (p != end && *p == '.') {
351 ++p;
352 while (p != end && *p >= '0' && *p <= '9') {
353 has_digits = true;
354 if (digits_count < numeric_traits<T>::digits10) {
355 significand = significand * 10 + (*p - '0');
356 exponent--;
357 }
358 digits_count++;
359 ++p;
360 }
361 }
362
363 if (!has_digits) {
364 if (endptr) *endptr = const_cast<char*>(start_conversion);
365 return static_cast<T>(0);
366 }
367
368 if (p != end && (*p == 'e' || *p == 'E')) {
369 ++p;
370 int exp_sign = 1;
371 if (p != end && *p == '+') {
372 ++p;
373 } else if (p != end && *p == '-') {
374 exp_sign = -1;
375 ++p;
376 }
377
378 if (p != end && *p >= '0' && *p <= '9') {
379 int exp_val = 0;
380 while (p != end && *p >= '0' && *p <= '9') {
381 if (exp_val < 10000) {
382 exp_val = exp_val * 10 + (*p - '0');
383 }
384 ++p;
385 }
386 exponent += exp_sign * exp_val;
387 } else {
388 --p;
389 if (p != start_conversion && (p[-1] == '+' || p[-1] == '-')) --p;
390 }
391 }
392
393 T result = significand;
394
395 if (exponent != 0) {
396 if (exponent > 400) {
397 if (endptr) *endptr = const_cast<char*>(p);
398 return sign * numeric_traits<T>::max();
399 } else if (exponent < -400) {
400 if (endptr) *endptr = const_cast<char*>(p);
401 return T(0);
402 } else {
403 result *= fast_pow10<T>(exponent);
404 }
405 }
406
407 result *= sign;
408 const T inf = numeric_traits<T>::infinity();
409 if (result == inf || result == -inf) {
410 result = sign * numeric_traits<T>::max();
411 }
412
413 if (endptr) *endptr = const_cast<char*>(p);
414 return result;
415}
