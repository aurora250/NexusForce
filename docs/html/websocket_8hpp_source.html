255 bool read_frame() noexcept {
256 try {
257 websocket_frame_header hdr{};
258 if (!receive_exact(socket_, &hdr, 2)) {
259 return false;
260 }
261
262 if (hdr.rsv1 || hdr.rsv2 || hdr.rsv3) {
263 send_close_frame(websocket_status::PROTOCOL_ERROR, "Reserved bits set");
264 return false;
265 }
266
267 const auto opcode = static_cast<websocket_opcode>(hdr.opcode);
268 const bool is_ctrl = (hdr.opcode >= 0x8);
269
270 if (is_ctrl && !hdr.fin) {
271 send_close_frame(websocket_status::PROTOCOL_ERROR, "Fragmented control frame");
272 return false;
273 }
274
275 uint64_t payload_len = hdr.payload_len;
276 if (payload_len == 126) {
277 uint16_t ext{};
278 if (!receive_exact(socket_, &ext, 2)) {
279 return false;
280 }
281 payload_len = endian::network_to_host<uint16_t>(ext);
282 } else if (payload_len == 127) {
283 uint64_t ext{};
284 if (!receive_exact(socket_, &ext, 8)) {
285 return false;
286 }
287 payload_len = endian::network_to_host<uint64_t>(ext);
288 }
289
290 if (is_ctrl && payload_len > 125) {
291 send_close_frame(websocket_status::PROTOCOL_ERROR, "Control frame payload too large");
292 return false;
293 }
294
295 if (payload_len > MAX_PAYLOAD_SIZE) {
296 send_close_frame(websocket_status::MESSAGE_TOO_BIG, "Payload exceeds limit");
297 return false;
298 }
299
300 uint32_t masking_key = 0;
301 if (hdr.masked) {
302 if (!receive_exact(socket_, &masking_key, 4)) {
303 return false;
304 }
305 }
306
307 string payload;
308 if (payload_len > 0) {
309 payload.resize(static_cast<size_t>(payload_len));
310 if (!receive_exact(socket_, payload.data(), static_cast<size_t>(payload_len))) {
311 return false;
312 }
313 }
314
315 if (hdr.masked && payload_len > 0) {
316 const auto* key_bytes = reinterpret_cast<const char*>(&masking_key);
317 for (size_t i = 0; i < static_cast<size_t>(payload_len); ++i) {
318 payload[i] ^= key_bytes[i % 4];
319 }
320 }
321
322 return dispatch(hdr, opcode, payload);
323
324 } catch (const exception& e) {
325 notify_error(e);
326 return false;
327 }
328 }

330 bool dispatch(const websocket_frame_header& hdr, websocket_opcode opcode, string payload) {
331 switch (opcode) {
332 case websocket_opcode::TEXT:
333 case websocket_opcode::BINARY: {
334 if (in_fragment_) {
335 send_close_frame(websocket_status::PROTOCOL_ERROR, "New data frame before fragment complete");
336 return false;
337 }
338 if (!hdr.fin) {
339 fragment_opcode_ = opcode;
340 fragment_buffer_ = payload;
341 in_fragment_ = true;
342 } else {
343 deliver_message(payload, opcode);
344 }
345 return true;
346 }
347 case websocket_opcode::CONTINUATION: {
348 if (!in_fragment_) {
349 send_close_frame(websocket_status::PROTOCOL_ERROR, "Unexpected continuation frame");
350 return false;
351 }
352 fragment_buffer_ += payload;
353 if (hdr.fin) {
354 deliver_message(fragment_buffer_, fragment_opcode_);
355 fragment_buffer_.clear();
356 in_fragment_ = false;
357 }
358 return true;
359 }
360 case websocket_opcode::PING: {
361 queue_frame(build_frame(websocket_opcode::PONG, payload, false), true);
362 return true;
363 }
364 case websocket_opcode::PONG: {
365 last_pong_ms_ = now_ms();
366 ping_pending_ = false;
367 return true;
368 }
369 case websocket_opcode::CLOSE: {
370 handle_close_frame(payload);
371 return false;
372 }
373 default: {
374 send_close_frame(websocket_status::PROTOCOL_ERROR, "Unknown opcode");
375 return false;
376 }
377 }
378 }

395 void handle_close_frame(string payload) {
396 auto status = websocket_status::NORMAL_CLOSURE;
397 string reason;
398
399 if (payload.size() >= 2) {
400 const uint16_t code = (static_cast<uint8_t>(payload[0]) << 8) | static_cast<uint8_t>(payload[1]);
401 status = static_cast<websocket_status>(code);
402 if (payload.size() > 2) {
403 reason = payload.substr(2);
404 }
405 }

412 void heartbeat_loop() noexcept {
413 last_pong_ms_ = now_ms();
414
415 while (running_) {
416 for (int i = 0; i < HEARTBEAT_INTERVAL_SEC * 10 && running_; ++i) {
417 this_thread::sleep_for(milliseconds(100));
418 }
419 if (!running_) {

422
423 const int64_t elapsed_ms = now_ms() - last_pong_ms_.load();
424 constexpr int64_t timeout_ms = static_cast<int64_t>(HEARTBEAT_TIMEOUT_SEC) * 1000;
425
426 if (ping_pending_.load() && elapsed_ms > timeout_ms) {
427 do_stop(websocket_status::ABNORMAL_CLOSURE, "Heartbeat timeout");
