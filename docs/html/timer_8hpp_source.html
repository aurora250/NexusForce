99 void run() {
100 while (!stopped_.load()) {
101 unique_lock<mutex> lock(mutex_);
102
103 if (nodes_.empty()) {
104 cv_.wait(lock, [this] { return stopped_.load() || !nodes_.empty(); });
105 if (stopped_.load()) {
106 break;
107 }
108 }
109
110 time_point now = clock_type::now();
111 while (!nodes_.empty() && nodes_.begin()->expire <= now) {
112 auto it = nodes_.begin();
113 node current_node = *it;
114 nodes_.erase(it);
115 node_map_.erase(current_node.id);
116
117 lock.unlock_quiet();
118 if (!stopped_.load()) {
119 current_node.handler();
120 }
121 lock.lock_quiet();
122 now = clock_type::now();
123 }
124
125 if (!nodes_.empty()) {
126 time_point next_expire = nodes_.begin()->expire;
127 cv_.wait_until(lock, next_expire);
128 }
129 }
130 }

166 token add_task(time_point expire, handler_type&& handler) {
167 unique_lock<mutex> lock(mutex_);
168 token id = next_id_++;
169
170 const bool is_earliest = nodes_.empty() || expire < nodes_.begin()->expire;
171
172 node new_node(expire, id, _NEFORCE move(handler));
173 auto result = nodes_.insert(new_node);
174 node_map_[id] = result.first;
175
176 lock.unlock_quiet();
177
178 if (is_earliest) {
179 cv_.notify_one();
180 }
181
182 return id;
183 }

192 bool cancel(token id) {
193 unique_lock<mutex> lock(mutex_);
194 auto it_map = node_map_.find(id);
195 if (it_map == node_map_.end()) {
196 return false;
197 }
198
199 const bool is_earliest = (it_map->second == nodes_.begin());
200 nodes_.erase(it_map->second);
201 node_map_.erase(it_map);
202
203 lock.unlock_quiet();
204
205 if (is_earliest) {
206 cv_.notify_one();
207 }
208
209 return true;
210 }
