139NEFORCE_ALWAYS_INLINE_INLINE void rb_tree_insert_rebalance(rb_tree_node_base* insert,
140 rb_tree_node_base*& root) noexcept {
141 insert->color_ = RB_TREE_RED;
142
143 while (insert != root && insert->parent_->color_ == RB_TREE_RED) {
144 if (insert->parent_ == insert->parent_->parent_->left_) {
145 rb_tree_node_base* y = insert->parent_->parent_->right_;
146
147 if (y != nullptr && y->color_ == RB_TREE_RED) {
148 insert->parent_->color_ = RB_TREE_BLACK;
149 y->color_ = RB_TREE_BLACK;
150 insert->parent_->parent_->color_ = RB_TREE_RED;
151 insert = insert->parent_->parent_;
152 } else {
153 if (insert == insert->parent_->right_) {
154 insert = insert->parent_;
155 _NEFORCE rb_tree_rotate_left(insert, root);
156 }
157 insert->parent_->color_ = RB_TREE_BLACK;
158 insert->parent_->parent_->color_ = RB_TREE_RED;
159 _NEFORCE rb_tree_rotate_right(insert->parent_->parent_, root);
160 }
161 } else {
162 rb_tree_node_base* y = insert->parent_->parent_->left_;
163 if (y != nullptr && y->color_ == RB_TREE_RED) {
164 insert->parent_->color_ = RB_TREE_BLACK;
165 y->color_ = RB_TREE_BLACK;
166 insert->parent_->parent_->color_ = RB_TREE_RED;
167 insert = insert->parent_->parent_;
168 } else {
169 if (insert == insert->parent_->left_) {
170 insert = insert->parent_;
171 _NEFORCE rb_tree_rotate_right(insert, root);
172 }
173 insert->parent_->color_ = RB_TREE_BLACK;
174 insert->parent_->parent_->color_ = RB_TREE_RED;
175 _NEFORCE rb_tree_rotate_left(insert->parent_->parent_, root);
176 }
177 }
178 }
179 root->color_ = RB_TREE_BLACK;

304 _NEFORCE rb_tree_rotate_right(w, root);
305 w = x_parent->right_;

309 if (w->right_ != nullptr) {
310 w->right_->color_ = RB_TREE_BLACK;
311 }
312 _NEFORCE rb_tree_rotate_left(x_parent, root);
313 break;
314 }
315 } else {
316 rb_tree_node_base* w = x_parent->left_;
317 NEFORCE_DEBUG_VERIFY(w != nullptr, "RB-tree structure corrupted: left sibling is null");
318 if (w == nullptr) {
319 x = x_parent;

321 continue;

325 w->color_ = RB_TREE_BLACK;

328 w = x_parent->left_;
329 }
330
331 if ((w->right_ == nullptr || w->right_->color_ == RB_TREE_BLACK) &&
332 (w->left_ == nullptr || w->left_->color_ == RB_TREE_BLACK)) {
333 w->color_ = RB_TREE_RED;
334 x = x_parent;
335 x_parent = x_parent->parent_;
336 } else {
337 if (w->left_ == nullptr || w->left_->color_ == RB_TREE_BLACK) {
338 if (w->right_ != nullptr) {
339 w->right_->color_ = RB_TREE_BLACK;
340 }
341
342 w->color_ = RB_TREE_RED;
343 _NEFORCE rb_tree_rotate_left(w, root);
344 w = x_parent->left_;

849 pair<iterator, bool> insert_unique(link_type node) {
850 link_type y = header_;
851 link_type x = root();
