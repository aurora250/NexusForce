141NEFORCE_ALWAYS_INLINE_INLINE void
142rb_tree_insert_rebalance(rb_tree_node_base* insert, rb_tree_node_base*& root) noexcept {
143 insert->color_ = RB_TREE_RED;
144
145 while (insert != root && insert->parent_->color_ == RB_TREE_RED) {
146 if (insert->parent_ == insert->parent_->parent_->left_) {
147 rb_tree_node_base* y = insert->parent_->parent_->right_;
148
149 if (y != nullptr && y->color_ == RB_TREE_RED) {
150 insert->parent_->color_ = RB_TREE_BLACK;
151 y->color_ = RB_TREE_BLACK;
152 insert->parent_->parent_->color_ = RB_TREE_RED;
153 insert = insert->parent_->parent_;
154 } else {
155 if (insert == insert->parent_->right_) {
156 insert = insert->parent_;
157 _NEFORCE rb_tree_rotate_left(insert, root);
158 }
159 insert->parent_->color_ = RB_TREE_BLACK;
160 insert->parent_->parent_->color_ = RB_TREE_RED;
161 _NEFORCE rb_tree_rotate_right(insert->parent_->parent_, root);
162 }
163 } else {
164 rb_tree_node_base* y = insert->parent_->parent_->left_;
165 if (y != nullptr && y->color_ == RB_TREE_RED) {
166 insert->parent_->color_ = RB_TREE_BLACK;
167 y->color_ = RB_TREE_BLACK;
168 insert->parent_->parent_->color_ = RB_TREE_RED;
169 insert = insert->parent_->parent_;
170 } else {

172 insert = insert->parent_;
173 _NEFORCE rb_tree_rotate_right(insert, root);
174 }
175 insert->parent_->color_ = RB_TREE_BLACK;
176 insert->parent_->parent_->color_ = RB_TREE_RED;

195rb_tree_erase_rebalance(rb_tree_node_base* erase, rb_tree_node_base*& root,
196 rb_tree_node_base*& leftmost, rb_tree_node_base*& rightmost) noexcept {

298 x_parent = x_parent->parent_;
299 } else {

302 w->color_ = RB_TREE_RED;

304 w = x_parent->right_;
305 }

308 if (w->right_ != nullptr) w->right_->color_ = RB_TREE_BLACK;
309 _NEFORCE rb_tree_rotate_left(x_parent, root);
310 break;
311 }
312 } else {

314 NEFORCE_DEBUG_VERIFY(w != nullptr, "RB-tree structure corrupted: left sibling is null");
315 if (w == nullptr) {
316 x = x_parent;
317 x_parent = x_parent->parent_;
318 continue;

321 if (w->color_ == RB_TREE_RED) {
322 w->color_ = RB_TREE_BLACK;
323 x_parent->color_ = RB_TREE_RED;
324 _NEFORCE rb_tree_rotate_right(x_parent, root);
325 w = x_parent->left_;
326 }
327
328 if ((w->right_ == nullptr || w->right_->color_ == RB_TREE_BLACK) &&
329 (w->left_ == nullptr || w->left_->color_ == RB_TREE_BLACK)) {
330 w->color_ = RB_TREE_RED;
331 x = x_parent;
332 x_parent = x_parent->parent_;
333 } else {
334 if (w->left_ == nullptr || w->left_->color_ == RB_TREE_BLACK) {
335 if (w->right_ != nullptr) {
336 w->right_->color_ = RB_TREE_BLACK;
337 }
338
339 w->color_ = RB_TREE_RED;
340 _NEFORCE rb_tree_rotate_left(w, root);
341 w = x_parent->left_;
342 }

345 if (w->left_ != nullptr) {

867 pair<iterator, bool> insert_unique(link_type node) {
868 link_type y = header_;
869 link_type x = root();
870 bool comp = true;
871 while (x != nullptr) {
872 y = x;
