338template <typename Traits>
339constexpr size_t char_traits_find_bmh(const char_traits_ptr_t<Traits> dest, const size_t dest_size, const size_t start,
340 const char_traits_ptr_t<Traits> rsc, const size_t rsc_size) noexcept {
341 if (rsc_size == 1) {
342 return _NEFORCE char_traits_find_char<Traits>(dest, dest_size, start, rsc[0]);
343 }
344
345 size_t skip[256];
346 for (auto& s: skip) {
347 s = rsc_size;
348 }
349 for (size_t i = 0; i + 1 < rsc_size; ++i) {
350 skip[static_cast<byte_t>(rsc[i])] = rsc_size - 1 - i;
351 }
352
353 const auto last = rsc[rsc_size - 1];
354 size_t i = start;
355 const size_t limit = dest_size - rsc_size;
356
357 while (i <= limit) {
358 if (Traits::eq(dest[i + rsc_size - 1], last)) {
359 if (Traits::compare(dest + i, rsc, rsc_size) == 0) {
360 return i;
361 }
362 }
363 i += skip[static_cast<byte_t>(dest[i + rsc_size - 1])];
364 }
365 return static_cast<size_t>(-1);
366}

381constexpr size_t char_traits_rfind_bmh(const char_traits_ptr_t<Traits> dest, const size_t dest_size, const size_t start,
382 const char_traits_ptr_t<Traits> rsc, const size_t rsc_size) noexcept {
383 if (rsc_size == 1) {
384 return _NEFORCE char_traits_rfind_char<Traits>(dest, dest_size, start, rsc[0]);
385 }

387 size_t skip[256];
388 for (auto& s: skip) {
389 s = rsc_size;
390 }
391 for (size_t i = 0; i + 1 < rsc_size; ++i) {
392 skip[static_cast<byte_t>(rsc[i])] = rsc_size - 1 - i;
393 }
394
395 const auto last = rsc[rsc_size - 1];
396 auto last_found = static_cast<size_t>(-1);
397 const size_t limit = _NEFORCE min(start, dest_size - rsc_size);

401 if (Traits::eq(dest[i + rsc_size - 1], last)) {
402 if (Traits::compare(dest + i, rsc, rsc_size) == 0) {
403 last_found = i;
404 ++i;
405 continue;
406 }
407 }
408 i += skip[static_cast<byte_t>(dest[i + rsc_size - 1])];
409 }

417 const size_t rsc_size) noexcept {
418 const auto may_match_end = dest + (dest_size - rsc_size) + 1;
