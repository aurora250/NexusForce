377template <typename Traits>
378constexpr size_t char_traits_find(const char_traits_ptr_t<Traits> dest, const size_t dest_size,
379 const size_t start, const char_traits_ptr_t<Traits> rsc, const size_t rsc_size) noexcept {
380 if (rsc_size > dest_size || start > dest_size - rsc_size) return static_cast<size_t>(-1);
381 if (rsc_size == 0) return start;
382
383 const auto may_match_end = dest + (dest_size - rsc_size) + 1;
384 for (auto if_match = dest + start; ; ++if_match) {
385 if_match = Traits::find(if_match, static_cast<size_t>(may_match_end - if_match), *rsc);
386 if (!if_match) return static_cast<size_t>(-1);
387
388 if (Traits::compare(if_match, rsc, rsc_size) == 0)
389 return static_cast<size_t>(if_match - dest);
390 }
391}
