714constexpr void uint128_t::divmod128(const uint128_t& dividend, const uint128_t& divisor, uint128_t& quotient,
715 uint128_t& remainder) {
716 if (!divisor) {
717 NEFORCE_THROW_EXCEPTION(math_exception("Division by zero"));
718 }
719#if defined(NEFORCE_SUPPORT_INTRINSIC_INT128) && !defined(NEFORCE_PLATFORM_WINDOWS)
720 const unsigned __int128 d = (static_cast<unsigned __int128>(dividend.hi) << 64) | dividend.lo;
721 const unsigned __int128 r = (static_cast<unsigned __int128>(divisor.hi) << 64) | divisor.lo;
722 const unsigned __int128 q = d / r;
723 const unsigned __int128 m = d % r;
724 quotient = uint128_t{static_cast<uint64_t>(q >> 64), static_cast<uint64_t>(q)};
725 remainder = uint128_t{static_cast<uint64_t>(m >> 64), static_cast<uint64_t>(m)};
726#else
727 if (divisor == uint128_t{1ULL}) {
728 quotient = dividend;
729 remainder = uint128_t{0ULL};
730 return;
731 }
732 if (dividend < divisor) {
733 quotient = uint128_t{0ULL};
734 remainder = dividend;
735 return;
736 }
737
738 const int bits = dividend.hi != 0 ? (128 - clz64(dividend.hi)) : (dividend.lo != 0 ? (64 - clz64(dividend.lo)) : 0);
739
740 quotient = uint128_t{0ULL};
741 remainder = uint128_t{0ULL};
742
743 for (int i = bits - 1; i >= 0; --i) {
744 remainder <<= 1;
745 const uint64_t bit = (i >= 64) ? ((dividend.hi >> (i - 64)) & 1ULL) : ((dividend.lo >> i) & 1ULL);
746 remainder.lo |= bit;
747 if (remainder >= divisor) {
748 remainder -= divisor;
749 if (i >= 64) {
750 quotient.hi |= (static_cast<uint64_t>(1) << (i - 64));
751 } else {
752 quotient.lo |= (static_cast<uint64_t>(1) << i);
753 }
754 }
755 }
756#endif
757}

759constexpr int128_t& int128_t::operator/=(const int128_t& other) {
760 if (!other) {
761 NEFORCE_THROW_EXCEPTION(math_exception("Division by zero"));
762 }
763 const bool neg_a = is_negative();
764 const bool neg_b = other.is_negative();
765 const int128_t abs_a = neg_a ? negation() : *this;
766 const int128_t abs_b = neg_b ? other.negation() : other;
767 const uint128_t q = abs_a.to_uint128() / abs_b.to_uint128();
768 const int128_t result(q.hi, q.lo);
769 *this = ((neg_a ^ neg_b) != 0) ? result.negation() : result;
770 return *this;
771}
