117NEFORCE_CONSTEXPR20 format_options parse_number_format(const string_view& fmt_str) {
118 format_options options;
119 size_t pos = 0;
120
121 if (fmt_str.empty()) return options;
122
123 bool found_align = false;
124 if (pos + 1 < fmt_str.size()) {
125 const char second_char = fmt_str[pos + 1];
126 const char first_char = fmt_str[pos];
127
128 if ((second_char == '<' || second_char == '>' ||
129 second_char == '^' || second_char == '=') &&
130 (first_char != '+' && first_char != '-' && first_char != ' ')) {
131 options.fill = first_char;
132 options.alignment = to_number_alignment(second_char);
133 pos += 2;
134 found_align = true;
135 }
136 }
137
138 if (!found_align && pos < fmt_str.size()) {
139 const char c = fmt_str[pos];
140 if (c == '<' || c == '>' || c == '^' || c == '=') {
141 options.alignment = to_number_alignment(c);
142 ++pos;
143 found_align = true;
144 }
145 }
146
147 if (pos < fmt_str.size()) {
148 const char c = fmt_str[pos];
149 if (c == '+' || c == '-' || c == ' ') {
150 options.sign_mode = c;
151 ++pos;
152 }
153 }
154
155 if (pos < fmt_str.size() && fmt_str[pos] == '#') {
156 options.show_base = true;
157 ++pos;
158 }
159
160 if (pos < fmt_str.size() && fmt_str[pos] == '0' &&
161 options.fill == ' ' && options.alignment == FORMAT_ALIGN::RIGHT) {
162 options.zero_pad = true;
163 options.fill = '0';
164 options.alignment = FORMAT_ALIGN::INTERNAL;
165 ++pos;
166 }
167
168 if (pos < fmt_str.size() && _NEFORCE is_digit(fmt_str[pos])) {
169 int width = 0;
170 while (pos < fmt_str.size() && _NEFORCE is_digit(fmt_str[pos])) {
171 width = width * 10 + (fmt_str[pos] - '0');
172 ++pos;
173 }
174 options.width = width;
175 }
176
177 if (pos < fmt_str.size() && fmt_str[pos] == '.') {
178 ++pos;
179 int precision = 0;
180 while (pos < fmt_str.size() && _NEFORCE is_digit(fmt_str[pos])) {
181 precision = precision * 10 + (fmt_str[pos] - '0');
182 ++pos;
183 }
184 options.precision = precision;
185 }
186
187 if (pos < fmt_str.size()) {
188 options.type = to_number_type(fmt_str[pos]);
189 ++pos;
190 }
191
192 return options;
193}

226 string prefix;

228 prefix += "-";

232 prefix += " ";
233 }
234
235 string number_str;
236 const int precision = options.precision >= 0 ? options.precision : 6;
237
238 switch (options.type) {

264 const string fill_str(fill_count, options.fill);

581NEFORCE_CONSTEXPR20 string format_impl(const string_view fmt, size_t& pos) {
582 string result;
583 while (pos < fmt.size()) {
584 if (fmt[pos] == '{') {
585 if (pos + 1 < fmt.size() && fmt[pos + 1] == '{') {
586 result += '{';
587 pos += 2;

593 result += '}';
594 pos += 2;
595 } else {
596 throw_exception(value_exception("Unmatched '}' in format string"));
597 }
598 } else {
599 result += fmt[pos];

601 }

617template <typename First, typename... Rest>
618NEFORCE_CONSTEXPR20 string format_impl(const string_view fmt, size_t& pos, First&& first, Rest&&... rest) {
619 string result;
620 while (pos < fmt.size()) {
621 if (fmt[pos] == '{') {
622 if (pos + 1 < fmt.size() && fmt[pos + 1] == '{') {
623 result += '{';
624 pos += 2;
625 continue;
626 }
627 ++pos;
628 size_t end_pos = pos;
629 int depth = 1;
630 while (end_pos < fmt.size() && depth > 0) {
631 if (fmt[end_pos] == '{') {
632 ++depth;
633 } else if (fmt[end_pos] == '}') {
634 --depth;
635 }
636 if (depth > 0) ++end_pos;
637 }
638 if (depth != 0) {
639 throw_exception(value_exception("Unmatched '{' in format string"));
640 }
641
642 const string_view spec_str = fmt.substr(pos, end_pos - pos);
643 pos = end_pos + 1;
644 format_options opts;
645 if (spec_str.empty()) {
646 opts = _INNER parse_number_format("");
647 } else if (spec_str[0] == ':') {
648 opts = _INNER parse_number_format(spec_str.substr(1));
649 } else {
650 opts = _INNER parse_number_format(spec_str);
651 }
652 result += formatter<decay_t<First>>()(_NEFORCE forward<First>(first), opts);
653 result += _INNER format_impl(fmt, pos, _NEFORCE forward<Rest>(rest)...);
654 return result;
655 } else if (fmt[pos] == '}') {
656 if (pos + 1 < fmt.size() && fmt[pos + 1] == '}') {
657 result += '}';
658 pos += 2;
659 } else {
660 throw_exception(value_exception("Unmatched '}' in format string"));
661 }
662 } else {
663 result += fmt[pos];
664 ++pos;
665 }
666 }
667 return result;
668}
