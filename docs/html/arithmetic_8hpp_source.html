554 // Use unsigned negation to avoid UB, matches x86 PABSB wrap-around behavior
555 result.data[i] =
556 val < 0 ? static_cast<byte_t>(static_cast<int8_t>(-static_cast<unsigned>(static_cast<uint8_t>(val))))
