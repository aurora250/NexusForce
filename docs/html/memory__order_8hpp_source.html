273constexpr memory_order cmpexch_failure_order(const memory_order mo) noexcept {
274#ifdef NEFORCE_SUPPORT_INTEL_TSX

276 const memory_order base_mo = mo & mask;
277 const memory_order failure_order = base_mo == memory_order_acq_rel ? memory_order_acquire
278 : base_mo == memory_order_release ? memory_order_relaxed
279 : base_mo;

282 return failure_order | modifiers;
