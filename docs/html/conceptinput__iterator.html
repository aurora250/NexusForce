template<typename Iterator>
concept input_iterator =
both_equality_comparable<Iterator, Iterator> && iterator_typedef<Iterator> && requires(Iterator it) {
{ *it } -> convertible_to<typename iterator_traits<Iterator>::value_type>;
{ ++it } -> same_as<Iterator&>;
{ it++ } -> same_as<Iterator>;
}
