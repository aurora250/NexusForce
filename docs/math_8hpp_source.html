286template <typename T>
287NEFORCE_PURE_FUNCTION NEFORCE_CONSTEXPR14 T logarithm_e(const T x) {
288 static_assert(is_floating_point_v<T>, "floating point required");
289 if (x <= 0) {
290 NEFORCE_THROW_EXCEPTION(math_exception("Logarithm domain error"));
291 }
292 uint32_t N = constants::TAYLOR_CONVERGENCE;
293 const T a = (x - 1) / (x + 1);
294 const T a_sqrt = a * a;
295 T nk = 2 * N + 1;
296 T y = 1.0 / nk;
297 while (N--) {
298 nk -= 2;
299 y = 1.0 / nk + a_sqrt * y;
300 }
301 return 2.0 * a * y;
302}

610NEFORCE_PURE_FUNCTION NEFORCE_CONSTEXPR14 decimal_t sine(decimal_t x) noexcept {
611 decimal_t sign = 1.0;
612 if (x < 0) {
613 sign = -1.0;
614 x = -x;
615 }
616 constexpr decimal_t twoPi = 2 * constants::PI;
617 x = x - twoPi * floor(x / twoPi);
618
619 if (x > constants::PI) {
620 x -= constants::PI;
621 sign = -sign;
622 }
623 if (x > constants::PI / 2) {
624 x = constants::PI - x;
625 }
626
627 decimal_t i = 1;
628 int32_t neg = 1;
629 decimal_t term = 0;
630 decimal_t idx = x;
631 decimal_t fac = 1;
632 decimal_t taylor = x;
633 do {
634 fac = fac * (i + 1) * (i + 2);
635 idx *= x * x;
636 neg = -neg;
637 term = idx / fac * neg;
638 taylor += term;
639 i += 2;
640 } while (absolute(term) > constants::EPSILON);
641 const auto fin = sign * taylor;
642 return around_zero(fin) ? 0 : fin;
643}

691NEFORCE_PURE_FUNCTION NEFORCE_CONSTEXPR14 decimal_t __arctangent_taylor(const decimal_t x) noexcept {
692 const decimal_t x_sq = x * x;
693 decimal_t term = x;
694 decimal_t sum = x;
695 decimal_t n = 1.0;
696 while (absolute(term) > constants::EPSILON) {

698 n += 2.0;
