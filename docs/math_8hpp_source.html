273MSTL_CONST_FUNCTION MSTL_CONSTEXPR14 T power(const T& x, UT n) noexcept {
274 constexpr UT zero = UT(0);
275 constexpr UT one = UT(1);
276 constexpr UT two = UT(2);
277
278 if (n == zero) return 1;
279 T result(1);
280 T base = x;
281 while (n > zero) {
282 if (n % two == one) {
283 result *= base;
284 }
285 base *= base;
286 n /= two;
287 }
288 return result;
289}

307MSTL_PURE_FUNCTION MSTL_CONSTEXPR14 decimal_t logarithm_e(const decimal_t x) noexcept {
308 uint32_t N = _CONSTANTS TAYLOR_CONVERGENCE;
309 const decimal_t a = (x - 1) / (x + 1);
310 const decimal_t a_sqrt = a * a;
311 decimal_t nk = 2 * N + 1;
312 decimal_t y = 1.0 / nk;
313 while (N--) {
314 nk -= 2;
315 y = 1.0 / nk + a_sqrt * y;
316 }
317 return 2.0 * a * y;
318}

613MSTL_PURE_FUNCTION MSTL_CONSTEXPR14 decimal_t sine(decimal_t x) noexcept {
614 decimal_t sign = 1.0;
615 if (x < 0) {
616 sign = -1.0;
617 x = -x;
618 }
619 constexpr decimal_t twoPi = 2 * _CONSTANTS PI;
620 x = x - twoPi * floor(x / twoPi);
621
622 if (x > _CONSTANTS PI) {
623 x -= _CONSTANTS PI;
624 sign = -sign;
625 }
626 if (x > _CONSTANTS PI / 2) {
627 x = _CONSTANTS PI - x;
628 }
629
630 decimal_t i = 1;
631 int32_t neg = 1;
632 decimal_t sum = 0;
633 decimal_t idx = x;
634 decimal_t fac = 1;
635 decimal_t taylor = x;
636 do {
637 fac = fac * (i + 1) * (i + 2);
638 idx *= x * x;
639 neg = -neg;
640 sum = idx / fac * neg;
641 taylor += sum;
642 i += 2;
643 } while (absolute(sum) > _CONSTANTS EPSILON);
644 const auto fin = sign * taylor;
645 return around_zero(fin) ? 0 : fin;
646}

695 const decimal_t x_sq = x * x;
