202Iterator2 partial_sort_copy(Iterator1 first, Iterator1 last, Iterator2 result_first, Iterator2 result_last, Compare comp) {
203 static_assert(is_ranges_input_iter_v<Iterator1>, "Iterator must be input_iterator");
204
205 if (result_first == result_last) return result_last;
206 Iterator2 result_real_last = result_first;

208 *result_real_last = *first;

466template <typename Iterator, typename Compare>
467void sort(Iterator first, Iterator last, Compare comp) {
468 if (first == last) return;
469
470 inner::__intro_sort_dispatch(first, last, inner::__log2_int(last - first) * 2, comp);
471 constexpr size_t threshhold = MEMORY_ALIGN_THRESHHOLD;
472
473 if (last - first > threshhold) {
474 _NEFORCE insertion_sort(first, first + threshhold, comp);
475 for (Iterator i = first + threshhold; i != last; ++i) {
476 inner::__insertion_sort_aux(i, *i, comp);
477 }
478 }
479 else {
480 _NEFORCE insertion_sort(first, last, comp);
481 }
482}
