191Iterator2 partial_sort_copy(Iterator1 first, Iterator1 last,
192 Iterator2 result_first, Iterator2 result_last, Compare comp) {
193 if (result_first == result_last) return result_last;
194 Iterator2 result_real_last = result_first;

447template <typename Iterator, typename Compare, enable_if_t<
448 is_ranges_rnd_iter_v<Iterator>, int> = 0>
449void sort(Iterator first, Iterator last, Compare comp) {
450 if (first == last) return;
451 _INNER __intro_sort_dispatch(first, last, _INNER __log2_int(last - first) * 2, comp);
452 if (last - first > MEMORY_ALIGN_THRESHHOLD) {
453 _NEFORCE insertion_sort(first, first + MEMORY_ALIGN_THRESHHOLD, comp);
454 for (Iterator i = first + MEMORY_ALIGN_THRESHHOLD; i != last; ++i) {
455 _INNER __insertion_sort_aux(i, *i, comp);
456 }
457 }
458 else {
459 _NEFORCE insertion_sort(first, last, comp);
460 }
461}
