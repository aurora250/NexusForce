736template <typename T, typename... Args>
737enable_if_t<!is_unbounded_array_v<T> && is_constructible_v<T, Args...>, shared_ptr<T>>
738make_shared(Args&&... args) {
739 auto const deleter = [](T* ptr) noexcept { ptr->~T(); };
740 using Counter = _INNER __smart_ptr_counter_impl_fused<T, decltype(deleter)>;
741 constexpr size_t align = max(alignof(T), alignof(Counter));
742 constexpr size_t offset = (sizeof(Counter) + align - 1) & ~(align - 1);
743 constexpr size_t size = offset + sizeof(T);
744#if NEFORCE_STANDARD_17
745 void* mem = ::operator new(size, static_cast<std::align_val_t>(align));
746 Counter* counter = static_cast<Counter*>(mem);
747#else
748 void* mem = ::operator new(size + align - 1);
749 size_t aligned_addr = (reinterpret_cast<size_t>(mem) + (align - 1)) & ~(align - 1);
750 Counter* counter = reinterpret_cast<Counter*>(aligned_addr);
751#endif
752 T* object = reinterpret_cast<T*>(reinterpret_cast<byte_t*>(counter) + offset);
753 try {
754 _NEFORCE construct(object, _NEFORCE forward<Args>(args)...);
755 } catch (...) {
756#if NEFORCE_STANDARD_17
757 operator delete(mem, static_cast<std::align_val_t>(align));
758#else
759 operator delete(mem);
760#endif
761 throw_exception(memory_exception("shared ptr construction failed."));
762 }
763 new (counter) Counter(object, mem, _NEFORCE move(deleter));
764 _INNER __setup_enable_shared_from(object, counter);
765 return _INNER __make_shared_fused(object, counter);
766}

799template <typename T, typename Alloc, typename... Args>
800enable_if_t<!is_array_v<T> && is_constructible_v<T, Args...>, shared_ptr<T>>
801allocate_shared(Alloc& alloc, Args&&... args) {
802 auto deleter = [](T* p) { p->~T(); };
803 using ControlBlock = _INNER __smart_ptr_counter_impl_allocated<T, decltype(deleter), Alloc>;
804
805 const size_t align = _NEFORCE max(alignof(ControlBlock), alignof(T));
806 const size_t offset = (sizeof(ControlBlock) + align - 1) & ~(align - 1);
807 const size_t total_size = offset + sizeof(T);
808 const size_t raw_size = total_size + align - 1;
809
810 using alloc_traits = allocator_traits<remove_cv_t<Alloc>>;
811 using byte_allocator = typename alloc_traits::template alloc_rebind_t<Alloc, byte_t>;
812 byte_allocator byte_alloc(alloc);
813
814 byte_t* raw_mem = allocator_traits<byte_allocator>::allocate(byte_alloc, raw_size);
815
816 const uintptr_t raw_addr = reinterpret_cast<uintptr_t>(raw_mem);
817 const uintptr_t aligned_addr = (raw_addr + align - 1) & ~static_cast<uintptr_t>(align - 1);
818 auto aligned_mem = reinterpret_cast<byte_t*>(aligned_addr);
819 T* object_ptr = reinterpret_cast<T*>(aligned_mem + offset);
820
821 try {
822 allocator_traits<Alloc>::construct(alloc, object_ptr, _NEFORCE forward<Args>(args)...);
823 } catch (...) {
824 allocator_traits<byte_allocator>::deallocate(byte_alloc, raw_mem, raw_size);
825 throw_exception(memory_exception("shared ptr ref object construction failed."));
826 }
827
828 ControlBlock* ctrl_block = nullptr;
829 try {
830 ctrl_block = ::new (aligned_mem) ControlBlock(object_ptr, raw_mem, raw_size, deleter, alloc);
831 } catch (...) {
832 allocator_traits<Alloc>::destroy(alloc, object_ptr);
833 allocator_traits<byte_allocator>::deallocate(byte_alloc, raw_mem, raw_size);
834 throw_exception(memory_exception("shared ptr control block construction failed."));
835 }
836
837 _INNER __setup_enable_shared_from(object_ptr, ctrl_block);
838 return _INNER __make_shared_fused(object_ptr, ctrl_block);
839}
