98 delete_this();

109 auto strong = strong_count_.load(memory_order_relaxed);
110 do {
111 if (strong == 0) return false;
112 } while (!strong_count_.compare_exchange_weak(
113 strong, strong + 1,
114 memory_order_release,
115 memory_order_relaxed
116 ));
117 return true;
118 }

773template <typename T, typename... Args>
774enable_if_t<!is_unbounded_array_v<T> && is_constructible_v<T, Args...>, shared_ptr<T>>
775make_shared(Args&&... args) {
776 auto const deleter =
777 [](T* ptr) noexcept(is_nothrow_destructible_v<T>) {
778 ptr->~T();
779 };
780 using Counter = inner::__smart_ptr_counter_impl_fused<T, decltype(deleter)>;
781 constexpr size_t align = max(alignof(T), alignof(Counter));
782 constexpr size_t offset = (sizeof(Counter) + align - 1) & ~(align - 1);
783 constexpr size_t size = offset + sizeof(T);
784#if NEFORCE_STANDARD_17
785 void* mem = operator new(size, std::align_val_t{ align });
786 auto* counter = static_cast<Counter*>(mem);
787#else
788 void* mem = operator new(size + align - 1);
789 size_t aligned_addr = (reinterpret_cast<size_t>(mem) + (align - 1)) & ~(align - 1);
790 Counter* counter = reinterpret_cast<Counter*>(aligned_addr);
791#endif
792 T* object = reinterpret_cast<T*>(reinterpret_cast<byte_t*>(counter) + offset);
793 try {
794 _NEFORCE construct(object, _NEFORCE forward<Args>( args)...);
795 } catch (...) {
796#if NEFORCE_STANDARD_17
797 operator delete(mem, std::align_val_t{ align });
798#else
799 operator delete(mem);
800#endif
801 NEFORCE_THROW_EXCEPTION(memory_exception("shared ptr construction failed."));
802 }
803 _NEFORCE construct(reinterpret_cast<Counter*>(counter), object, mem, align, _NEFORCE move(deleter));
804 inner::__setup_enable_shared_from(object, counter);
805 return inner::__make_shared_fused(object, counter);
806}

842template <typename T, typename Alloc, typename... Args>
843enable_if_t<!is_array_v<T> && is_constructible_v<T, Args...>, shared_ptr<T>>
844allocate_shared(Alloc& alloc, Args&&... args) {
845 auto deleter = [](T* p) { p->~T(); };
846 using ControlBlock = inner::__smart_ptr_counter_impl_allocated<T, decltype(deleter), Alloc>;
847
848 const size_t align = _NEFORCE max(alignof(ControlBlock), alignof(T));
849 const size_t offset = (sizeof(ControlBlock) + align - 1) & ~(align - 1);
850 const size_t total_size = offset + sizeof(T);
851 const size_t raw_size = total_size + align - 1;
852
853 using alloc_traits = allocator_traits<remove_cv_t<Alloc>>;
854 using byte_allocator = typename alloc_traits::template alloc_rebind_t<Alloc, byte_t>;
855 byte_allocator byte_alloc(alloc);
856
857 byte_t* raw_mem = allocator_traits<byte_allocator>::allocate(byte_alloc, raw_size);
858
859 const uintptr_t raw_addr = reinterpret_cast<uintptr_t>(raw_mem);
860 const uintptr_t aligned_addr = (raw_addr + align - 1) & ~static_cast<uintptr_t>(align - 1);
861 auto aligned_mem = reinterpret_cast<byte_t*>(aligned_addr);
862 T* object_ptr = reinterpret_cast<T*>(aligned_mem + offset);
863
864 try {
865 allocator_traits<Alloc>::construct(alloc, object_ptr, _NEFORCE forward<Args>(args)...);
866 } catch (...) {
867 allocator_traits<byte_allocator>::deallocate(byte_alloc, raw_mem, raw_size);
868 NEFORCE_THROW_EXCEPTION(memory_exception("shared ptr ref object construction failed."));
869 }
870
871 ControlBlock* ctrl_block = nullptr;
872 try {
873 ctrl_block = _NEFORCE construct(
874 reinterpret_cast<ControlBlock*>(aligned_mem),
875 object_ptr, raw_mem, raw_size, deleter, alloc);
876 } catch (...) {
877 allocator_traits<Alloc>::destroy(alloc, object_ptr);
878 allocator_traits<byte_allocator>::deallocate(byte_alloc, raw_mem, raw_size);
879 NEFORCE_THROW_EXCEPTION(memory_exception("shared ptr control block construction failed."));
880 }
881
882 inner::__setup_enable_shared_from(object_ptr, ctrl_block);
883 return inner::__make_shared_fused(object_ptr, ctrl_block);
884}

962template <typename T>
963class smart_pointer_atomic {
964public:
965 using value_type = T;
966
967private:
968 struct atomic_counter {
969 public:
970 using count_type = inner::__smart_ptr_counter;
971
972 private:
973 mutable atomic_base<uintptr_t> value_{0};
974
975 static constexpr uintptr_t lock_bit{1};
976
977 static void dereference(count_type* counter, true_type) noexcept {
978 counter->decref_strong();
979 }
980
981 static void dereference(count_type* counter, false_type) noexcept {
982 counter->decref_weak();
983 }
984
985 public:
986 constexpr atomic_counter() noexcept = default;
987
988 explicit atomic_counter(count_type* counter) noexcept
989 : value_(reinterpret_cast<uintptr_t>(counter)) {
990 counter = nullptr;
991 }
992
993 ~atomic_counter() {
994 auto value = value_.load(memory_order_relaxed);
995 NEFORCE_CONSTEXPR_ASSERT(!(value & lock_bit));
996 if (auto counter = reinterpret_cast<count_type*>(value)) {
997 this->dereference(counter, is_shared_ptr<T>());
998 }
999 }
1000
1001 atomic_counter(const atomic_counter&) = delete;
1002 atomic_counter& operator =(const atomic_counter&) = delete;
1003
1004 count_type* lock(memory_order mo) const noexcept {
1005 if (mo != memory_order_seq_cst) {
1006 mo = memory_order_acquire;
1007 }
1008 auto cur = value_.load(memory_order_relaxed);
1009 while (cur & lock_bit) {
1010 this_thread::relax();
1011 cur = value_.load(memory_order_relaxed);
1012 }
1013
1014 while (!value_.compare_exchange_strong(cur, cur | lock_bit, mo, memory_order_relaxed)) {
1015 this_thread::relax();
1016 cur = cur & ~lock_bit;
1017 }
1018 return reinterpret_cast<count_type*>(cur);
1019 }
1020
1021 void unlock(memory_order mo) const noexcept {
1022 value_.fetch_sub(1, mo);
1023 }
1024
1025 count_type* swap_unlock(count_type* counter, memory_order mo) noexcept {
1026 if (mo != memory_order_seq_cst) {
1027 mo = memory_order_release;
1028 }
