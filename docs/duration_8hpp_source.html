474template <typename Rep, typename Period>
475void sleep_for(const duration<Rep, Period>& time) {
476 if (time <= time.zero()) return;
477
478#ifdef MSTL_PLATFORM_WINDOWS__
479 auto ns = duration_cast<nanoseconds>(time);
480 ::LARGE_INTEGER li{};
481 li.QuadPart = -(ns.count() / 100);
482
483 const ::HANDLE timer = ::CreateWaitableTimerW(nullptr, 1, nullptr);
484 if (timer) {
485 ::SetWaitableTimer(timer, &li, 0, nullptr, nullptr, 0);
486 ::WaitForSingleObject(timer, INFINITE);
487 ::CloseHandle(timer);
488 }
489#elif defined(MSTL_PLATFORM_LINUX__)
490 auto s = duration_cast<seconds>(time);
491 auto ns = duration_cast<nanoseconds>(time - s);
492 ::timespec ts{static_cast<std::time_t>(s.count()), static_cast<long>(ns.count())};
493 while (::nanosleep(&ts, &ts) == -1) {}
494#endif
495}
