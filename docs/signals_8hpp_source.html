493 void emit(Types... args) {
494 vector<slot_entry> snapshot;
495 {
496 lock<mutex> lk(mutex_);
497 if (is_blocked()) return;
498 snapshot = slots_;
499 }
500
501 for (auto& slot : snapshot) {
502 if (!slot.connected_flag || !(*slot.connected_flag)) continue;
503
504 const callback_result res = slot.callback(args...);
505 if (res == callback_result::erase) {
506 *slot.connected_flag = false;
507 }
508 }
509
510 {
511 lock<mutex> lk(mutex_);
512 slots_.erase(
513 _NEFORCE remove_if(slots_.begin(), slots_.end(),
514 [](const slot_entry& s) {
515 return !s.connected_flag || !(*s.connected_flag);
516 }
517 ),
518 slots_.end()
519 );
