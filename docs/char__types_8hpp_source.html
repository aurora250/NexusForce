113template <typename CharT>
114MSTL_PURE_FUNCTION MSTL_CONSTEXPR14 bool is_ctype(const CharT c, uint64_t mask_low, uint64_t mask_high) noexcept {
115 static_assert(is_character_v<CharT>, "character type is necessary");
116 const auto uc = static_cast<make_unsigned_t<CharT>>(c);
117 if (uc > 127) return false;
118 if (uc <= 63) return (mask_low & (1ULL << uc)) != 0;
119 const auto offset = uc - 64;
120 return (mask_high & (1ULL << offset)) != 0;
121}

260template <typename CharT>
261MSTL_CONST_FUNCTION MSTL_CONSTEXPR14 bool is_xdigit(const CharT c) noexcept {
262 static_assert(is_character_v<CharT>, "character type is necessary");
263 const auto uc = static_cast<make_unsigned_t<CharT>>(c);
264 if (uc > 127) return false;
265 const bool is_09 = (uc & 0xF0) == 0x30 && (uc & 0x0F) <= 0x09;
266 const bool is_AF = (uc & 0xF0) == 0x40 && (uc & 0x0F) >= 0x01 && (uc & 0x0F) <= 0x06;

268 return is_09 || is_AF || is_af;
269}
