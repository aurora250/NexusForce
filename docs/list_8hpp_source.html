682 void insert(iterator position, Iterator first, Iterator last) {
683 if (first == last) return;
684
685 link_type original_prev = position.base()->prev;
686 link_type current_prev = original_prev;
687 link_type first_inserted = nullptr;
688
689 try {
690 while (first != last) {
691 link_type temp = list::create_node(*first);
692 temp->prev = current_prev;
693 temp->next = position.base();
694 current_prev->next = temp;
695 position.base()->prev = temp;
696 if (first_inserted == nullptr) {
697 first_inserted = temp;
698 }
699 current_prev = temp;
700 ++pair_.value;
701 ++first;
702 }
703 } catch (...) {
704 if (first_inserted != nullptr) {
705 link_type to_delete = first_inserted;
706 while (to_delete != position.base()) {
707 link_type next = to_delete->next;
708 list::destroy_node(to_delete);
709 --pair_.value;
710 to_delete = next;
711 }
712 original_prev->next = position.base();
713 position.base()->prev = original_prev;
714 }
715 throw;
716 }
