63 if (expire > other.expire) return false;

65 }

88 void run() {
89 while (!stopped_.load()) {
90 smart_lock<mutex> lock(mutex_);
91
92 if (nodes_.empty()) {
93 cv_.wait(lock, [this] {
94 return stopped_.load() || !nodes_.empty();
95 });
96 if (stopped_.load()) break;
97 }
98
99 time_point now = clock_type::now();
100 while (!nodes_.empty() && nodes_.begin()->expire <= now) {
101 auto it = nodes_.begin();
102 node current_node = *it;
103 nodes_.erase(it);
104 node_map_.erase(current_node.id);
105
106 lock.unlock_quiet();
107 if (!stopped_.load()) {
108 current_node.handler();
109 }
110 lock.lock_quiet();
111 now = clock_type::now();
112 }
113
114 if (!nodes_.empty()) {
115 time_point next_expire = nodes_.begin()->expire;
116 cv_.wait_until(lock, next_expire);
117 }
118 }
119 }

128 }

154 token add_task(time_point expire, handler_type&& handler) {
155 smart_lock<mutex> lock(mutex_);
156 token id = next_id_++;
157
158 const bool is_earliest = nodes_.empty() || expire < nodes_.begin()->expire;
159
160 node new_node(expire, id, _NEFORCE move(handler));
161 auto result = nodes_.insert(new_node);
162 node_map_[id] = result.first;
163
164 lock.unlock_quiet();
165
166 if (is_earliest) {
167 cv_.notify_one();
168 }
169
170 return id;
171 }

180 bool cancel(token id) {
181 smart_lock<mutex> lock(mutex_);
182 auto it_map = node_map_.find(id);
183 if (it_map == node_map_.end()) {
184 return false;
185 }
186
187 const bool is_earliest = (it_map->second == nodes_.begin());
188 nodes_.erase(it_map->second);
189 node_map_.erase(it_map);
190
191 lock.unlock_quiet();
192
193 if (is_earliest) {
194 cv_.notify_one();
195 }
196
197 return true;
198 }

205 nodes_.clear();
206 node_map_.clear();
207 lock.unlock_quiet();

209 }
