776 iter_value_t<Iterator> value = *initial;
777 Iterator ptr1 = initial;
778 Iterator ptr2 = ptr1 + shift;

780 *ptr1 = *ptr2;

783 ptr2 += shift;

785 ptr2 = first + (shift - (last - ptr2));
