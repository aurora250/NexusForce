188 Iterator first,
189 iter_difference_t<Iterator> hole_index,
190 iter_difference_t<Iterator> len,
191 T value, Compare comp) {
192 auto top_index = hole_index;
193 auto child = 2 * hole_index + 1;
194 while (child < len) {
