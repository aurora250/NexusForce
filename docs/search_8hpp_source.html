43constexpr Iterator lower_bound(Iterator first, Iterator last, const T& value, Compare comp) {
44 static_assert(is_ranges_fwd_iter_v<Iterator>, "Iterator must be forward_iterator");
45
46 using Distance = iter_difference_t<Iterator>;
47 Distance len = _NEFORCE distance(first, last);
48 Distance half;
49 Iterator middle;
50 while (len > 0) {
51 half = len >> 1;
52 middle = first;
53 _NEFORCE advance(middle, half);
54 if (comp(*middle, value)) {
55 first = middle;
56 ++first;
57 len = len - half - 1;
58 }

60 }

454template <typename Iterator1, typename Iterator2, typename BinaryPredicate>
455constexpr Iterator1 search(
456 Iterator1 first1, Iterator1 last1, Iterator2 first2,
457 Iterator2 last2, BinaryPredicate binary_pred) {
458
459 static_assert(is_ranges_fwd_iter_v<Iterator1> && is_ranges_fwd_iter_v<Iterator2>, "Iterator must be forward_iterator");
460
461 const auto d1 = _NEFORCE distance(first1, last1);
462 const auto d2 = _NEFORCE distance(first2, last2);
463 if (d1 < d2) return last1;
464
465 Iterator1 current1 = first1;
466 Iterator2 current2 = first2;
467
468 while (current2 != last2) {
469 if (binary_pred(*current1, *current2)) {
470 ++current1;
471 ++current2;
472 } else {
473 if (d1 == d2) return last1;
474
475 current1 = ++first1;
476 current2 = first2;
477 --d1;
478 }
479 }
480 return first1;
481}

508template <typename Iterator, typename T>
509constexpr Iterator search_n(Iterator first, Iterator last, const size_t count, const T& value) {
510 static_assert(is_ranges_fwd_iter_v<Iterator>, "Iterator must be forward_iterator");
511
512 first = _NEFORCE find(first, last, value);
513 while (first != last) {
514 size_t n = count - 1;
515 Iterator i = first;
516 ++i;
517 while (i != last && n != 0 && *i == value) {
518 ++i;
519 --n;
520 }
521 if (n == 0) return first;
522
523 first = _NEFORCE find(i, last, value);
524 }
525 return last;

541constexpr Iterator search_n(Iterator first, Iterator last, const size_t count, const T& value, BinaryPredicate binary_pred) {
542 static_assert(is_ranges_fwd_iter_v<Iterator>, "Iterator must be forward_iterator");
543
544 while (first != last) {
545 if (binary_pred(*first, value)) break;
546 ++first;

550 size_t n = count - 1;
