82 if (first == last) return;

87 while (swapped) {

102 --prev;

106 }

108 ++left;
109 }

178template <typename Iterator, typename Compare, enable_if_t<
179 is_ranges_rnd_iter_v<Iterator>, int> = 0>
180NEFORCE_CONSTEXPR20 void shell_sort(Iterator first, Iterator last, Compare comp) {
181 if (first == last) return;
182 using Distance = iter_difference_t<Iterator>;
183 using T = iter_value_t<Iterator>;
184 Distance dist = _NEFORCE distance(first, last);
185 for (Distance gap = dist / 2; gap > 0; gap /= 2) {
186 for (Iterator i = first + gap; i < last; ++i) {
187 T temp = *i;
188 Iterator j;
189 for (j = i; j >= first + gap && comp(temp, *(j - gap)); j -= gap)
190 *j = *(j - gap);
191 *j = temp;
192 }
193 }
194}

223template <typename Iterator, typename Compare, typename IndexMapper, enable_if_t<

225NEFORCE_CONSTEXPR20 void counting_sort(Iterator first, Iterator last, Compare comp, IndexMapper mapper) {
226 if (first == last) return;
227 auto min_max = _NEFORCE minmax_element(first, last, comp);
228 auto min_val = mapper(*min_max.first);
229 auto max_val = mapper(*min_max.second);
230 const auto range = static_cast<size_t>(max_val - min_val + 1);
231 vector<int> count(range, 0);
232
233 for (Iterator it = first; it != last; ++it) {
234 auto value = mapper(*it);
235 if (value < min_val || value > max_val) {
236 throw_exception(iterator_exception("element out of range for counting sort."));
237 }
238 ++count[static_cast<size_t>(value - min_val)];
239 }
240
241 for (size_t i = 1; i < count.size(); ++i)
242 count[i] += count[i - 1];
243
244 vector<iter_value_t<Iterator>> sorted(_NEFORCE distance(first, last));
245 auto bound = _NEFORCE make_reverse_iterator(first);
246
247 for (auto rit = _NEFORCE make_reverse_iterator(last); rit != bound; ++rit) {
248 auto value = mapper(*rit);
249 const auto index = static_cast<size_t>(value - min_val);
250 size_t position = --count[index];
251 sorted[position] = *rit;
252 }
253 _NEFORCE copy(sorted.begin(), sorted.end(), first);
254}
