79template <typename Iterator, typename T, typename Compare, enable_if_t<is_ranges_fwd_iter_v<Iterator>, int> = 0>
80constexpr pair<Iterator, Iterator> equal_range(Iterator first, Iterator last, const T& value, Compare comp) {
81 using Distance = iter_difference_t<Iterator>;
82 Distance len = _MSTL distance(first, last);
83 Distance half;
84 Iterator middle, left, right;
85 while (len > 0) {
86 half = len >> 1;
87 middle = first;
88 _MSTL advance(middle, half);
89 if (comp(*middle, value)) {
90 first = middle;
91 ++first;
92 len = len - half - 1;
93 } else if (comp(value, *middle)) {
94 len = half;
95 } else {
96 left = _MSTL lower_bound(first, middle, value, comp);
97 _MSTL advance(first, len);
98 right = _MSTL upper_bound(++middle, first, value, comp);
99 return pair<Iterator, Iterator>(left, right);
100 }
101 }
102 return pair<Iterator, Iterator>(first, first);
103}

243template <typename Iterator, typename Compare, enable_if_t<is_ranges_input_iter_v<Iterator>, int> = 0>
244pair<iter_value_t<Iterator>, iter_value_t<Iterator>>
245constexpr minmax(Iterator first, Iterator last, Compare comp) {
246 using T = iter_value_t<Iterator>;
247 if (first == last) {
248 return _MSTL make_pair(T(), T());
249 }
250 T min_val = *first;
251 T max_val = *first;
252
253 ++first;
254 for (; first != last; ++first) {
255 if (comp(*first, min_val))
256 min_val = *first;
257 else if (!comp(*first, max_val))
258 max_val = *first;
259 }
260 return _MSTL make_pair(min_val, max_val);
261}
