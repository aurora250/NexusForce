381 template <typename Rep, typename Period>
382 future_status wait_for(const duration<Rep, Period>& relative_time) {
383 if (status.load(memory_order_acquire) == status::ready) {
384 return future_status::ready;
385 }
386
387 if (is_deferred_future()) {
388 return future_status::deferred;
389 }
390
391 if (relative_time > relative_time.zero() &&
392 status.load_when_equal_for(status::ready, memory_order_acquire, relative_time)) {
393 complete_async();
394 return future_status::ready;
395 }
396 return future_status::timeout;
397 }

406 template <typename Clock, typename Dur>
407 future_status wait_until(const time_point<Clock, Dur>& absolute_time) {
408 static_assert(is_clock_v<Clock>, "Clock type must be clock_t");
409 if (status.load(memory_order_acquire) == status::ready) {
410 return future_status::ready;
411 }
412
413 if (is_deferred_future()) {
414 return future_status::deferred;
415 }
416
417 if (status.load_when_equal_until(status::ready, memory_order_acquire, absolute_time)) {
418 complete_async();
419 return future_status::ready;
420 }
421 return future_status::timeout;

433 bool did_set = false;

435 call_once(flag, &state_base::do_set, this, _NEFORCE addressof(func), _NEFORCE addressof(did_set));
