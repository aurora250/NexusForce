134 NEFORCE_NODISCARD reference dereference() const noexcept {
135 NEFORCE_DEBUG_VERIFY(current_ && container_, "Attempting to dereference on a null pointer");
136 NEFORCE_DEBUG_VERIFY(first_ <= current_ && current_ < last_, "Attempting to dereference out of boundary");
137 return *current_;

145 void increment() noexcept {
146 NEFORCE_DEBUG_VERIFY(current_ && container_, "Attempting to increment a null pointer");
147 ++current_;
148 if (current_ == last_) {
149 deque_iterator::change_buff(node_ + 1);
150 current_ = first_;

159 void decrement() noexcept {
160 NEFORCE_DEBUG_VERIFY(current_ && container_, "Attempting to decrement a null pointer");
161 if (current_ == first_) {
162 deque_iterator::change_buff(node_ - 1);
163 current_ = last_;
164 }
165 --current_;

174 void advance(difference_type off) noexcept {
175 NEFORCE_DEBUG_VERIFY((current_ && container_) || off == 0, "Attempting to advance a null pointer");
176 const difference_type offset = off + (current_ - first_);
177 if (offset >= 0 && offset < buffer_size) {
178 current_ += off;
179 } else {
180 difference_type node_offset = offset > 0 ?
181 offset / buffer_size :
182 -static_cast<difference_type>((-offset - 1) / buffer_size) - 1;
183 deque_iterator::change_buff(node_ + node_offset);
184 current_ = first_ + (offset - node_offset * buffer_size);
185 }

355 void create_map_and_nodes(const size_type n) {
356 size_type node_nums = n / buffer_size + (n % buffer_size ? 1 : 0);
357 map_size_pair_.value = _NEFORCE max(init_map_size, node_nums + 2);
358
359 try {
360 map_pair_.value = deque::create_map(map_size_pair_.value);
361 } catch (...) {
362 map_pair_.value = nullptr;
363 map_size_pair_.value = 0;
364 throw;
365 }
366
367 map_pointer nstart = map_pair_.value + (map_size_pair_.value - node_nums) / 2;
368 map_pointer nfinish = node_nums == 0 ? nstart : nstart + node_nums - 1;
369
370 try {
371 deque::create_nodes(nstart, nfinish);
372 } catch (...) {
373 deque::destroy_nodes(map_pair_.value, map_pair_.value + map_size_pair_.value - 1);
374 map_pair_.get_base().deallocate(map_pair_.value, map_size_pair_.value);
375 map_pair_.value = nullptr;
376 map_size_pair_.value = 0;
377 throw;

382 start_.current_ = start_.first_;
383 finish_.current_ = n == 0 ?
384 start_.first_ :
385 finish_.first_ + (n % buffer_size ? n % buffer_size : buffer_size);
386 start_.container_ = this;
387 finish_.container_ = this;

501 void reallocate_map(const size_type n, const bool add_at_front) {
502 const size_type begin_left = start_.current_ - start_.first_;
503
504 if (add_at_front && begin_left < n) {
505 const size_t needed = (n - begin_left) / buffer_size + 1;
506
507 if (needed > static_cast<size_type>(start_.node_ - map_pair_.value)) {
508 const size_type new_size = _NEFORCE max(
509 map_size_pair_.value << 1,
510 map_size_pair_.value + needed + init_map_size);
511 map_pointer map = deque::create_map(new_size);
512 const size_type old_buf = finish_.node_ - start_.node_ + 1;
513 const size_type new_buf = needed + old_buf;
514
515 auto begin = map + (new_size - new_buf) / 2;
516 auto mid = begin + needed;
517 auto end = mid + old_buf;
518
519 deque::create_nodes(begin, mid - 1);
520 for (auto begin1 = mid, begin2 = start_.node_;
521 begin1 != end; ++begin1, ++begin2) {
522 *begin1 = *begin2;
523 }
524
525 map_pair_.get_base().deallocate(map_pair_.value, map_size_pair_.value);
526 map_pair_.value = map;
527 map_size_pair_.value = new_size;
528 start_ = iterator(*mid + (start_.current_ - start_.first_), mid, this);
529 finish_ = iterator(*(end - 1) + (finish_.current_ - finish_.first_), end - 1, this);
530 return;
531 }
532
533 deque::create_nodes(start_.node_ - needed, start_.node_ - 1);
534 return;
535 }
536
537 const size_type end_left = finish_.last_ - finish_.current_ - 1;
538
539 if (!add_at_front && end_left < n) {
540
541 const size_type needed = (n - end_left) / buffer_size + 1;
542
543 if (needed > static_cast<size_type>((map_pair_.value + map_size_pair_.value) - finish_.node_ - 1)) {
544 const size_type new_size = _NEFORCE max(
545 map_size_pair_.value << 1,
546 map_size_pair_.value + needed + init_map_size);
547 map_pointer map = deque::create_map(new_size);
548 const size_type old_buf = finish_.node_ - start_.node_ + 1;
549 const size_type new_buf = needed + old_buf;
550
551 auto begin = map + (new_size - new_buf) / 2;
552 auto mid = begin + old_buf;
553 auto end = mid + needed;
554
555 for (auto begin1 = begin, begin2 = start_.node_;
556 begin1 != mid; ++begin1, ++begin2) {
557 *begin1 = *begin2;
558 }
559 deque::create_nodes(mid, end - 1);
560
561 map_pair_.get_base().deallocate(map_pair_.value, map_size_pair_.value);
562 map_pair_.value = map;
563 map_size_pair_.value = new_size;
564 start_ = iterator(*begin + (start_.current_ - start_.first_), begin, this);
565 finish_ = iterator(*(mid - 1) + (finish_.current_ - finish_.first_), mid - 1, this);
566 return;
567 }
568
569 deque::create_nodes(finish_.node_ + 1, finish_.node_ + needed);
570 }

581 template <typename Iterator>
582 void insert_ranges_n(iterator position, Iterator first, Iterator last, size_type n) {
583 difference_type dist_before = position - start_;
584
585 if (dist_before < static_cast<difference_type>(size() / 2)) {
586 deque::reallocate_map(n, true);
587 auto old_start = start_;
588 auto new_start = start_ - n;
589 position = start_ + dist_before;
590
591 try {
592 if (dist_before >= n) {
593 iterator start_n = start_ + n;
594 _NEFORCE uninitialized_copy(start_, start_n, new_start);
595 start_ = new_start;
596 _NEFORCE copy(start_n, position, old_start);
597 _NEFORCE copy(first, last, position - n);
598 } else {
599 auto mid = _NEFORCE next(first, n - dist_before);
600 _NEFORCE uninitialized_copy(first, mid,
601 _NEFORCE uninitialized_copy(start_, position, new_start));
602 start_ = new_start;
603 _NEFORCE copy(mid, last, old_start);
604 }
605 } catch (...) {
606 if (new_start.node_ != start_.node_) {
607 deque::destroy_nodes(new_start.node_, start_.node_ - 1);
608 }
609 throw;
610 }
611 } else {
612 deque::reallocate_map(n, false);
613 auto old_finish = finish_;
614 auto new_finish = finish_ + n;
615 const size_type dist_after = size() - dist_before;
616 position = finish_ - dist_after;
617
618 try {
619 if (dist_after > n) {
620 auto finish_n = finish_ - n;
621 _NEFORCE uninitialized_copy(finish_n, finish_, finish_);
622 finish_ = new_finish;
623 _NEFORCE copy_backward(position, finish_n, old_finish);
624 _NEFORCE copy(first, last, position);
625 } else {
626 auto mid = _NEFORCE next(first, dist_after);
627 _NEFORCE uninitialized_copy(position, finish_,
628 _NEFORCE uninitialized_copy(mid, last, finish_));
629 finish_ = new_finish;
630 _NEFORCE copy(first, mid, position);
631 }
632 } catch (...) {
633 if (new_finish.node_ != finish_.node_) {
634 deque::destroy_nodes(finish_.node_ + 1, new_finish.node_);
635 }
636 throw;
637 }
638 }
