377 NEFORCE_NODISCARD constexpr bitset& operator <<=(const uint32_t pos) noexcept {
378 if (pos >= N) {
379 reset();
380 return *this;
381 }
382 const size_t block_shift = pos / bits_per_block;
383 const size_t bit_shift = pos % bits_per_block;
384
385 if (block_shift > 0) {
386 for (size_t i = block_count - 1; i >= block_shift; --i) {
387 blocks[i] = blocks[i - block_shift];
388 }
389 for (size_t i = 0; i < block_shift; ++i) {
390 blocks[i] = 0;
391 }
392 }
393 if (bit_shift > 0) {
394 for (size_t i = block_count - 1; i > 0; --i) {
395 blocks[i] = (blocks[i] << bit_shift) | (blocks[i - 1] >> (bits_per_block - bit_shift));
396 }
397 blocks[0] <<= bit_shift;
398 }
399 blocks[block_count - 1] &= last_block_mask();
400
401 return *this;
402 }
