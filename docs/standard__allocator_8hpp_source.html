130NEFORCE_ALLOC_OPTIMIZE NEFORCE_CONSTEXPR20 void* __allocate_aux(const alloc_size_t bytes) {
131#ifdef NEFORCE_COMPILER_MSVC
132 if (bytes >= MEMORY_BIG_ALLOC_THRESHHOLD) {
133 const size_t block_size = MEMORY_NO_USER_SIZE + bytes;
134 if (block_size <= bytes) {
135 NEFORCE_THROW_EXCEPTION(memory_exception("invalid block size."));
136 }
137 const auto holder = reinterpret_cast<uintptr_t>(operator new(block_size));
138 NEFORCE_DEBUG_VERIFY(holder != 0, "invalid argument");
139 const auto ptr = reinterpret_cast<void*>((holder + MEMORY_NO_USER_SIZE) & ~(MEMORY_BIG_ALLOC_ALIGN - 1));
140 static_cast<uintptr_t*>(ptr)[-1] = holder;
141#ifdef NEFORCE_STATE_DEBUG
142 static_cast<uintptr_t*>(ptr)[-2] = MEMORY_BIG_ALLOC_SENTINEL;
143#endif
144 return ptr;
145 }
146#endif
147 return operator new(bytes);
148}

230template <size_t Align>
231void __deallocate_aux(void*& ptr, inner::alloc_size_t& bytes) noexcept {
232#ifdef NEFORCE_COMPILER_MSVC
233 if (bytes >= MEMORY_BIG_ALLOC_THRESHHOLD) {
234 bytes += MEMORY_NO_USER_SIZE;
235 const uintptr_t* const user_ptr = static_cast<uintptr_t*>(ptr);
236 const uintptr_t holder = user_ptr[-1];
237 NEFORCE_DEBUG_VERIFY(user_ptr[-2] == MEMORY_BIG_ALLOC_SENTINEL, "invalid sentinel.");
238#ifdef NEFORCE_STATE_DEBUG
239 constexpr uintptr_t min_shift = 2 * sizeof(void*);
240#else
241 constexpr uintptr_t min_shift = sizeof(void*);
242#endif
243 const uintptr_t shift = reinterpret_cast<uintptr_t>(ptr) - holder;
244 NEFORCE_DEBUG_VERIFY(shift >= min_shift && shift <= MEMORY_NO_USER_SIZE, "invalid argument.");
245 ptr = reinterpret_cast<void*>(holder);
246 }
247#endif
248#if defined(NEFORCE_STANDARD_14) && defined(NEFORCE_COMPILER_MSVC)
249 operator delete(ptr, bytes);
