105template <size_t Align>
106MSTL_ALLOC_OPTIMIZE MSTL_CONSTEXPR20 void* __allocate_aux(const alloc_size_t bytes) {
107#ifdef MSTL_COMPILER_MSVC__
108 if (bytes >= MEMORY_BIG_ALLOC_THRESHHOLD) {
109 const size_t block_size = MEMORY_NO_USER_SIZE + bytes;
110 if (block_size <= bytes)
111 throw_exception(memory_exception("invalid block size."));
112 const auto holder = reinterpret_cast<uintptr_t>(operator new(block_size));
113 MSTL_DEBUG_VERIFY(holder != 0, "invalid argument");
114 const auto ptr = reinterpret_cast<void*>(
115 (holder + MEMORY_NO_USER_SIZE) & ~(MEMORY_BIG_ALLOC_ALIGN - 1)); // align the memory address
116 static_cast<uintptr_t*>(ptr)[-1] = holder;
117#ifdef MSTL_STATE_DEBUG__
118 static_cast<uintptr_t*>(ptr)[-2] = MEMORY_BIG_ALLOC_SENTINEL;
119#endif
120 return ptr;
121 }
122#endif
123 return operator new(bytes);
124}
