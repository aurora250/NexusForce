184 bool read_frame() {
185 try {
186 if (!socket_.is_open()) {
187 return false;
188 }
189
190 websocket_frame_header header;
191 ssize_t n = socket_.receive(memory_view<char>(reinterpret_cast<char*>(&header), 1));
192 if (n <= 0) return false;
193
194 if (header.rsv1 || header.rsv2 || header.rsv3) {
195 close(WEBSOCKET_STATUS::PROTOCOL_ERROR, "Reserved bits not zero");
196 return false;
197 }
198
199 uint64_t payload_len = header.payload_len;
200 if (payload_len == 126) {
201 uint16_t net_len;
202 n = socket_.receive(memory_view<char>(reinterpret_cast<char*>(&net_len), 2));
203 if (n != 2) return false;
204 payload_len = ntohs(net_len);
205 } else if (payload_len == 127) {
206 uint64_t net_len;
207 n = socket_.receive(memory_view<char>(reinterpret_cast<char*>(&net_len), 8));
208 if (n != 8) return false;
209 payload_len = ntohll(net_len);
210 }
211
212 constexpr uint64_t MAX_MESSAGE_SIZE = 64 * 1024 * 1024; // 64MB
213 if (payload_len > MAX_MESSAGE_SIZE) {
214 close(WEBSOCKET_STATUS::MESSAGE_TOO_BIG, "Message too large");
215 return false;
216 }
217
218 uint32_t masking_key = 0;
219 if (header.masked) {
220 n = socket_.receive(memory_view<char>(reinterpret_cast<char*>(&masking_key), 4));
221 if (n != 4) return false;
222 }
223
224 string payload;
225 if (payload_len > 0) {
226 payload.resize(payload_len);
227 n = socket_.receive(memory_view<char>(&payload[0], payload_len));
228 if (static_cast<size_t>(n) != payload_len) return false;
229 }
230
231 if (header.masked) {
232 for (size_t i = 0; i < payload_len; ++i) {
233 payload[i] ^= reinterpret_cast<char*>(&masking_key)[i % 4];
234 }
235 }
236
237 websocket_opcode opcode = static_cast<websocket_opcode>(header.opcode);
238
239 if ((opcode == websocket_opcode::CLOSE ||
240 opcode == websocket_opcode::PING ||
241 opcode == websocket_opcode::PONG) && payload_len > 125) {
242 close(WEBSOCKET_STATUS::PROTOCOL_ERROR, "Control frame too large");
243 return false;
244 }
245
246 if (opcode == websocket_opcode::CLOSE) {
247 handle_close_frame(payload);
248 return false;
249 } else if (opcode == websocket_opcode::PING) {
250 queue_frame(build_frame(websocket_opcode::PONG, payload, false));
251 return true;
252 } else if (opcode == websocket_opcode::PONG) {
253 last_pong_time_ = steady_clock::now();
254 return true;
255 } else if (opcode == websocket_opcode::TEXT ||
256 opcode == websocket_opcode::BINARY ||
257 opcode == websocket_opcode::CONTINUATION) {
258 if (on_message_) {
259 auto self = this->shared_from_this();
260 on_message_(payload, move(opcode));
261 }
262 return true;
263 } else {
264 close(WEBSOCKET_STATUS::PROTOCOL_ERROR, "Unknown opcode");
265 return false;
266 }
267 } catch (const exception& e) {
268 // handle
269 return false;
270 }
271 }

281 void handle_close_frame(const string& payload) {
282 WEBSOCKET_STATUS status = WEBSOCKET_STATUS::NORMAL_CLOSURE;
283 string reason;
284
285 if (payload.size() >= 2) {
286 uint16_t code = (static_cast<uint8_t>(payload[0]) << 8) | static_cast<uint8_t>(payload[1]);
287 status = static_cast<WEBSOCKET_STATUS>(code);
288 if (payload.size() > 2) {
289 reason = payload.substr(2);
290 }
291 }
292
293 if (running_) {
294 string close_payload;
295 close_payload.push_back((static_cast<uint16_t>(status) >> 8) & 0xFF);
296 close_payload.push_back(static_cast<uint16_t>(status) & 0xFF);
297 close_payload += reason;
298
299 auto close_frame = build_frame(websocket_opcode::CLOSE, close_payload, false);
300 queue_frame(move(close_frame));
301
302 auto start = steady_clock::now();
303 while (!write_queue_.empty() && steady_clock::now() - start < seconds(1)) {
304 this_thread::sleep_for(milliseconds(10));
305 }
306 }
307
308 if (on_close_) {
309 on_close_(move(status), reason);
310 }
311
312 stop();
313 }

363
