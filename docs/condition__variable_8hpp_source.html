122 cv_status wait_until(unique_lock<mutex>& lock,
123 const time_point<Clock, Dur>& util) {
124 const typename Clock::time_point entry = Clock::now();
125 const auto atime = clock_t::now() + ceil<clock_t::duration>(util - entry);
126
127 if (this->__wait_until_impl(lock, atime) == cv_status::no_timeout) {
128 return cv_status::no_timeout;
129 }
130 if (Clock::now() < util) {
131 return cv_status::no_timeout;
132 }
