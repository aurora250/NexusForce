265 template <typename Clock, typename Dur>
266 cv_status wait_until(unique_lock<mutex>& lock, const time_point<Clock, Dur>& util) {
267 const typename Clock::time_point entry = Clock::now();
268 const auto atime = clock_type::now() + ceil<clock_type::duration>(util - entry);
269
270 if (this->__wait_until_impl(lock, atime) == cv_status::success) {
271 return cv_status::success;
272 }
273 if (Clock::now() < util) {
274 return cv_status::success;
275 }
276 return cv_status::timeout;
277 }
