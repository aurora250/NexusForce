118NEFORCE_CONSTEXPR20 string escape(const string_view str) {
119 string result;
120 result.reserve(str.length() + str.length() / 4);
121
122 for (const char c : str) {
123 switch (c) {
124 case '\"': {
125 result += "\\\"";
126 break;
127 }
128 case '\'': {
129 result += "\\\'";
130 break;
131 }
132 case '\\': {
133 result += "\\\\";
134 break;
135 }
136 case '\b': {
137 result += "\\b";
138 break;
139 }
140 case '\f': {
141 result += "\\f";
142 break;
143 }
144 case '\n': {
145 result += "\\n";
146 break;
147 }
148 case '\r': {
149 result += "\\r";
150 break;
151 }
152 case '\t': {
153 result += "\\t";
154 break;
155 }
156 case '\v': {
157 result += "\\v";
158 break;
159 }
160 default: {
161 if (static_cast<byte_t>(c) < 0x20) {
162 result += "\\u";
163 constexpr char hex[] = "0123456789abcdef";
164 result += "00";
165 result += hex[(c >> 4) & 0x0F];
166 result += hex[c & 0x0F];
167 } else {
168 result += c;
169 }
170 break;
171 }
172 }
173 }
174 return result;
175}

320NEFORCE_NODISCARD NEFORCE_CONSTEXPR20 string address_string(const void* p) {
321 if (p == nullptr) return {"nullptr"};
322
323 const uintptr_t addr_val = reinterpret_cast<uintptr_t>(p);
324 constexpr size_t hex_digit_count = sizeof(void*) * 2;
325 constexpr char hex_digits[] = "0123456789abcdef";
326 uintptr_t mask = _INNER ADDRESS_MASK;
327 int shift = _INNER ADDRESS_SHIFT;
328
329 string result{"0x"};
330 result.reserve(2 + hex_digit_count);
331
332 for (size_t i = 0; i < hex_digit_count; ++i) {
333 const byte_t digit = static_cast<byte_t>((addr_val & mask) >> shift);
334 result += hex_digits[digit];
335 mask >>= 4;
336 shift -= 4;
337 }
338 return result;
339}
