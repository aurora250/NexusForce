56NEFORCE_CONSTEXPR20 void append_utf8_char(basic_string<T>& result, uint32_t cp) {
57 if (cp > 0x10FFFF || _NEFORCE is_high_surrogate(cp) || _NEFORCE is_low_surrogate(cp)) {
58 _INNER __append_utf8_char_aux(result);
59 return;
60 }
61
62 if (cp <= 0x7F) {
63 result.push_back(static_cast<T>(cp));
64 } else if (cp <= 0x7FF) {
65 result.push_back(static_cast<T>(0xC0 | (cp >> 6)));
66 result.push_back(static_cast<T>(0x80 | (cp & 0x3F)));

70 result.push_back(static_cast<T>(0x80 | (cp & 0x3F)));
71 } else {

73 result.push_back(static_cast<T>(0x80 | ((cp >> 12) & 0x3F)));

90constexpr bool decode_utf8_char(const byte_t* data, size_t& i, const size_t len, uint32_t& cp) noexcept {
91 if (i >= len) {
92 cp = 0xFFFD;
93 return false;
94 }
95
96 const byte_t b1 = data[i++];
97 if ((b1 & 0x80) == 0) {
98 cp = b1;
99 return true;
100 } else if ((b1 & 0xE0) == 0xC0) {
101 if (i >= len) return false;
102 const byte_t b2 = data[i++];
103 if ((b2 & 0xC0) != 0x80) return false;
104 cp = ((b1 & 0x1F) << 6) | (b2 & 0x3F);
105 return cp >= 0x80;
106 } else if ((b1 & 0xF0) == 0xE0) {
107 if (i + 1 >= len) return false;
108 const byte_t b2 = data[i++];
109 const byte_t b3 = data[i++];
110 if ((b2 & 0xC0) != 0x80 || (b3 & 0xC0) != 0x80) return false;
111 cp = ((b1 & 0x0F) << 12) | ((b2 & 0x3F) << 6) | (b3 & 0x3F);
112 return cp >= 0x800 && !(cp >= 0xD800 && cp <= 0xDFFF);
113 } else if ((b1 & 0xF8) == 0xF0) {
114 if (i + 2 >= len) return false;
115 const byte_t b2 = data[i++];
116 const byte_t b3 = data[i++];
117 const byte_t b4 = data[i++];
118 if ((b2 & 0xC0) != 0x80 || (b3 & 0xC0) != 0x80 || (b4 & 0xC0) != 0x80) return false;
119 cp = ((b1 & 0x07) << 18) | ((b2 & 0x3F) << 12) | ((b3 & 0x3F) << 6) | (b4 & 0x3F);
120 return cp >= 0x10000 && cp <= 0x10FFFF;
121 }
122
123 cp = 0xFFFD;
124 return false;
125}

140constexpr bool get_utf16_codepoint(const T* obj, size_t index, const size_t len, uint32_t& cp, size_t& consumed) {
141 const auto c1 = static_cast<uint32_t>(obj[index]);
142 consumed = 1;
143
144 if (_NEFORCE is_high_surrogate(c1)) {
145 if (index + 1 < len) {
146 const auto c2 = static_cast<uint32_t>(obj[index + 1]);
147 if (_NEFORCE is_low_surrogate(c2)) {
148 cp = _NEFORCE combine_surrogates(c1, c2);
149 consumed = 2;
150 return true;
151 }
152 }
153 cp = 0xFFFD;
154 return true;
155 } else if (_NEFORCE is_low_surrogate(c1)) {
156 cp = 0xFFFD;
157 return true;
158 }
159
160 cp = c1;
161 return true;

219NEFORCE_CONSTEXPR20 void codepoint_to_utf16(basic_string<T>& result, uint32_t cp) {
220 if (!_INNER is_valid_unicode_codepoint(cp)) {
221 result.push_back(0xFFFD);
222 return;
223 }
224
225 if (cp <= 0xFFFF) {
226 if (cp >= 0xD800 && cp <= 0xDFFF) {
227 result.push_back(0xFFFD);
228 } else {
229 result.push_back(static_cast<T>(cp));
230 }
231 } else {
232 const uint32_t adjusted = cp - 0x10000;
233 const auto high_surrogate = static_cast<T>((adjusted >> 10) + 0xD800);
234 const auto low_surrogate = static_cast<T>((adjusted & 0x3FF) + 0xDC00);
235 result.push_back(high_surrogate);
236 result.push_back(low_surrogate);
237 }
