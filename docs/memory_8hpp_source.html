127MSTL_CONSTEXPR14 void* memory_move(void* dest, const void* src, size_t count) noexcept {
128 if(dest == nullptr || src == nullptr) return nullptr;
129
130 void* res = dest;
131 auto dest_v = static_cast<volatile byte_t*>(dest);
132 auto src_v = static_cast<const volatile byte_t*>(src);
133 if (dest_v < src_v) {
134 while (count--) {
135 *dest_v = *src_v;
136 dest_v = dest_v + 1;
137 src_v = src_v + 1;
138 }
139 } else if (dest_v > src_v) {
140 while (count--) {
141 *(dest_v + count) = *(src_v + count);
142 }
143 }
144 return res;
145}

210MSTL_CONSTEXPR14 void* memory_find_pattern(
211 const void* data, const size_t data_len,
212 const void* pattern, const size_t pattern_len) noexcept {
213 if (data == nullptr || pattern == nullptr ||
214 data_len == 0 || pattern_len == 0 || pattern_len > data_len) {
215 return nullptr;
216 }
217 const auto data_ptr = static_cast<const byte_t*>(data);
218 const auto pattern_ptr = static_cast<const byte_t*>(pattern);
219 const size_t last_possible = data_len - pattern_len + 1;
220
221 for (size_t i = 0; i < last_possible; ++i) {
222 if (data_ptr[i] == pattern_ptr[0]) {
223 bool match = true;
224 for (size_t j = 1; j < pattern_len; ++j) {
225 if (data_ptr[i + j] != pattern_ptr[j]) {
226 match = false;
227 break;
228 }
229 }
230 if (match) {
231 return const_cast<byte_t*>(data_ptr + i);
232 }
233 }
234 }
235 return nullptr;
236}
