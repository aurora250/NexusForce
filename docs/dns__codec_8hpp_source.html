44 static dns_query_result parse_response(cbyte_view data) {
45 if (data.size() < sizeof(dns_header)) {
46 throw_exception(dns_exception::parse_error("Response too short"));
47 }
48
49 dns_query_result result;
50 size_t offset = 0;
51
52 dns_header header;
53 memory_copy(&header, data.data(), sizeof(dns_header));
54 offset += sizeof(dns_header);
55
56 header.id = ntohs(header.id);
57 header.flags = ntohs(header.flags);
58 header.qdcount = ntohs(header.qdcount);
59 header.ancount = ntohs(header.ancount);
60 header.nscount = ntohs(header.nscount);
61 header.arcount = ntohs(header.arcount);
62
63 result.response_code = static_cast<DNS_RESPONSE>(header.flags & 0x000F);
64 result.truncated = (header.flags & 0x0200) != 0;
65 result.recursive_available = (header.flags & 0x0080) != 0;
66
67 for (uint16_t i = 0; i < header.qdcount; ++i) {
68 skip_domain_name(data, offset);
69 offset += 4;
70 }
71
72 result.answers = parse_records(data, offset, header.ancount);
73 result.authorities = parse_records(data, offset, header.nscount);
74 result.additional = parse_records(data, offset, header.arcount);
75 result.id = header.id;
76
77 return result;
78 }

81 static void encode_domain_name(byte_vector& buffer, const string& domain) {
82 size_t start = 0;
83 size_t pos;
84
85 while ((pos = domain.find('.', start)) != string::npos) {
86 size_t len = pos - start;
87 if (len > 63) {
88 throw_exception(dns_exception::parse_error("Label too long"));
89 }
90
91 buffer.push_back(static_cast<byte_t>(len));
92 buffer.insert(buffer.end(),
93 reinterpret_cast<const byte_t*>(domain.data() + start),
94 reinterpret_cast<const byte_t*>(domain.data() + pos));
95 start = pos + 1;
96 }
97
98 if (start < domain.length()) {
99 size_t len = domain.length() - start;
100 if (len > 63) {
101 throw_exception(dns_exception::parse_error("Label too long"));
102 }
103
104 buffer.push_back(static_cast<byte_t>(len));
105 buffer.insert(buffer.end(),
106 reinterpret_cast<const byte_t*>(domain.data() + start),
107 reinterpret_cast<const byte_t*>(domain.data() + domain.length()));
108 }
109
110 buffer.push_back(0);
111 }

113 static string decode_domain_name(cbyte_view data, size_t& offset, int max_jumps = 5) {
114 string name;
115 bool jumped = false;
116 size_t original_offset = offset;
117 int jumps = 0;
118
119 while (offset < data.size()) {
120 uint8_t len = data[offset];
121
122 if ((len & 0xC0) == 0xC0) {
123 if (offset + 1 >= data.size()) {
124 throw_exception(dns_exception::parse_error("Invalid pointer"));
125 }
126
127 if (!jumped) {
128 original_offset = offset + 2;
129 }
130
131 uint16_t pointer = ((len & 0x3F) << 8) | data[offset + 1];
132 offset = pointer;
133 jumped = true;
134
135 if (++jumps > max_jumps) {
136 throw_exception(dns_exception::parse_error("Too many jumps"));
137 }
138 continue;
139 }
140
141 if (len == 0) {
142 offset++;
143 break;
144 }
145
146 if (len > 63) {
147 throw_exception(dns_exception::parse_error("Invalid label length"));
148 }
149
150 if (offset + 1 + len > data.size()) {
151 throw_exception(dns_exception::parse_error("Domain name truncated"));
152 }
153
154 if (!name.empty()) {
155 name += '.';
156 }
157
158 name.append(reinterpret_cast<const char*>(&data[offset + 1]), len);
159 offset += 1 + len;
160 }
161
162 if (jumped) {
163 offset = original_offset;
164 }
165
166 return name;
167 }

197 static dns_record parse_record(cbyte_view data, size_t& offset) {
198 dns_record record;
199
200 // NAME
201 record.name = decode_domain_name(data, offset);
202
203 // TYPE
204 if (offset + 2 > data.size()) {
205 throw_exception(dns_exception::parse_error("Incomplete record"));
206 }
207 uint16_t type_val;
208 memory_copy(&type_val, &data[offset], 2);
209 record.type = static_cast<DNS_RECORD>(ntohs(type_val));
210 offset += 2;
211
212 // CLASS
213 if (offset + 2 > data.size()) {
214 throw_exception(dns_exception::parse_error("Incomplete record"));
215 }
216 uint16_t class_val;
217 memory_copy(&class_val, &data[offset], 2);
218 record.class_type = static_cast<DNS_QUERY>(ntohs(class_val));
219 offset += 2;
220
221 // TTL
222 if (offset + 4 > data.size()) {
223 throw_exception(dns_exception::parse_error("Incomplete record"));
224 }
225 uint32_t ttl_val;
226 memory_copy(&ttl_val, &data[offset], 4);
227 record.ttl = ntohl(ttl_val);
228 offset += 4;
229
230 // RDLENGTH
231 if (offset + 2 > data.size()) {
232 throw_exception(dns_exception::parse_error("Incomplete record"));
233 }
234 uint16_t rdlength;
235 memory_copy(&rdlength, &data[offset], 2);
236 rdlength = ntohs(rdlength);
237 offset += 2;
238
239 // RDATA
240 if (offset + rdlength > data.size()) {
241 throw_exception(dns_exception::parse_error("RDATA truncated"));
242 }
243
244 record.data = parse_rdata(data.view(offset, rdlength), record.type);
245 offset += rdlength;
246
247 return record;
248 }

250 static string parse_rdata(cbyte_view rdata, DNS_RECORD type) {
251 switch (type) {
252 case DNS_RECORD::A: {
253 if (rdata.size() != 4) {
254 throw_exception(dns_exception::parse_error("Invalid A record length"));
255 }
256 char ip[INET_ADDRSTRLEN];
257 if (!inet_ntop(AF_INET, rdata.data(), ip, sizeof(ip))) {
258 throw_exception(dns_exception::parse_error("Failed to parse A record"));
259 }
260 return ip;
261 }
262
263 case DNS_RECORD::AAAA: {
264 if (rdata.size() != 16) {
265 throw_exception(dns_exception::parse_error("Invalid AAAA record length"));
266 }
267 char ip[INET6_ADDRSTRLEN];
268 if (!inet_ntop(AF_INET6, rdata.data(), ip, sizeof(ip))) {
269 throw_exception(dns_exception::parse_error("Failed to parse AAAA record"));
270 }
271 return ip;
272 }
273
274 case DNS_RECORD::CNAME:
275 case DNS_RECORD::NS:
276 case DNS_RECORD::PTR: {
277 size_t offset = 0;
278 return decode_domain_name(rdata, offset);
279 }
280
281 case DNS_RECORD::MX: {
282 if (rdata.size() < 2) {
283 throw_exception(dns_exception::parse_error("Invalid MX record"));
284 }
285 uint16_t preference;
286 memory_copy(&preference, rdata.data(), 2);
287 preference = ntohs(preference);
288
289 size_t offset = 2;
290 string exchange = decode_domain_name(rdata, offset);
291
292 return to_string(preference) + " " + exchange;
293 }
294
295 case DNS_RECORD::TXT: {
296 string result;
297 size_t offset = 0;
298 while (offset < rdata.size()) {
299 uint8_t len = rdata[offset++];
300 if (offset + len > rdata.size()) {
301 throw_exception(dns_exception::parse_error("Invalid TXT record"));
302 }
303 result.append(reinterpret_cast<const char*>(&rdata[offset]), len);
304 offset += len;
305 }
306 return result;
307 }
308
309 default: {
310 string hex;
311 for (auto b : rdata) {
312 hex += format("{#02x}", b);
313 }
314 return hex;
315 }
316 }
317 }
