83 template <typename T>
84 json_builder& value_impl(unique_ptr<T> value) {
85 if (contexts_.empty()) {
86 if (root_) {
87 NEFORCE_THROW_EXCEPTION(json_exception("Multiple root values not allowed"));
88 }
89 root_ = _NEFORCE move(value);
90 } else {
91 const auto& top = contexts_.top();
92 if (top.type == array) {
93 top.array_ptr->add_element(_NEFORCE move(value));
94 } else if (top.type == object) {
95 if (current_key_.empty()) {
96 NEFORCE_THROW_EXCEPTION(json_exception("No key set for value in object"));
97 }
98 top.object_ptr->add_member(current_key_, _NEFORCE move(value));
99 current_key_.clear();
100 }
101 }
102 return *this;
103 }
