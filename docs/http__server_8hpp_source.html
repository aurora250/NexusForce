94 void handle_client(socket_type client_socket) {
95 try {
96 http_request request = this->parse_request(
97 static_cast<tcp_socket*>(&client_socket),
98 session_manager_,
99 cookie_name_,
100 max_header_size_,
101 max_body_size_);
102
103 if (client_socket.is_ssl()) {
104 request.set_header(HTTP_KEY::X_Forwarded_Proto, "https");
105 }
106
107 if (enable_websocket_ && this->try_websocket_upgrade(client_socket, request)) {
108 return;
109 }
110
111 http_session* sess = this->get_or_create_session(request, true, session_manager_, cookie_name_);
112 this->handle_request_with_forward(client_socket, request, sess);
113 } catch (const http_exception& e) {
114 this->send_error_response(
115 static_cast<tcp_socket*>(&client_socket),
116 HTTP_STATUS::S4_BAD_REQUEST,
117 e.what());
118 } catch (const exception& e) {
119 this->send_error_response(
120 static_cast<tcp_socket*>(&client_socket),
121 HTTP_STATUS::S5_INTERNAL_ERROR,
122 e.what());
123 } catch (...) {
124 this->send_error_response(
125 static_cast<tcp_socket*>(&client_socket),
126 HTTP_STATUS::S5_INTERNAL_ERROR,
127 "Unknown internal error");
128 }
129 }

131 bool try_websocket_upgrade(socket_type& client_socket, http_request& request) {
132 const string_view upgrade = request.header("Upgrade");
133 const string_view connection = request.header("Connection");
134
135 if (upgrade != "websocket" || connection.find("Upgrade") == string::npos) {
136 return false;
137 }
138
139 const string_view key = request.header("Sec-WebSocket-Key");
140 if (key.empty()) {
141 return false;
142 }
143
144 string accept = compute_websocket_accept(key);
145
146 http_response upgrade_response;
147 upgrade_response.status = HTTP_STATUS::S1_SWITCH_PROTOCOL;
148 upgrade_response.status_message = "Switching Protocols";
149 upgrade_response.set_header("Upgrade", "websocket");
150 upgrade_response.set_header("Connection", "Upgrade");
151 upgrade_response.set_header("Sec-WebSocket-Accept", move(accept));
152
153 send_response(static_cast<tcp_socket*>(&client_socket), upgrade_response);
154
155 return ws_server_.handle_upgrade(request, _NEFORCE move(client_socket));
156 }

158 void handle_request_with_forward(socket_type& client_socket, http_request& request, http_session* sess) {
159 int forward_count = 0;
160
161 while (forward_count < max_forward_count) {
162 http_response response = router_.handle_request(request);
163
164 if (sess) {
165 add_session_cookie(request, response, sess, cookie_name_);
166 }
167
168 if (!response.forward_path.empty()) {
169 request.path = move(response.forward_path);
170 request.parameters.clear();
171 parse_parameters(request);
172 forward_count++;
173 continue;
174 }
175
176 send_response(static_cast<tcp_socket*>(&client_socket), response);
177 break;
178 }
179
180 if (forward_count >= max_forward_count) {
181 this->send_error_response(
182 static_cast<tcp_socket*>(&client_socket),
183 HTTP_STATUS::S5_INTERNAL_ERROR,
184 "Too many forwards");
185 }
186 }
